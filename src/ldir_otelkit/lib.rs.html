1//! Shared tracing subscriber initialization for LDIR binaries.
2//!
3//! Provides convenience functions for setting up `tracing-subscriber`
4//! with common configurations used across the workspace.
5
6use tracing_subscriber::layer::SubscriberExt;
7use tracing_subscriber::util::SubscriberInitExt;
8
9/// Initialize tracing with a simple `fmt` subscriber and env-filter.
10///
11/// Reads `RUST_LOG` for the filter level (defaults to `info`).
12pub fn init() {
13    tracing_subscriber::fmt()
14        .with_env_filter(
15            tracing_subscriber::EnvFilter::try_from_default_env()
16                .unwrap_or_else(|_| tracing_subscriber::EnvFilter::new("info")),
17        )
18        .init();
19}
20
21/// Initialize tracing with a chrome-trace layer (for flamegraph viewing).
22///
23/// Returns the chrome-layer guard; the caller must keep it alive for the
24/// duration of tracing (it is leaked here for convenience in simple binaries).
25pub fn init_chrome() {
26    let (chrome_layer, _guard) = tracing_chrome::ChromeLayerBuilder::new().build();
27
28    tracing_subscriber::registry()
29        .with(chrome_layer)
30        .with(tracing_subscriber::EnvFilter::new("info"))
31        .init();
32
33    // Leak the guard so the chrome layer stays alive.
34    // In benchmarks the process exits shortly after, so this is fine.
35    std::mem::forget(_guard);
36}
