1//! ldc — LDIR Compiler CLI.
2//!
3//! Compiles documents in multiple formats to PDF, HTML, EPUB, TXT, DOCX, SIR2, or LDIR.
4//!
5//! # Usage
6//!
7//! ```sh
8//! ldc input.md -o output.pdf
9//! ldc input.typ -f html -o output.html
10//! ldc input.adoc -f epub -o output.epub
11//! ldc input.org -f txt  -o output.txt
12//! ldc input.docx -f docx -o output.docx
13//! ldc input.html -f sir2 -o output.sir2
14//! ldc input.md input.typ -o output.html
15//! ```

145fn detect_input_format(path: &Path) -> &'static str {
146    match path.extension().and_then(|e| e.to_str()).unwrap_or("") {
147        "md" | "markdown" => "markdown",
148        "tex" | "latex" => "latex",
149        "typ" | "typst" => "typst",
150        "html" | "htm" => "html",
151        "adoc" | "asciidoc" => "asciidoc",
152        "org" => "org",
153        "docx" => "docx",
154        _ => "unknown",
155    }
156}

158fn detect_format_from_extension(ext: &str) -> &str {
159    match ext {
160        "html" | "htm" => "html",
161        "epub" => "epub",

163        "docx" => "docx",
164        "sir2" => "sir2",
165        "ldir" => "ldir",
166        _ => "pdf",
167    }

170fn parse_to_sir_v2(text: &str, format: &str) -> SIRModuleV2 {
171    match format {
172        "typst" => ldir_typst::parse_typst(text),
173        "html" => ldir_html_reader::parse_html(text),
174        "asciidoc" => ldir_adoc::parse_asciidoc(text),
175        "org" => ldir_org::parse_org(text),
176        "markdown" => {
177            let v1 = ldir_md::parse_markdown(text);
178            ldir_core::compiler::v1_to_v2::convert_v1_to_v2(&v1)
179        }
180        "latex" => {
181            let v1 = ldir_tex::parse_tex(text);
182            ldir_core::compiler::v1_to_v2::convert_v1_to_v2(&v1)
183        }
184        _ => SIRModuleV2::new(),
185    }
186}

188fn parse_docx_to_sir_v2(bytes: &[u8]) -> SIRModuleV2 {
189    ldir_docx_reader::parse_docx(bytes)
190}

192fn parse_input_to_sir_v2(path: &Path) -> Result<SIRModuleV2> {
193    let format = detect_input_format(path);
194    if format == "unknown" {
195        anyhow::bail!(
196            "unsupported input format for '{}'. Supported: .md, .tex, .typ, .html, .htm, .adoc, .org, .docx",
197            path.display()
198        );
199    }
200    if format == "docx" {
201        let bytes =
202            std::fs::read(path).with_context(|| format!("failed to read {}", path.display()))?;
203        Ok(parse_docx_to_sir_v2(&bytes))
204    } else {
205        let text = std::fs::read_to_string(path)
206            .with_context(|| format!("failed to read {}", path.display()))?;
207        Ok(parse_to_sir_v2(&text, format))
208    }
209}

211fn merge_modules(inputs: &[PathBuf]) -> Result<SIRModuleV2> {
212    let mut merged = SIRModuleV2::new();
213    let mut id_offset: u32 = 0;
214    for input in inputs {
215        let mut module = parse_input_to_sir_v2(input)
216            .with_context(|| format!("failed to parse {}", input.display()))?;
217        eprintln!(
218            "[ldir] parsed {} → {} nodes",
219            input.display(),
220            module.body.len()
221        );
222        for node in module.body.iter_mut() {
223            node.id += id_offset;
224            node.parent_id = node.parent_id.map(|p| p + id_offset);
225            node.child_ids = node.child_ids.iter().map(|c| c + id_offset).collect();
226        }
227        let count = module.body.len() as u32;
228        for node in module.body.iter() {
229            merged.body.push(node.clone());
230        }
231        id_offset += count;
232    }
233    Ok(merged)
234}

236fn write_output(module: &SIRModuleV2, output_path: &Path, format: &str) -> Result<()> {
237    match format {
238        "html" => {
239            let html = ldir_html::HtmlRenderer::new().render(module);
240            eprintln!("[ldir] generated HTML ({} bytes)", html.len());
241            std::fs::write(output_path, &html)
242                .with_context(|| format!("failed to write {}", output_path.display()))?;
243        }
244        "epub" => {
245            let epub_bytes = ldir_epub::EpubBuilder::new()
246                .build(module)
247                .map_err(|e| anyhow::anyhow!(e))?;
248            eprintln!("[ldir] generated EPUB ({} bytes)", epub_bytes.len());
249            std::fs::write(output_path, &epub_bytes)
250                .with_context(|| format!("failed to write {}", output_path.display()))?;
251        }
252        "txt" => {
253            let m = module.clone();
254            let text = ldir_txt::TextRenderer::new().render(&m);
255            eprintln!("[ldir] generated text ({} bytes)", text.len());
256            std::fs::write(output_path, &text)
257                .with_context(|| format!("failed to write {}", output_path.display()))?;
258        }
259        "docx" => {
260            let docx_bytes = ldir_docx::DocxBuilder::new()
261                .build(module)
262                .map_err(|e| anyhow::anyhow!(e))?;
263            eprintln!("[ldir] generated DOCX ({} bytes)", docx_bytes.len());
264            std::fs::write(output_path, &docx_bytes)
265                .with_context(|| format!("failed to write {}", output_path.display()))?;
266        }
267        "sir2" => {
268            let bytes = ldir_ir::sir::v2::SIRBinaryWriter::write(module);
269            eprintln!(
270                "[ldir] wrote S-IR v2 binary ({} bytes) → {}",
271                bytes.len(),
272                output_path.display()
273            );
274            std::fs::write(output_path, &bytes)
275                .with_context(|| format!("failed to write {}", output_path.display()))?;
276        }
277        "ldir" => {
278            let text = ldir_ir::sir::v2::module_to_text(module);
279            eprintln!(
280                "[ldir] wrote .ldir text ({} bytes) → {}",
281                text.len(),
282                output_path.display()
283            );
284            std::fs::write(output_path, &text)
285                .with_context(|| format!("failed to write {}", output_path.display()))?;
286        }
287        _ => anyhow::bail!("unsupported output format: {}", format),
288    }
289    eprintln!("[ldir] wrote {}", output_path.display());
290    Ok(())
291}

293/// Try to load a font variant by searching common paths.
294fn find_font_variant(base_path: &str, variant_names: &[&str]) -> Option<Vec<u8>> {
295    if let Some(dir) = std::path::Path::new(base_path).parent() {
296        for name in variant_names {
297            let candidate = dir.join(name);
298            if let Ok(data) = std::fs::read(&candidate)
299                && ttf_parser::Face::parse(&data, 0).is_ok()
300            {
301                eprintln!("[ldir]   variant: {}", candidate.display());
302                return Some(data);
303            }
304        }
305    }
306    None
307}

314fn load_font_variants(primary_path: &str) -> Vec<(u32, Vec<u8>)> {
315    let base_name = std::path::Path::new(primary_path)
316        .file_stem()
317        .and_then(|s| s.to_str())
318        .unwrap_or("");
319
320    let is_dejavu = base_name.contains("DejaVu");
321    let is_liberation = base_name.contains("Liberation");
322    let is_noto = base_name.contains("Noto");
323
324    let variants: Vec<FontVariant> = vec![
325        FontVariant {
326            id: 0,
327            search_names: vec![],
328        },
329        FontVariant {
330            id: 1,
331            search_names: if is_dejavu {
332                vec!["DejaVuSans-Bold.ttf"]
333            } else if is_liberation {
334                vec!["LiberationSans-Bold.ttf"]
335            } else if is_noto {
336                vec!["NotoSans-Bold.ttf"]
337            } else {
338                vec!["-Bold.ttf", "-Bold.otf"]
339            },
340        },
341        FontVariant {
342            id: 2,
343            search_names: if is_dejavu {
344                vec!["DejaVuSans-Oblique.ttf"]
345            } else if is_liberation {
346                vec!["LiberationSans-Italic.ttf"]
347            } else if is_noto {
348                vec!["NotoSans-Italic.ttf"]
349            } else {
350                vec!["-Italic.ttf", "-Oblique.ttf", "-Italic.otf"]
351            },
352        },
353        FontVariant {
354            id: 3,
355            search_names: if is_dejavu {
356                vec!["DejaVuSans-BoldOblique.ttf"]
357            } else if is_liberation {
358                vec!["LiberationSans-BoldItalic.ttf"]
359            } else if is_noto {
360                vec!["NotoSans-BoldItalic.ttf"]
361            } else {
362                vec!["-BoldItalic.ttf", "-BoldOblique.ttf", "-BoldItalic.otf"]
363            },
364        },
365        FontVariant {
366            id: 4,
367            search_names: if is_dejavu {
368                vec!["DejaVuSansMono.ttf", "../DejaVuSansMono/DejaVuSansMono.ttf"]
369            } else if is_liberation {
370                vec!["LiberationMono-Regular.ttf"]
371            } else if is_noto {
372                vec!["NotoSansMono-Regular.ttf"]
373            } else {
374                vec!["Mono-Regular.ttf", "-Mono.ttf", "monospace.ttf"]
375            },
376        },
377    ];
378
379    let mut loaded = Vec::new();
380    for variant in &variants {
381        if variant.id == 0 {
382            continue;
383        }
384        if let Some(data) = find_font_variant(primary_path, &variant.search_names) {
385            loaded.push((variant.id, data));
386        }
387    }
388    loaded
389}

391/// Resolve font data: first by family name via database, then by file path,
392/// then by scanning common system font directories.
393fn resolve_font(cli: &Cli, font_db: &FontDatabase) -> Option<(Arc<Vec<u8>>, String)> {
394    if let Some(ref path) = cli.font_path {
395        let data = std::fs::read(path)
396            .with_context(|| format!("failed to read font: {}", path.display()))
397            .ok()?;
398        ttf_parser::Face::parse(&data, 0)
399            .with_context(|| format!("invalid font file: {}", path.display()))
400            .ok()?;
401        eprintln!("[ldir] using font: {}", path.display());
402        let path_str = path.to_string_lossy().to_string();
403        return Some((Arc::new(data), path_str));
404    }
405
406    if let Some(ref family) = cli.font
407        && let Some(id) = font_db.query(family)
408        && let Some(data) = font_db.face_data(id)
409    {
410        eprintln!("[ldir] using font family: {} (from system fonts)", family);
411        return Some((data, format!("system:{}", family)));
412    }
413
414    if let Some(ref family) = cli.font {
415        eprintln!(
416            "[ldir] warning: font family '{}' not found in system fonts",
417            family
418        );
419    }
420
421    let candidates = [
422        "/usr/share/fonts/TTF/DejaVuSans.ttf",
423        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
424        "/usr/share/fonts/TTF/LiberationSans-Regular.ttf",
425        "/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf",
426        "/usr/share/fonts/noto/NotoSans-Regular.ttf",
427        "/usr/share/fonts/opentype/noto/NotoSans-Regular.otf",
428        "/usr/share/fonts/truetype/noto/NotoSans-Regular.ttf",
429    ];
430    for path in &candidates {
431        if let Ok(data) = std::fs::read(path)
432            && ttf_parser::Face::parse(&data, 0).is_ok()
433        {
434            eprintln!("[ldir] using system font: {}", path);
435            return Some((Arc::new(data), path.to_string()));
436        }
437    }
438
439    if let Some(id) = font_db.query("DejaVu Sans")
440        && let Some(data) = font_db.face_data(id)
441    {
442        eprintln!("[ldir] using font family: DejaVu Sans (auto-detected)");
443        return Some((data, "system:DejaVu Sans".to_string()));
444    }
445
446    None
447}

449/// Load font variant data using the font database for style variants.
450fn load_font_variants_from_db(
451    cli: &Cli,
452    font_db: &FontDatabase,
453    primary_family: &str,
454) -> Vec<(u32, Arc<Vec<u8>>)> {
455    let mut variants = Vec::new();
456
457    let family = cli.font.as_deref().unwrap_or(primary_family);
458
459    let bold_styles = [
460        (fontdb::Weight::BOLD, fontdb::Style::Normal),
461        (fontdb::Weight::BOLD, fontdb::Style::Italic),
462    ];
463    if let Some((_, data)) = try_query_style(font_db, family, &bold_styles) {
464        eprintln!("[ldir] loaded bold font variant from database");
465        variants.push((1, data));
466    }
467
468    let italic_styles = [
469        (fontdb::Weight::NORMAL, fontdb::Style::Italic),
470        (fontdb::Weight::NORMAL, fontdb::Style::Oblique),
471    ];
472    if let Some((_, data)) = try_query_style(font_db, family, &italic_styles) {
473        eprintln!("[ldir] loaded italic font variant from database");
474        variants.push((2, data));
475    }
476
477    let bold_italic_styles = [
478        (fontdb::Weight::BOLD, fontdb::Style::Italic),
479        (fontdb::Weight::BOLD, fontdb::Style::Oblique),
480    ];
481    if let Some((_, data)) = try_query_style(font_db, family, &bold_italic_styles) {
482        eprintln!("[ldir] loaded bold italic font variant from database");
483        variants.push((3, data));
484    }
485
486    let mono_family = cli.font_mono.as_deref().unwrap_or("");
487    let mono_id = if !mono_family.is_empty() {
488        font_db.query(mono_family)
489    } else {
490        font_db.query_monospace()
491    };
492    if let Some(id) = mono_id
493        && let Some(data) = font_db.face_data(id)
494    {
495        eprintln!("[ldir] loaded monospace font variant from database");
496        variants.push((4, data));
497    }
498
499    variants
500}

502fn try_query_style(
503    font_db: &FontDatabase,
504    family: &str,
505    styles: &[(fontdb::Weight, fontdb::Style)],
506) -> Option<(fontdb::ID, Arc<Vec<u8>>)> {
507    for &(weight, style) in styles {
508        if let Some(id) = font_db.query_family_style(family, weight, style)
509            && let Some(data) = font_db.face_data(id)
510        {
511            return Some((id, data));
512        }
513    }
514    None
515}

517fn list_system_fonts(font_db: &FontDatabase) {
518    let mut families: Vec<(String, Vec<String>)> = Vec::new();
519
520    for face in font_db.face_info_iter() {
521        if let Some(name_tuple) = face.families.first() {
522            let name_str = &name_tuple.0;
523            if let Some(entry) = families.iter_mut().find(|(f, _)| f == name_str) {
524                entry.1.push(format!("{:?}", face.style));
525            } else {
526                families.push((name_str.clone(), vec![format!("{:?}", face.style)]));
527            }
528        }
529    }
530
531    families.sort_by(|a, b| a.0.cmp(&b.0));
532
533    eprintln!(
534        "Available system fonts ({} faces, {} families):",
535        font_db.face_count(),
536        families.len()
537    );
538    for (i, (family, styles)) in families.iter().enumerate() {
539        let styles_str = styles.join(", ");
540        eprintln!("  {:>3}. {} ({})", i + 1, family, styles_str);
541    }
542}

544fn emit_pdf(
545    gir_doc: &GIRDocument,
546    cli: &Cli,
547    font_data: &Option<Arc<Vec<u8>>>,
548    variant_fonts: &[(u32, Arc<Vec<u8>>)],
549    output: &Path,
550) -> Result<()> {
551    let conformance: ldir_pdf::conformance::PdfConformance =
552        cli.pdfa_level.parse().map_err(|e| anyhow::anyhow!("{e}"))?;
553
554    let options = ldir_pdf::converter::PdfOptions {
555        title: cli.title.clone(),
556        author: cli.author.clone(),
557        subject: cli.subject.clone(),
558        header_left: cli.header_left.clone(),
559        header_right: cli.header_right.clone(),
560        footer_left: cli.footer_left.clone(),
561        footer_right: cli.footer_right.clone(),
562        conformance,
563        ..Default::default()
564    };
565
566    let total_fonts = 1 + variant_fonts.len();
567
568    let pdf_bytes = if let Some(primary) = font_data {
569        let mut font_data_vec: Vec<Arc<Vec<u8>>> = vec![primary.clone()];
570        font_data_vec.resize(5, primary.clone());
571        for (id, data) in variant_fonts {
572            let idx = *id as usize;
573            if idx < font_data_vec.len() {
574                font_data_vec[idx] = data.clone();
575            }
576        }
577
578        let font_faces: Vec<ldir_pdf::font::FontFace> = font_data_vec
579            .iter()
580            .filter_map(|data| ldir_pdf::font::FontFace::from_bytes(data).ok())
581            .collect();
582
583        ldir_pdf::converter::gir_to_pdf_with_fonts_and_options(gir_doc, &font_faces, &options)
584    } else {
585        ldir_pdf::converter::gir_to_pdf_with_fonts_and_options(gir_doc, &[], &options)
586    };
587    eprintln!(
588        "[ldir] generated PDF ({} bytes){} ({} fonts)",
589        pdf_bytes.len(),
590        if font_data.is_some() {
591            " [embedded font]"
592        } else {
593            " [fallback Helvetica]"
594        },
595        total_fonts,
596    );
597
598    std::fs::write(output, &pdf_bytes)
599        .with_context(|| format!("failed to write {}", output.display()))?;
600    eprintln!("[ldir] wrote {}", output.display());
601    Ok(())
602}

604fn main() -> Result<()> {
605    let cli = Cli::parse();
606
607    if cli.inputs.is_empty() && !cli.list_fonts {
608        anyhow::bail!("no input files specified. Usage: ldc <INPUT> [INPUTS...] -o <OUTPUT>");
609    }
610
611    let mut font_db = FontDatabase::new();
612    let loaded = font_db.load_system_fonts();
613    if loaded > 0 {
614        eprintln!("[ldir] loaded {} system fonts", loaded);
615    }
616
617    if cli.list_fonts {
618        list_system_fonts(&font_db);
619        return Ok(());
620    }
621
622    let first_input = &cli.inputs[0];
623
624    let effective_format = if let Some(ref output) = cli.output {
625        let out_ext = output.extension().and_then(|e| e.to_str()).unwrap_or("");
626        detect_format_from_extension(out_ext)
627    } else {
628        &cli.format
629    };
630
631    let output = match cli.output {
632        Some(ref p) => p.clone(),
633        None => {
634            let stem = first_input
635                .file_stem()
636                .context("cannot determine output filename")?;
637            let ext = match effective_format {
638                "html" => "html",
639                "epub" => "epub",
640                "txt" => "txt",
641                "docx" => "docx",
642                "sir2" => "sir2",
643                "ldir" => "ldir",
644                "gir" => "gir",
645                _ => "pdf",
646            };
647            first_input.with_file_name(stem).with_extension(ext)
648        }
649    };
650
651    if V2_FORMATS.contains(&effective_format) {
652        let mut module = merge_modules(&cli.inputs)?;
653        if let Some(ref title) = cli.title {
654            module.metadata.title = Some(title.clone());
655        }
656        if let Some(ref author) = cli.author {
657            module.metadata.author = Some(author.clone());
658        }
659        eprintln!("[ldir] merged → S-IR v2 ({} nodes)", module.body.len());
660        return write_output(&module, &output, effective_format);
661    }
662
663    let font_info = resolve_font(&cli, &font_db);
664    let font_data = font_info.as_ref().map(|(arc, _)| arc.clone());
665    let font_path_str = font_info.as_ref().map(|(_, s)| s.clone());
666    if font_data.is_none() {
667        eprintln!("[ldir] warning: no font found, using ASCII monospace stub");
668    }
669
670    let primary_family = cli.font.as_deref().unwrap_or("DejaVu Sans");
671    let mut variant_fonts: Vec<(u32, Arc<Vec<u8>>)> =
672        load_font_variants_from_db(&cli, &font_db, primary_family);
673
674    if let Some(ref path_str) = font_path_str
675        && !path_str.starts_with("system:")
676    {
677        for (id, data) in load_font_variants(path_str) {
678            eprintln!(
679                "[ldir] loaded font variant (id={}): {} bytes",
680                id,
681                data.len()
682            );
683            variant_fonts.push((id, Arc::new(data)));
684        }
685    }
686
687    let margin_pt = (cli.margin * 72.0) as i32;
688    let page_size_name = cli.page_size.as_deref();
689    let page_dims = if let (Some(w), Some(h)) = (cli.page_width, cli.page_height) {
690        Some((w as i32, h as i32))
691    } else {
692        None
693    };
694    let mut module = merge_modules(&cli.inputs)?;
695    if let Some(ref title) = cli.title {
696        module.metadata.title = Some(title.clone());
697    }
698    if let Some(ref author) = cli.author {
699        module.metadata.author = Some(author.clone());
700    }
701    eprintln!("[ldir] merged → S-IR v2 ({} nodes)", module.body.len());
702
703    let (pw, ph) = if let Some((w, h)) = page_dims {
704        (w, h)
705    } else if let Some(name) = page_size_name {
706        ldir_core::compiler::context::parse_page_size(name).unwrap_or((
707            ldir_core::compiler::context::DEFAULT_PAGE_WIDTH_PT,
708            ldir_core::compiler::context::DEFAULT_PAGE_HEIGHT_PT,
709        ))
710    } else {
711        (
712            ldir_core::compiler::context::DEFAULT_PAGE_WIDTH_PT,
713            ldir_core::compiler::context::DEFAULT_PAGE_HEIGHT_PT,
714        )
715    };
716
717    let mut ctx = ldir_core::compiler::context::CompileContext::with_font_margins_and_page(
718        font_data.clone(),
719        margin_pt,
720        margin_pt,
721        margin_pt,
722        margin_pt,
723        pw,
724        ph,
725    );
726    ctx.drop_caps_enabled = cli.drop_caps;
727    for (id, data) in &variant_fonts {
728        ctx.set_font_variant(*id as usize, Some(data.clone()));
729    }
730    ctx.font_db = Some(Arc::new(font_db));
731    ctx.font_family = cli
732        .font
733        .clone()
734        .unwrap_or_else(|| "DejaVu Sans".to_string());
735    ctx.font_mono_family = cli.font_mono.clone().unwrap_or_default();
736
737    let gir_doc = if cli.lir {
738        use ldir_core::compile_sir_to_lir;
739        use ldir_pdf::lir_render::render_lir_to_gir;
740
741        eprintln!("[ldir] compiling via L-IR pipeline (S-IR → L-IR → G-IR)");
742        let lir_doc = compile_sir_to_lir(&module, &ctx)
743            .map_err(|e| anyhow::anyhow!("L-IR compilation failed: {e}"))?;
744        eprintln!("[ldir] L-IR layout → {} pages", lir_doc.pages.len());
745        let gir = render_lir_to_gir(&lir_doc);
746        eprintln!("[ldir] L-IR → G-IR render complete");
747        gir
748    } else if let Some(ref bib_path) = cli.bibliography {
749        let bib_content = std::fs::read_to_string(bib_path)
750            .with_context(|| format!("failed to read bibliography: {}", bib_path.display()))?;
751        let bib_entries = ldir_core::compiler::bibtex::parse_bib(&bib_content)
752            .map_err(|e| anyhow::anyhow!("BibTeX parse error: {}", e))?;
753        eprintln!(
754            "[ldir] loaded bibliography: {} entries from {}",
755            bib_entries.len(),
756            bib_path.display()
757        );
758        ldir_core::compiler::v2_compile::compile_v2_document_with_bib(
759            &module,
760            &mut ctx,
761            &bib_entries,
762        )
763        .map_err(|e| anyhow::anyhow!("v2 compilation failed: {e}"))?
764    } else {
765        ldir_core::compiler::v2_compile::compile_v2_document(&module, &mut ctx)
766            .map_err(|e| anyhow::anyhow!("v2 compilation failed: {e}"))?
767    };
768    eprintln!("[ldir] compiled → {} pages", gir_doc.page_count());
769
770    if let Err(errors) = ldir_core::verifier::check_gir(&gir_doc) {
771        eprintln!("[ldir] G-IR verification warnings:");
772        for err in &errors {
773            eprintln!("  - {}", err);
774        }
775    }
776
777    match effective_format {
778        "pdf" => emit_pdf(&gir_doc, &cli, &font_data, &variant_fonts, &output)?,
779        "gir" => {
780            let bytes = ldir_core::emitter::binary::emit_gir(&gir_doc);
781            std::fs::write(&output, &bytes)
782                .with_context(|| format!("failed to write {}", output.display()))?;
783            eprintln!(
784                "[ldir] wrote G-IR binary ({} bytes) → {}",
785                bytes.len(),
786                output.display()
787            );
788        }
789        "sir" => {
790            let bytes = ldir_ir::sir::v2::serialize_module(&module);
791            std::fs::write(&output, &bytes)
792                .with_context(|| format!("failed to write {}", output.display()))?;
793            eprintln!(
794                "[ldir] wrote S-IR v2 ({} bytes) → {}",
795                bytes.len(),
796                output.display()
797            );
798        }
799        _ => anyhow::bail!("unsupported output format: {}", effective_format),
800    }
801
802    Ok(())
803}
