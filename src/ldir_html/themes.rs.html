629pub fn slugify(text: &str) -> String {
630    let mut slug = String::with_capacity(text.len());
631    for c in text.chars() {
632        match c {
633            'a'..='z' | '0'..='9' => slug.push(c),
634            'A'..='Z' => slug.push(c.to_ascii_lowercase()),
635            ' ' | '-' => slug.push('-'),
636            _ => {}
637        }
638    }
639    let trimmed = slug.trim_matches('-');
640    if trimmed.is_empty() {
641        "heading".to_string()
642    } else {
643        trimmed.to_string()
644    }
645}

651pub fn generate_toc(headings: &[(u32, String)], max_depth: u32) -> String {
652    let mut html = String::from("<nav class=\"toc\">\n<h2>Table of Contents</h2>\n<ul>\n");
653    let mut current_depth = 0u32;
654
655    for &(level, ref text) in headings {
656        if level > max_depth {
657            continue;
658        }
659        while current_depth < level {
660            html.push_str("<ul>\n");
661            current_depth += 1;
662        }
663        while current_depth > level {
664            html.push_str("</ul>\n");
665            current_depth -= 1;
666        }
667        let anchor = slugify(text);
668        html.push_str(&format!(
669            "<li><a href=\"#{}\">{}</a></li>\n",
670            escape_html(&anchor),
671            escape_html(text)
672        ));
673    }
674
675    while current_depth > 0 {
676        html.push_str("</ul>\n");
677        current_depth -= 1;
678    }
679
680    html.push_str("</ul>\n</nav>\n");
681    html
682}

684fn escape_html(s: &str) -> String {
685    let mut out = String::with_capacity(s.len());
686    for c in s.chars() {
687        match c {
688            '&' => out.push_str("&amp;"),
689            '<' => out.push_str("&lt;"),
690            '>' => out.push_str("&gt;"),
