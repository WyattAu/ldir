1use clap::Parser;
2use std::fs;
3use std::path::PathBuf;
4
5/// LDIR IR Disassembler — converts binary S-IR to human-readable .ldir text.
6#[derive(Parser)]
7#[command(name = "ldir-dis", version, about)]
8struct Cli {
9    /// Input file (binary .sir2 or .ldir text file)
10    input: PathBuf,
11
12    /// Output file (default: stdout)
13    #[arg(short, long)]
14    output: Option<PathBuf>,
15
16    /// Output format
17    #[arg(short = 'f', long, value_name = "FORMAT", default_value = "text")]
18    format: String,
19}
20
21fn main() {
22    let cli = Cli::parse();
23
24    let bytes = fs::read(&cli.input).unwrap_or_else(|e| {
25        eprintln!("[ldir-dis] Error reading {}: {}", cli.input.display(), e);
26        std::process::exit(1);
27    });
28
29    let module = match ldir_ir::sir::v2::deserialize_module(&bytes) {
30        Ok(m) => m,
31        Err(_) => {
32            let text = String::from_utf8(bytes).unwrap_or_else(|e| {
33                eprintln!(
34                    "[ldir-dis] Error: not valid binary S-IR or UTF-8 text: {}",
35                    e
36                );
37                std::process::exit(1);
38            });
39            if let Some(ref out) = cli.output {
40                fs::write(out, &text).unwrap_or_else(|e| {
41                    eprintln!("[ldir-dis] Error writing {}: {}", out.display(), e);
42                    std::process::exit(1);
43                });
44            } else {
45                print!("{}", text);
46            }
47            return;
48        }
49    };
50
51    let output = match cli.format.as_str() {
52        "text" => ldir_ir::sir::v2::module_to_text(&module),
53        "json" => serde_json::to_string_pretty(&module).unwrap_or_else(|e| {
54            eprintln!("[ldir-dis] Error serializing to JSON: {}", e);
55            std::process::exit(1);
56        }),
57        other => {
58            eprintln!(
59                "[ldir-dis] Unknown format: {} (use 'text' or 'json')",
60                other
61            );
62            std::process::exit(1);
63        }
64    };
65
66    match cli.output {
67        Some(ref out) => {
68            fs::write(out, &output).unwrap_or_else(|e| {
69                eprintln!("[ldir-dis] Error writing {}: {}", out.display(), e);
70                std::process::exit(1);
71            });
72            eprintln!(
73                "[ldir-dis] Wrote {} bytes to {}",
74                output.len(),
75                out.display()
76            );
77        }
78        None => {
79            print!("{}", output);
80        }
81    }
82}
