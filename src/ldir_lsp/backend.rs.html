81    async fn did_open(&self, params: DidOpenTextDocumentParams) {
82        let doc = params.text_document;
83        let uri = doc.uri.clone();
84        let uri_str = uri.to_string();
85        let diag = diagnostics::compute_diagnostics(&doc.text, &uri);
86        let state = DocumentState {
87            text: doc.text,
88            uri: doc.uri,
89        };
90        {
91            let mut docs = self.documents.write().await;
92            docs.insert(uri_str, state);
93        }
94        self.client.publish_diagnostics(uri, diag, None).await;
95    }
96
97    async fn did_change(&self, params: DidChangeTextDocumentParams) {
98        let uri = params.text_document.uri;
99        let uri_str = uri.to_string();
100        let text = {
101            let mut docs = self.documents.write().await;
102            if let Some(state) = docs.get_mut(&uri_str) {
103                for change in params.content_changes {
104                    state.text = change.text;
105                }
106                Some(state.text.clone())
107            } else {
108                None
109            }
110        };
111
112        let Some(text) = text else {
113            return;
114        };
115
116        let diag = diagnostics::compute_diagnostics(&text, &uri);
117        self.preview.trigger(&text, &uri_str).await;
118        self.client.publish_diagnostics(uri, diag, None).await;
119    }

131    async fn hover(&self, params: HoverParams) -> Result<Option<Hover>> {
132        let docs = self.documents.read().await;
133        let uri = &params.text_document_position_params.text_document.uri;
134        let pos = params.text_document_position_params.position;
135        let Some(state) = docs.get(&uri.to_string()) else {
136            return Ok(None);
137        };
138        let syms = symbols::extract_symbols(&state.text, uri);
139        for sym in &syms {
140            let r = sym.range;
141            if pos.line >= r.start.line
142                && pos.line <= r.end.line
143                && pos.character >= r.start.character
144                && pos.character <= r.end.character
145            {
146                let name = sym.name.clone();
147                let hover_text = match &sym.detail {
148                    Some(d) => format!("{name}\n\n{d}"),
149                    None => name,
150                };
151                return Ok(Some(Hover {
152                    contents: HoverContents::Scalar(MarkedString::String(hover_text)),
153                    range: None,
154                }));
155            }
156        }
157        Ok(None)
158    }

160    async fn goto_definition(
161        &self,
162        params: GotoDefinitionParams,
163    ) -> Result<Option<GotoDefinitionResponse>> {
164        let docs = self.documents.read().await;
165        let uri = &params.text_document_position_params.text_document.uri;
166        let pos = params.text_document_position_params.position;
167        let Some(state) = docs.get(&uri.to_string()) else {
168            return Ok(None);
169        };
170        let ext = crate::detect_extension(uri.path());
171        let Some(line_text) = state.text.lines().nth(pos.line as usize) else {
172            return Ok(None);
173        };
174        match ext {
175            "tex" => {
176                if let Some(label) = extract_ref_label(line_text) {
177                    let pattern = format!("\\label{{{label}}}");
178                    for (li, l) in state.text.lines().enumerate() {
179                        if let Some(col) = l.find(&pattern) {
180                            let start = col as u32;
181                            let end = start + pattern.len() as u32;
182                            return Ok(Some(GotoDefinitionResponse::Scalar(Location {
183                                uri: uri.clone(),
184                                range: Range {
185                                    start: Position {
186                                        line: li as u32,
187                                        character: start,
188                                    },
189                                    end: Position {
190                                        line: li as u32,
191                                        character: end,
192                                    },
193                                },
194                            })));
195                        }
196                    }
197                }
198            }
199            "md" => {
200                if let Some(target) = extract_md_link_target(line_text) {
201                    for (li, l) in state.text.lines().enumerate() {
202                        if l.contains(&format!("# {target}"))
203                            || l.contains(&format!("## {target}"))
204                            || l.contains(&format!("### {target}"))
205                            || l.contains(&format!("#{target}"))
206                        {
207                            return Ok(Some(GotoDefinitionResponse::Scalar(Location {
208                                uri: uri.clone(),
209                                range: Range {
210                                    start: Position {
211                                        line: li as u32,
212                                        character: 0,
213                                    },
214                                    end: Position {
215                                        line: li as u32,
216                                        character: l.len() as u32,
217                                    },
218                                },
219                            })));
220                        }
221                    }
222                }
223            }
224            _ => {}
225        }
226        Ok(None)
227    }

229    async fn references(&self, params: ReferenceParams) -> Result<Option<Vec<Location>>> {
230        let docs = self.documents.read().await;
231        let uri = &params.text_document_position.text_document.uri;
232        let pos = params.text_document_position.position;
233        let Some(state) = docs.get(&uri.to_string()) else {
234            return Ok(None);
235        };
236        let ext = crate::detect_extension(uri.path());
237        let Some(line_text) = state.text.lines().nth(pos.line as usize) else {
238            return Ok(None);
239        };
240        match ext {
241            "tex" => {
242                if let Some(label) = extract_label_name(line_text) {
243                    let ref_cmds = [
244                        "\\ref{",
245                        "\\eqref{",
246                        "\\cref{",
247                        "\\Cref{",
248                        "\\autoref{",
249                        "\\cite{",
250                    ];
251                    let mut locations = Vec::new();
252                    for (doc_uri, doc_state) in docs.iter() {
253                        if let Ok(doc_url) = Url::parse(doc_uri) {
254                            for (li, line) in doc_state.text.lines().enumerate() {
255                                for cmd in &ref_cmds {
256                                    let pattern = format!("{cmd}{label}}}");
257                                    if let Some(col) = line.find(&pattern) {
258                                        let start = col as u32;
259                                        let end = start + pattern.len() as u32;
260                                        locations.push(Location {
261                                            uri: doc_url.clone(),
262                                            range: Range {
263                                                start: Position {
264                                                    line: li as u32,
265                                                    character: start,
266                                                },
267                                                end: Position {
268                                                    line: li as u32,
269                                                    character: end,
270                                                },
271                                            },
272                                        });
273                                        break;
274                                    }
275                                }
276                            }
277                        }
278                    }
279                    return Ok(Some(locations));
280                }
281            }
282            "md" => {
283                if let Some(heading) = extract_md_heading_text(line_text) {
284                    let mut locations = Vec::new();
285                    let wiki_link = format!("[[{heading}]]");
286                    for (doc_uri, doc_state) in docs.iter() {
287                        if let Ok(doc_url) = Url::parse(doc_uri) {
288                            for (li, line) in doc_state.text.lines().enumerate() {
289                                if let Some(col) = line.find(&wiki_link) {
290                                    let start = col as u32;
291                                    let end = start + wiki_link.len() as u32;
292                                    locations.push(Location {
293                                        uri: doc_url.clone(),
294                                        range: Range {
295                                            start: Position {
296                                                line: li as u32,
297                                                character: start,
298                                            },
299                                            end: Position {
300                                                line: li as u32,
301                                                character: end,
302                                            },
303                                        },
304                                    });
305                                }
306                            }
307                        }
308                    }
309                    return Ok(Some(locations));
310                }
311            }
312            _ => {}
313        }
314        Ok(None)
315    }

422    async fn code_action(&self, params: CodeActionParams) -> Result<Option<CodeActionResponse>> {
423        let uri = params.text_document.uri;
424        let mut actions: Vec<CodeActionOrCommand> = Vec::new();
425
426        for diagnostic in &params.context.diagnostics {
427            let (title, new_text) = if diagnostic.message.contains("Unclosed link") {
428                ("Insert closing parenthesis", ")")
429            } else if diagnostic.message.contains("Unclosed brace") {
430                ("Insert closing brace", "}")
431            } else {
432                continue;
433            };
434
435            let end = diagnostic.range.end;
436            let text_edit = TextEdit {
437                range: Range {
438                    start: Position {
439                        line: end.line,
440                        character: end.character,
441                    },
442                    end: Position {
443                        line: end.line,
444                        character: end.character,
445                    },
446                },
447                new_text: new_text.to_string(),
448            };
449
450            let mut changes = HashMap::new();
451            changes.insert(uri.clone(), vec![text_edit]);
452
453            actions.push(CodeActionOrCommand::CodeAction(CodeAction {
454                title: title.to_string(),
455                kind: Some(CodeActionKind::QUICKFIX),
456                diagnostics: Some(vec![diagnostic.clone()]),
457                edit: Some(WorkspaceEdit {
458                    changes: Some(changes),
459                    ..Default::default()
460                }),
461                is_preferred: Some(true),
462                command: None,
463                disabled: None,
464                data: None,
465            }));
466        }
467
468        if actions.is_empty() {
469            Ok(None)
470        } else {
471            Ok(Some(actions))
472        }
473    }

475    async fn rename(&self, params: RenameParams) -> Result<Option<WorkspaceEdit>> {
476        let docs = self.documents.read().await;
477        let uri = &params.text_document_position.text_document.uri;
478        let pos = params.text_document_position.position;
479        let new_name = params.new_name;
480        let Some(state) = docs.get(&uri.to_string()) else {
481            return Ok(None);
482        };
483        let ext = crate::detect_extension(uri.path());
484        if ext != "tex" {
485            return Ok(None);
486        }
487        let Some(line_text) = state.text.lines().nth(pos.line as usize) else {
488            return Ok(None);
489        };
490        let prefixes = [
491            "\\label{",
492            "\\ref{",
493            "\\eqref{",
494            "\\cref{",
495            "\\Cref{",
496            "\\autoref{",
497            "\\cite{",
498        ];
499        let Some((label, _, _)) = find_key_at_position(line_text, pos.character, &prefixes) else {
500            return Ok(None);
501        };
502        let mut changes: HashMap<Url, Vec<TextEdit>> = HashMap::new();
503        for (doc_uri_str, doc_state) in docs.iter() {
504            let Ok(doc_url) = Url::parse(doc_uri_str) else {
505                continue;
506            };
507            let mut edits = Vec::new();
508            for (li, line) in doc_state.text.lines().enumerate() {
509                for &cmd in &prefixes {
510                    let pattern = format!("{cmd}{label}}}");
511                    let mut offset = 0;
512                    while let Some(idx) = line[offset..].find(&pattern) {
513                        let abs_col = offset + idx;
514                        let key_start = abs_col + cmd.len();
515                        let key_end = abs_col + pattern.len() - 1;
516                        edits.push(TextEdit {
517                            range: Range {
518                                start: Position {
519                                    line: li as u32,
520                                    character: key_start as u32,
521                                },
522                                end: Position {
523                                    line: li as u32,
524                                    character: key_end as u32,
525                                },
526                            },
527                            new_text: new_name.clone(),
528                        });
529                        offset = abs_col + 1;
530                    }
531                }
532            }
533            if !edits.is_empty() {
534                changes.insert(doc_url, edits);
535            }
536        }
537        Ok(Some(WorkspaceEdit {
538            changes: Some(changes),
539            ..Default::default()
540        }))
541    }

544fn extract_ref_label(line: &str) -> Option<String> {
545    for prefix in ["\\ref{", "\\eqref{", "\\cref{", "\\Cref{", "\\autoref{"] {
546        if let Some(start) = line.find(prefix) {
547            let rest = &line[start + prefix.len()..];
548            if let Some(end) = rest.find('}') {
549                let label = rest[..end].trim().to_string();
550                if !label.is_empty() {
551                    return Some(label);
552                }
553            }
554        }
555    }
556    None
557}

559fn extract_md_link_target(line: &str) -> Option<String> {
560    if let Some(start) = line.find("[[") {
561        let rest = &line[start + 2..];
562        if let Some(end) = rest.find("]]") {
563            let target = rest[..end].trim().to_string();
564            if !target.is_empty() {
565                return Some(target);
566            }
567        }
568    }
569    None
570}

572fn extract_label_name(line: &str) -> Option<String> {
573    let prefix = "\\label{";
574    if let Some(start) = line.find(prefix) {
575        let rest = &line[start + prefix.len()..];
576        if let Some(end) = rest.find('}') {
577            let label = rest[..end].trim().to_string();
578            if !label.is_empty() {
579                return Some(label);
580            }
581        }
582    }
583    None

586fn find_key_at_position(line: &str, col: u32, prefixes: &[&str]) -> Option<(String, u32, u32)> {
587    for &prefix in prefixes {
588        let mut offset = 0;
589        while let Some(idx) = line[offset..].find(prefix) {
590            let abs_idx = offset + idx;
591            let key_start = abs_idx + prefix.len();
592            let rest = &line[key_start..];
593            if let Some(brace_pos) = rest.find('}') {
594                let key = rest[..brace_pos].trim().to_string();
595                let key_end = key_start + brace_pos;
596                if !key.is_empty() && col >= key_start as u32 && col < key_end as u32 {
597                    return Some((key, key_start as u32, key_end as u32));
598                }
599                offset = key_start + brace_pos + 1;
600            } else {
601                break;
602            }
603        }
604    }
605    None
606}

608fn extract_md_heading_text(line: &str) -> Option<String> {
609    let trimmed = line.trim_start();
610    let hashes: &str;
611    if let Some(h) = trimmed.strip_prefix("#### ") {
612        hashes = h;
613    } else if let Some(h) = trimmed.strip_prefix("### ") {
614        hashes = h;
615    } else if let Some(h) = trimmed.strip_prefix("## ") {
616        hashes = h;
617    } else if let Some(h) = trimmed.strip_prefix("# ") {
618        hashes = h;
619    } else {
620        return None;
621    }
622    let heading = hashes.trim().to_string();
623    if !heading.is_empty() {
624        Some(heading)
625    } else {
626        None
627    }
628}

757fn completion_latex(prefix: &str, full_text: &str) -> Vec<CompletionItem> {
758    if let Some(stripped) = prefix.strip_suffix("\\begin{") {
759        let typed = stripped.to_lowercase();
760        let mut items = Vec::new();
761        for &(name, insert) in LATEX_ENVIRONMENTS {
762            if name.starts_with(&typed) {
763                items.push(CompletionItem {
764                    label: name.to_string(),
765                    detail: Some(format!("environment {name}")),
766                    kind: Some(CompletionItemKind::SNIPPET),
767                    insert_text: Some(insert.to_string()),
768                    insert_text_format: Some(InsertTextFormat::SNIPPET),
769                    filter_text: Some(name.to_string()),
770                    ..Default::default()
771                });
772            }
773        }
774        return items;
775    }
776
777    if let Some(rest) = find_in_braces(prefix, "\\ref{")
778        .or_else(|| find_in_braces(prefix, "\\cite{"))
779        .or_else(|| find_in_braces(prefix, "\\eqref{"))
780        .or_else(|| find_in_braces(prefix, "\\cref{"))
781        .or_else(|| find_in_braces(prefix, "\\Cref{"))
782        .or_else(|| find_in_braces(prefix, "\\autoref{"))
783    {
784        let typed = rest.to_lowercase();
785        return extract_labels(full_text)
786            .into_iter()
787            .filter(|l| l.to_lowercase().contains(&typed))
788            .map(|label| CompletionItem {
789                label: label.clone(),
790                detail: Some(format!("label: {label}")),
791                kind: Some(CompletionItemKind::REFERENCE),
792                insert_text: Some(label),
793                ..Default::default()
794            })
795            .collect();
796    }
797
798    if let Some(stripped) = prefix.strip_suffix('\\') {
799        let typed = stripped.to_lowercase();
800        let mut items = Vec::new();
801        for &(name, insert) in LATEX_COMMANDS {
802            if name.starts_with(&typed) {
803                items.push(CompletionItem {
804                    label: name.to_string(),
805                    detail: Some(insert.to_string()),
806                    kind: Some(CompletionItemKind::KEYWORD),
807                    insert_text: Some(insert.to_string()),
808                    filter_text: Some(name.to_string()),
809                    ..Default::default()
810                });
811            }
812        }
813        return items;
814    }
815
816    Vec::new()
817}

886fn find_in_braces<'a>(line: &'a str, prefix: &str) -> Option<&'a str> {
887    if let Some(start) = line.rfind(prefix) {
888        let after = &line[start + prefix.len()..];
889        if !after.contains('}') {
890            return Some(after);
891        }
892    }
893    None
894}

896fn extract_labels(text: &str) -> Vec<String> {
897    let prefix = "\\label{";
898    let mut labels = Vec::new();
899    for line in text.lines() {
900        let mut search = line;
901        while let Some(pos) = search.find(prefix) {
902            let rest = &search[pos + prefix.len()..];
903            if let Some(end) = rest.find('}') {
904                labels.push(rest[..end].to_string());
905                search = &rest[end + 1..];
906            } else {
907                break;
908            }
909        }
910    }
911    labels.sort();
912    labels.dedup();
913    labels
914}
