22/// Parameters for the preview status notification.
23#[derive(Debug, Clone, Serialize, Deserialize)]
24#[serde(rename_all = "camelCase")]
25pub struct PreviewStatusParams {
26    /// Document URI.
27    pub uri: String,
28    /// Status message (e.g., "compiling", "ready", "error: ...").
29    pub status: String,
30    /// Path to the generated PDF, if available.
31    #[serde(skip_serializing_if = "Option::is_none")]
32    pub pdf_path: Option<String>,
33    /// Path to the generated HTML preview, if available.
34    #[serde(skip_serializing_if = "Option::is_none")]
35    pub html_path: Option<String>,
36}

63/// Manages debounced preview compilation for the LSP server.
64#[derive(Debug, Clone)]
65pub struct PreviewManager {
66    output_path: Arc<PathBuf>,
67    debounce: Duration,
68    notify: Arc<Notify>,
69    enabled: Arc<AtomicBool>,
70    running: Arc<AtomicBool>,
71    pending: Arc<Mutex<Option<PendingCompile>>>,
72    #[allow(dead_code)]
73    incremental: Arc<std::sync::Mutex<Option<IncrementalState>>>,
74    #[allow(dead_code)]
75    client: Client,
76}
77
78impl PreviewManager {
79    /// Create a new preview manager that compiles to `output_path`.
80    pub fn new(client: Client, output_path: PathBuf) -> Self {
81        let notify = Arc::new(Notify::new());
82        let running = Arc::new(AtomicBool::new(true));
83        let pending = Arc::new(Mutex::new(None));
84        let enabled = Arc::new(AtomicBool::new(false));
85        let debounce = Duration::from_millis(DEFAULT_DEBOUNCE_MS);
86        let output_path = Arc::new(output_path);
87        let incremental = Arc::new(std::sync::Mutex::new(None));
88
89        let bg_notify = notify.clone();
90        let bg_running = running.clone();
91        let bg_pending = pending.clone();
92        let bg_output_path = output_path.clone();
93        let bg_client = client.clone();
94        let bg_incremental = incremental.clone();
95
96        tokio::spawn(async move {
97            Self::background_task(
98                bg_notify,
99                bg_running,
100                bg_pending,
101                bg_output_path,
102                bg_client,
103                bg_incremental,
104                debounce,
105            )
106            .await;
107        });
108
109        Self {
110            output_path,
111            debounce,
112            notify,
113            enabled,
114            running,
115            pending,
116            incremental,
117            client,
118        }
119    }
120
121    /// Enable or disable preview compilation.
122    pub fn set_enabled(&self, enabled: bool) {
123        self.enabled.store(enabled, Ordering::Relaxed);
124    }
125
126    /// Check if preview compilation is enabled.
127    pub fn is_enabled(&self) -> bool {
128        self.enabled.load(Ordering::Relaxed)
129    }
130
131    /// Returns the debounce duration.
132    pub fn debounce(&self) -> Duration {
133        self.debounce
134    }
135
136    /// Returns the output directory path.
137    pub fn output_path(&self) -> &PathBuf {
138        &self.output_path
139    }
140
141    /// Trigger a debounced recompilation of the given document.
142    pub async fn trigger(&self, text: &str, uri: &str) {
143        if !self.enabled.load(Ordering::Relaxed) {
144            return;
145        }
146
147        let extension = detect_extension(uri).to_string();
148        if !matches!(extension.as_str(), "md" | "tex" | "typ") {
149            return;
150        }
151
152        let mut guard = self.pending.lock().await;
153        *guard = Some(PendingCompile {
154            text: text.to_string(),
155            extension,
156            uri: uri.to_string(),
157        });
158        drop(guard);
159
160        self.notify.notify_one();
161    }
162
163    /// Shut down the background compilation task.
164    pub fn shutdown(&self) {
165        self.running.store(false, Ordering::Relaxed);
166        self.notify.notify_one();
167    }

169    async fn background_task(
170        notify: Arc<Notify>,
171        running: Arc<AtomicBool>,
172        pending: Arc<Mutex<Option<PendingCompile>>>,
173        output_path: Arc<PathBuf>,
174        #[allow(dead_code)] client: Client,
175        #[allow(dead_code)] incremental: Arc<std::sync::Mutex<Option<IncrementalState>>>,
176        debounce: Duration,
177    ) {
178        loop {
179            notify.notified().await;
180
181            if !running.load(Ordering::Relaxed) {
182                break;
183            }
184
185            loop {
186                tokio::select! {
187                    biased;
188                    _ = notify.notified() => {
189                        continue;
190                    }
191                    _ = time::sleep(debounce) => {
192                        break;
193                    }
194                }
195            }
196
197            if !running.load(Ordering::Relaxed) {
198                break;
199            }
200
201            let compile = {
202                let mut guard = pending.lock().await;
203                guard.take()
204            };
205
206            let Some(compile) = compile else {
207                continue;
208            };
209
210            let uri = compile.uri.clone();
211            let _ = client
212                .send_notification::<PreviewStatus>(PreviewStatusParams {
213                    uri: uri.clone(),
214                    status: "compiling".to_string(),
215                    pdf_path: None,
216                    html_path: None,
217                })
218                .await;
219
220            let output = (*output_path).clone();
221            let inc = incremental.clone();
222            let result = tokio::task::spawn_blocking(move || {
223                compile_to_pdf(
224                    &compile.text,
225                    &compile.extension,
226                    &compile.uri,
227                    &output,
228                    &inc,
229                )
230            })
231            .await;
232
233            match result {
234                Ok(Ok(res)) => {
235                    tracing::info!("preview PDF written: {}", res.pdf_path.display());
236                    let _ = client
237                        .send_notification::<PreviewStatus>(PreviewStatusParams {
238                            uri,
239                            status: "ready".to_string(),
240                            pdf_path: Some(res.pdf_path.to_string_lossy().to_string()),
241                            html_path: Some(res.html_path.to_string_lossy().to_string()),
242                        })
243                        .await;
244                }
245                Ok(Err(err)) => {
246                    tracing::warn!("preview compile error: {}", err);
247                    let _ = client
248                        .send_notification::<PreviewStatus>(PreviewStatusParams {
249                            uri,
250                            status: format!("error: {}", err),
251                            pdf_path: None,
252                            html_path: None,
253                        })
254                        .await;
255                }
256                Err(err) => {
257                    tracing::error!("preview task panicked: {}", err);
258                    let _ = client
259                        .send_notification::<PreviewStatus>(PreviewStatusParams {
260                            uri,
261                            status: "error: compile task panicked".to_string(),
262                            pdf_path: None,
263                            html_path: None,
264                        })
265                        .await;
266                }
267            }
268        }
269    }
270}

272fn parse_to_sir(text: &str, extension: &str) -> Result<SIRModuleV2, String> {
273    match extension {
274        "md" => {
275            let v1 = ldir_md::parse_markdown(text);
276            Ok(ldir_core::compiler::v1_to_v2::convert_v1_to_v2(&v1))
277        }
278        "tex" => {
279            let v1 = ldir_tex::parse_tex(text);
280            Ok(ldir_core::compiler::v1_to_v2::convert_v1_to_v2(&v1))
281        }
282        "typ" => Ok(ldir_typst::parse_typst(text)),
283        _ => Err(format!("unsupported format: .{}", extension)),
284    }
285}

287/// Result of a preview compilation.

290    pdf_path: PathBuf,
291    html_path: PathBuf,
292}

294fn compile_to_pdf(
295    text: &str,
296    extension: &str,
297    uri: &str,
298    output_dir: &Path,
299    incremental: &std::sync::Mutex<Option<IncrementalState>>,
300) -> Result<CompileResult, String> {
301    let module = parse_to_sir(text, extension)?;
302    let new_module = Arc::new(module);
303
304    // Try incremental recompilation first
305    {
306        let mut state_guard = incremental
307            .lock()
308            .map_err(|e| format!("lock poisoned: {e}"))?;
309        if let Some(state) = state_guard.as_mut() {
310            state.layout.update_sir(Arc::clone(&new_module));
311            let ctx = ldir_core::compiler::context::CompileContext::new();
312            match state.layout.recompile_lir(&state.lir, &ctx) {
313                Ok(new_lir) => {
314                    if Arc::ptr_eq(&new_lir, &state.lir) {
315                        let html_path = derive_html_path(uri, output_dir)?;
316                        return Ok(CompileResult {
317                            pdf_path: state.pdf_path.clone(),
318                            html_path,
319                        });
320                    }
321                    state.lir = new_lir;
322                }
323                Err(e) => {
324                    tracing::debug!(
325                        "incremental LIR recompile failed, falling back to full compile: {e}"
326                    );
327                }
328            }
329            drop(state_guard);
330        }
331    }
332
333    // Full compilation: always generate HTML (no font deps)
334    let html = compile_to_html(&new_module)?;
335
336    let html_path = derive_html_path(uri, output_dir)?;
337    if let Some(parent) = html_path.parent() {
338        std::fs::create_dir_all(parent).map_err(|e| format!("failed to create directory: {e}"))?;
339    }
340    std::fs::write(&html_path, &html).map_err(|e| format!("failed to write HTML: {e}"))?;
341
342    // Also attempt PDF if fonts are available
343    let pdf_path = derive_pdf_path(uri, output_dir)?;
344    let mut compile_ctx = ldir_core::compiler::context::CompileContext::new();
345    match ldir_core::compiler::v2_compile::compile_v2_document(&new_module, &mut compile_ctx) {
346        Ok(gir_doc) => {
347            let pdf_bytes = ldir_pdf::converter::gir_to_pdf(&gir_doc);
348            if let Some(parent) = pdf_path.parent() {
349                let _ = std::fs::create_dir_all(parent);
350            }
351            let _ = std::fs::write(&pdf_path, &pdf_bytes);
352        }
353        Err(_) => {
354            // PDF generation failed (likely no fonts); HTML preview still works
355        }
356    }
357
358    // Store incremental state for next edit
359    let ctx_lir = ldir_core::compiler::context::CompileContext::new();
360    let lir = ldir_core::compile_sir_to_lir(&new_module, &ctx_lir)
361        .map(Arc::new)
362        .unwrap_or_else(|_| Arc::new(LIRDocument::default()));
363
364    let mut state_guard = incremental
365        .lock()
366        .map_err(|e| format!("lock poisoned: {e}"))?;
367    *state_guard = Some(IncrementalState {
368        layout: IncrementalLayout::new(new_module),
369        lir,
370        pdf_path: pdf_path.clone(),
371    });
372
373    Ok(CompileResult {
374        pdf_path,
375        html_path,
376    })
377}

379/// Compile S-IR module to HTML (always works, no font deps).
380fn compile_to_html(module: &SIRModuleV2) -> Result<String, String> {
381    let mut renderer = ldir_html::HtmlRenderer::new();
382    Ok(renderer.render(module))
383}

385fn derive_html_path(uri: &str, output_dir: &Path) -> Result<PathBuf, String> {
386    let url = url::Url::parse(uri).map_err(|e| format!("invalid URI: {e}"))?;
387    let path = url
388        .to_file_path()
389        .map_err(|_| "URI is not a file path".to_string())?;
390    let stem = path
391        .file_stem()
392        .ok_or("cannot determine file stem")?
393        .to_string_lossy()
394        .to_string();
395    Ok(output_dir.join(format!("{stem}.html")))
396}

398fn derive_pdf_path(uri: &str, output_dir: &Path) -> Result<PathBuf, String> {
399    let url = url::Url::parse(uri).map_err(|e| format!("invalid URI: {e}"))?;
400    let path = url
401        .to_file_path()
402        .map_err(|_| "URI is not a file path".to_string())?;
403
404    let stem = path
405        .file_stem()
406        .ok_or("cannot determine file stem")?
407        .to_string_lossy()
408        .to_string();
409
410    Ok(output_dir.join(format!("{stem}.pdf")))
411}
