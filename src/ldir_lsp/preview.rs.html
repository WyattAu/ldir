21/// Parameters for the preview status notification.
22#[derive(Debug, Clone, Serialize, Deserialize)]
23#[serde(rename_all = "camelCase")]
24pub struct PreviewStatusParams {
25    /// Document URI.
26    pub uri: String,
27    /// Status message (e.g., "compiling", "ready", "error: ...").
28    pub status: String,
29    /// Path to the generated PDF, if available.
30    #[serde(skip_serializing_if = "Option::is_none")]
31    pub pdf_path: Option<String>,
32}
33
34/// LSP notification for preview status updates.
35#[derive(Debug)]
36pub struct PreviewStatus;
37
38impl lsp_types::notification::Notification for PreviewStatus {
39    type Params = PreviewStatusParams;
40    const METHOD: &'static str = "ldir/previewStatus";
41}

43/// A pending compilation request waiting to be debounced.
44#[derive(Debug)]
45struct PendingCompile {
46    text: String,
47    extension: String,
48    uri: String,
49}

51/// Manages debounced preview compilation for the LSP server.
52#[derive(Debug)]
53pub struct PreviewManager {
54    output_path: Arc<PathBuf>,
55    debounce: Duration,
56    notify: Arc<Notify>,
57    enabled: Arc<AtomicBool>,
58    running: Arc<AtomicBool>,
59    pending: Arc<Mutex<Option<PendingCompile>>>,
60    client: Client,
61}
62
63impl PreviewManager {
64    /// Create a new preview manager that compiles to `output_path`.
65    pub fn new(client: Client, output_path: PathBuf) -> Self {
66        let notify = Arc::new(Notify::new());
67        let running = Arc::new(AtomicBool::new(true));
68        let pending = Arc::new(Mutex::new(None));
69        let enabled = Arc::new(AtomicBool::new(false));
70        let debounce = Duration::from_millis(DEFAULT_DEBOUNCE_MS);
71        let output_path = Arc::new(output_path);
72
73        let bg_notify = notify.clone();
74        let bg_running = running.clone();
75        let bg_pending = pending.clone();
76        let bg_output_path = output_path.clone();
77        let bg_client = client.clone();
78
79        tokio::spawn(async move {
80            Self::background_task(
81                bg_notify,
82                bg_running,
83                bg_pending,
84                bg_output_path,
85                bg_client,
86                debounce,
87            )
88            .await;
89        });
90
91        Self {
92            output_path,
93            debounce,
94            notify,
95            enabled,
96            running,
97            pending,
98            client,
99        }
100    }
101
102    /// Enable or disable preview compilation.
103    pub fn set_enabled(&self, enabled: bool) {
104        self.enabled.store(enabled, Ordering::Relaxed);
105    }
106
107    /// Check if preview compilation is enabled.
108    pub fn is_enabled(&self) -> bool {
109        self.enabled.load(Ordering::Relaxed)
110    }
111
112    /// Returns the debounce duration.
113    pub fn debounce(&self) -> Duration {
114        self.debounce
115    }
116
117    /// Returns the output directory path.
118    pub fn output_path(&self) -> &PathBuf {
119        &self.output_path
120    }
121
122    /// Trigger a debounced recompilation of the given document.
123    pub async fn trigger(&self, text: &str, uri: &str) {
124        if !self.enabled.load(Ordering::Relaxed) {
125            return;
126        }
127
128        let extension = detect_extension(uri).to_string();
129        if !matches!(extension.as_str(), "md" | "tex" | "typ") {
130            return;
131        }
132
133        let mut guard = self.pending.lock().await;
134        *guard = Some(PendingCompile {
135            text: text.to_string(),
136            extension,
137            uri: uri.to_string(),
138        });
139        drop(guard);
140
141        self.notify.notify_one();
142    }
143
144    /// Shut down the background compilation task.
145    pub fn shutdown(&self) {
146        self.running.store(false, Ordering::Relaxed);
147        self.notify.notify_one();
148    }
149
150    async fn background_task(
151        notify: Arc<Notify>,
152        running: Arc<AtomicBool>,
153        pending: Arc<Mutex<Option<PendingCompile>>>,
154        output_path: Arc<PathBuf>,
155        #[allow(dead_code)] client: Client,
156        debounce: Duration,
157    ) {
158        loop {
159            notify.notified().await;
160
161            if !running.load(Ordering::Relaxed) {
162                break;
163            }
164
165            loop {
166                tokio::select! {
167                    biased;
168                    _ = notify.notified() => {
169                        continue;
170                    }
171                    _ = time::sleep(debounce) => {
172                        break;
173                    }
174                }
175            }
176
177            if !running.load(Ordering::Relaxed) {
178                break;
179            }
180
181            let compile = {
182                let mut guard = pending.lock().await;
183                guard.take()
184            };
185
186            let Some(compile) = compile else {
187                continue;
188            };
189
190            let uri = compile.uri.clone();
191            let _ = client
192                .send_notification::<PreviewStatus>(PreviewStatusParams {
193                    uri: uri.clone(),
194                    status: "compiling".to_string(),
195                    pdf_path: None,
196                })
197                .await;
198
199            let output = (*output_path).clone();
200            let result = tokio::task::spawn_blocking(move || {
201                compile_to_pdf(&compile.text, &compile.extension, &compile.uri, &output)
202            })
203            .await;
204
205            match result {
206                Ok(Ok(pdf_path)) => {
207                    tracing::info!("preview PDF written: {}", pdf_path.display());
208                    let _ = client
209                        .send_notification::<PreviewStatus>(PreviewStatusParams {
210                            uri,
211                            status: "ready".to_string(),
212                            pdf_path: Some(pdf_path.to_string_lossy().to_string()),
213                        })
214                        .await;
215                }
216                Ok(Err(err)) => {
217                    tracing::warn!("preview compile error: {}", err);
218                    let _ = client
219                        .send_notification::<PreviewStatus>(PreviewStatusParams {
220                            uri,
221                            status: format!("error: {}", err),
222                            pdf_path: None,
223                        })
224                        .await;
225                }
226                Err(err) => {
227                    tracing::error!("preview task panicked: {}", err);
228                    let _ = client
229                        .send_notification::<PreviewStatus>(PreviewStatusParams {
230                            uri,
231                            status: "error: compile task panicked".to_string(),
232                            pdf_path: None,
233                        })
234                        .await;
235                }
236            }
237        }
238    }
239}

241fn compile_to_pdf(
242    text: &str,
243    extension: &str,
244    uri: &str,
245    output_dir: &Path,
246) -> Result<PathBuf, String> {
247    let module = match extension {
248        "md" => {
249            let v1 = ldir_md::parse_markdown(text);
250            ldir_core::compiler::v1_to_v2::convert_v1_to_v2(&v1)
251        }
252        "tex" => {
253            let v1 = ldir_tex::parse_tex(text);
254            ldir_core::compiler::v1_to_v2::convert_v1_to_v2(&v1)
255        }
256        "typ" => ldir_typst::parse_typst(text),
257        _ => return Err(format!("unsupported format: .{}", extension)),
258    };
259
260    let mut ctx = ldir_core::compiler::context::CompileContext::new();
261    let gir_doc = ldir_core::compiler::v2_compile::compile_v2_document(&module, &mut ctx)
262        .map_err(|e| format!("compile error: {e}"))?;
263
264    let pdf_bytes = ldir_pdf::converter::gir_to_pdf(&gir_doc);
265
266    let pdf_path = derive_pdf_path(uri, output_dir)?;
267    if let Some(parent) = pdf_path.parent() {
268        std::fs::create_dir_all(parent).map_err(|e| format!("failed to create directory: {e}"))?;
269    }
270
271    std::fs::write(&pdf_path, &pdf_bytes).map_err(|e| format!("failed to write PDF: {e}"))?;
272
273    Ok(pdf_path)
274}

276fn derive_pdf_path(uri: &str, output_dir: &Path) -> Result<PathBuf, String> {
277    let url = url::Url::parse(uri).map_err(|e| format!("invalid URI: {e}"))?;
278    let path = url
279        .to_file_path()
280        .map_err(|_| "URI is not a file path".to_string())?;
281
282    let stem = path
283        .file_stem()
284        .ok_or("cannot determine file stem")?
285        .to_string_lossy()
286        .to_string();
287
288    Ok(output_dir.join(format!("{stem}.pdf")))
289}
