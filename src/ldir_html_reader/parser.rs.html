34fn parse_html_dimension(s: &str) -> Option<Dimension> {
35    let s = s.trim();
36    if let Some(v) = s
37        .strip_suffix("px")
38        .and_then(|v| v.trim().parse::<f64>().ok())
39    {
40        return Some(Dimension::Pt(v));
41    }
42    if let Some(v) = s
43        .strip_suffix('%')
44        .and_then(|v| v.trim().parse::<f64>().ok())
45    {
46        return Some(Dimension::Percent(v));
47    }
48    if let Some(v) = s
49        .strip_suffix("pt")
50        .and_then(|v| v.trim().parse::<f64>().ok())
51    {
52        return Some(Dimension::Pt(v));
53    }
54    if let Some(v) = s
55        .strip_suffix("mm")
56        .and_then(|v| v.trim().parse::<f64>().ok())
57    {
58        return Some(Dimension::Mm(v));
59    }
60    if let Some(v) = s
61        .strip_suffix("cm")
62        .and_then(|v| v.trim().parse::<f64>().ok())
63    {
64        return Some(Dimension::Cm(v));
65    }
66    if let Some(v) = s
67        .strip_suffix("in")
68        .and_then(|v| v.trim().parse::<f64>().ok())
69    {
70        return Some(Dimension::In(v));
71    }
72    s.parse::<f64>().ok().map(Dimension::Pt)
73}

75fn extract_width_from_style(style: &str) -> Option<f64> {
76    let style_lower = style.to_ascii_lowercase();
77    let start = style_lower.find("width:")?;
78    let after = &style_lower[start + 6..];
79    let end = after.find(';').unwrap_or(after.len());
80    let val = after[..end].trim();
81    let val = val.strip_suffix("px").unwrap_or(val).trim();
82    val.parse::<f64>().ok()
83}

85fn decode_entities(s: &str) -> String {
86    let mut out = String::with_capacity(s.len());
87    let bytes = s.as_bytes();
88    let mut i = 0;
89
90    while i < bytes.len() {
91        if bytes[i] == b'&' {
92            let semi = s[i..].find(';');
93            if let Some(pos) = semi {
94                let entity = &s[i..i + pos + 1];
95                let decoded = match entity {
96                    "&amp;" => Some('&'),
97                    "&lt;" => Some('<'),
98                    "&gt;" => Some('>'),
99                    "&quot;" => Some('"'),
100                    "&apos;" => Some('\''),
101                    "&nbsp;" => Some('\u{00A0}'),
102                    "&mdash;" => Some('—'),
103                    "&ndash;" => Some('–'),
104                    "&copy;" => Some('©'),
105                    "&reg;" => Some('®'),
106                    "&trade;" => Some('™'),
107                    "&hellip;" => Some('…'),
108                    "&laquo;" => Some('«'),
109                    "&raquo;" => Some('»'),
110                    "&bull;" => Some('•'),
111                    "&middot;" => Some('·'),
112                    _ => {
113                        if let Some(hex_str) =
114                            entity.strip_prefix("&#x").and_then(|e| e.strip_suffix(';'))
115                        {
116                            if let Ok(code) = u32::from_str_radix(hex_str, 16) {
117                                char::from_u32(code)
118                            } else {
119                                None
120                            }
121                        } else if let Some(dec_str) =
122                            entity.strip_prefix("&#").and_then(|e| e.strip_suffix(';'))
123                        {
124                            if let Ok(code) = dec_str.parse::<u32>() {
125                                char::from_u32(code)
126                            } else {
127                                None
128                            }
129                        } else {
130                            None
131                        }
132                    }
133                };
134                match decoded {
135                    Some(c) => {
136                        out.push(c);
137                        i += pos + 1;
138                    }
139                    None => {
140                        out.push_str(entity);
141                        i += pos + 1;
142                    }
143                }
144            } else {
145                out.push('&');
146                i += 1;
147            }
148        } else {
149            out.push(bytes[i] as char);
150            i += 1;
151        }
152    }
153    out
154}

156fn tokenize(input: &str) -> Vec<HtmlToken> {
157    let mut tokens = Vec::new();
158    let bytes = input.as_bytes();
159    let mut i = 0;
160
161    while i < bytes.len() {
162        if bytes[i] == b'<' {
163            if i + 4 <= bytes.len() && &input[i..i + 4] == "<!--" {
164                if let Some(end) = input[i..].find("-->") {
165                    tokens.push(HtmlToken::Comment);
166                    i += end + 3;
167                } else {
168                    i = bytes.len();
169                }
170                continue;
171            }
172
173            if i + 9 <= bytes.len() && input[i..i + 9].eq_ignore_ascii_case("<!doctype") {
174                if let Some(end) = input[i..].find('>') {
175                    i += end + 1;
176                } else {
177                    i = bytes.len();
178                }
179                continue;
180            }
181
182            if i + 1 < bytes.len() && bytes[i + 1] == b'?' {
183                if let Some(end) = input[i..].find("?>") {
184                    i += end + 2;
185                } else {
186                    i = bytes.len();
187                }
188                continue;
189            }
190
191            i += 1;
192
193            let is_closing = if i < bytes.len() && bytes[i] == b'/' {
194                i += 1;
195                true
196            } else {
197                false
198            };
199
200            let tag_start = i;
201            while i < bytes.len() && bytes[i].is_ascii_alphanumeric() {
202                i += 1;
203            }
204            let tag_name = input[tag_start..i].to_ascii_lowercase();
205
206            if tag_name.is_empty() {
207                tokens.push(HtmlToken::Text {
208                    content: "<".to_string(),
209                });
210                continue;
211            }
212
213            let mut attrs = Vec::new();
214            let mut self_closing = false;
215
216            loop {
217                while i < bytes.len() && bytes[i] == b' ' {
218                    i += 1;
219                }
220                if i >= bytes.len() {
221                    break;
222                }
223                if bytes[i] == b'>' {
224                    i += 1;
225                    break;
226                }
227                if bytes[i] == b'/' && i + 1 < bytes.len() && bytes[i + 1] == b'>' {
228                    i += 2;
229                    self_closing = true;
230                    break;
231                }
232
233                let attr_name_start = i;
234                while i < bytes.len()
235                    && bytes[i] != b'='
236                    && bytes[i] != b'>'
237                    && bytes[i] != b'/'
238                    && bytes[i] != b' '
239                {
240                    i += 1;
241                }
242                let attr_name = input[attr_name_start..i].to_ascii_lowercase();
243
244                while i < bytes.len() && bytes[i] == b' ' {
245                    i += 1;
246                }
247
248                if i < bytes.len() && bytes[i] == b'=' {
249                    i += 1;
250                    while i < bytes.len() && bytes[i] == b' ' {
251                        i += 1;
252                    }
253                    let attr_value = if i < bytes.len() && (bytes[i] == b'"' || bytes[i] == b'\'') {
254                        let quote = bytes[i];
255                        i += 1;
256                        let val_start = i;
257                        while i < bytes.len() && bytes[i] != quote {
258                            i += 1;
259                        }
260                        let val = input[val_start..i].to_string();
261                        if i < bytes.len() {
262                            i += 1;
263                        }
264                        val
265                    } else {
266                        let val_start = i;
267                        while i < bytes.len()
268                            && bytes[i] != b'>'
269                            && bytes[i] != b' '
270                            && bytes[i] != b'/'
271                        {
272                            i += 1;
273                        }
274                        input[val_start..i].to_string()
275                    };
276                    attrs.push((attr_name, attr_value));
277                } else {
278                    attrs.push((attr_name, String::new()));
279                }
280            }
281
282            if is_closing {
283                tokens.push(HtmlToken::CloseTag { tag: tag_name });
284            } else {
285                tokens.push(HtmlToken::OpenTag {
286                    tag: tag_name,
287                    attrs,
288                    self_closing,
289                });
290            }
291        } else {
292            let text_start = i;
293            while i < bytes.len() && bytes[i] != b'<' {
294                i += 1;
295            }
296            let raw = &input[text_start..i];
297            let decoded = decode_entities(raw);
298            if !decoded.is_empty() {
299                tokens.push(HtmlToken::Text { content: decoded });
300            }
301        }
302    }
303
304    tokens
305}

307fn build_dom(tokens: &[HtmlToken]) -> DomNode {
308    let root = DomNode {
309        tag: String::new(),
310        attrs: Vec::new(),
311        children: Vec::new(),
312        text: None,
313    };
314    let mut stack: Vec<DomNode> = vec![root];
315
316    for token in tokens {
317        match token {
318            HtmlToken::OpenTag {
319                tag,
320                attrs,
321                self_closing,
322            } => {
323                let void = matches!(
324                    tag.as_str(),
325                    "br" | "hr"
326                        | "img"
327                        | "input"
328                        | "meta"
329                        | "link"
330                        | "area"
331                        | "base"
332                        | "col"
333                        | "embed"
334                        | "source"
335                        | "track"
336                        | "wbr"
337                );
338                let node = DomNode {
339                    tag: tag.clone(),
340                    attrs: attrs.clone(),
341                    children: Vec::new(),
342                    text: None,
343                };
344                if !*self_closing && !void {
345                    stack.push(node);
346                } else {
347                    if let Some(parent) = stack.last_mut() {
348                        parent.children.push(node);
349                    }
350                }
351            }
352            HtmlToken::CloseTag { tag } => {
353                let skip_tags = ["head", "style", "script", "noscript"];
354                if skip_tags.contains(&tag.as_str()) {
355                    while let Some(current) = stack.last() {
356                        if current.tag == *tag {
357                            let popped = stack.pop();
358                            if let Some(popped_node) = popped
359                                && let Some(parent) = stack.last_mut()
360                            {
361                                parent.children.push(popped_node);
362                            }
363                            break;
364                        }
365                        let popped = stack.pop();
366                        if stack.len() <= 1 {
367                            if let Some(n) = popped {
368                                stack[0].children.push(n);
369                            }
370                            break;
371                        }
372                    }
373                    continue;
374                }
375
376                let mut found = false;
377                for j in (1..stack.len()).rev() {
378                    if stack[j].tag == *tag {
379                        let popped = stack.remove(j);
380                        if let Some(parent) = stack.last_mut() {
381                            parent.children.push(popped);
382                        }
383                        found = true;
384                        break;
385                    }
386                }
387                let _ = found;
388            }
389            HtmlToken::Text { content } => {
390                if let Some(current) = stack.last_mut() {
391                    current
392                        .text
393                        .get_or_insert_with(String::new)
394                        .push_str(content);
395                }
396            }
397            HtmlToken::Comment => {}
398        }
399    }
400
401    while stack.len() > 1 {
402        let popped = stack.pop();
403        let parent = stack.last_mut();
404        if let (Some(popped), Some(parent)) = (popped, parent) {
405            parent.children.push(popped);
406        }
407    }
408
409    stack
410        .pop()
411        .unwrap_or_else(|| unreachable!("stack always contains root element"))
412}

414struct Converter {
415    module: SIRModuleV2,
416    next_id: u32,
417}

419impl Converter {
420    fn new() -> Self {
421        let module = SIRModuleV2::from_source("html", "");
422        Self { module, next_id: 0 }
423    }
424
425    fn alloc_id(&mut self) -> u32 {
426        let id = self.next_id;
427        self.next_id += 1;
428        id
429    }
430
431    fn convert_dom(&mut self, dom: &DomNode) {
432        let body = Self::find_body(dom);
433        let children: &[DomNode] = match &body {
434            Some(node) => &node.children,
435            None => &dom.children,
436        };
437
438        if let Some(lang) = body.as_ref().and_then(|b| attr(&b.attrs, "lang")) {
439            self.module.metadata.language = lang;
440        } else {
441            for child in &dom.children {
442                if child.tag == "html" {
443                    if let Some(lang) = attr(&child.attrs, "lang") {
444                        self.module.metadata.language = lang;
445                    }
446                    break;
447                }
448            }
449        }
450
451        for child in children {
452            self.convert_node(child, None);
453        }
454    }
455
456    fn find_body(dom: &DomNode) -> Option<&DomNode> {
457        for child in &dom.children {
458            if child.tag == "html" {
459                for html_child in &child.children {
460                    if html_child.tag == "body" {
461                        return Some(html_child);
462                    }
463                }
464                return None;
465            }
466            if child.tag == "body" {
467                return Some(child);
468            }
469        }
470        None
471    }
472
473    fn convert_node(&mut self, dom: &DomNode, parent_id: Option<u32>) -> Option<u32> {
474        let skip_tags = ["head", "style", "script", "noscript", "html", "!doctype"];
475        if skip_tags.contains(&dom.tag.as_str()) {
476            for child in &dom.children {
477                self.convert_node(child, parent_id);
478            }
479            return None;
480        }
481
482        let tag = dom.tag.as_str();
483
484        if matches!(tag, "meta" | "link" | "title") {
485            return None;
486        }
487
488        let node_type = match tag {
489            "h1" => Some(NodeType::Chapter),
490            "h2" => Some(NodeType::Section),
491            "h3" => Some(NodeType::Subsection),
492            "h4" => Some(NodeType::Subsubsection),
493            "h5" | "h6" => Some(NodeType::Subsubsection),
494            "p" => Some(NodeType::Paragraph),
495            "strong" | "b" => Some(NodeType::Bold),
496            "em" | "i" => Some(NodeType::Italic),
497            "code" => Some(NodeType::Mono),
498            "u" => Some(NodeType::Underline),
499            "s" | "del" | "strike" => Some(NodeType::Strikethrough),
500            "a" => {
501                let url = attr(&dom.attrs, "href").unwrap_or_default();
502                let title = attr(&dom.attrs, "title");
503                Some(NodeType::Link { url, title })
504            }
505            "img" => {
506                let source = attr(&dom.attrs, "src").unwrap_or_default();
507                let alt = attr(&dom.attrs, "alt").unwrap_or_default();
508                let width = attr(&dom.attrs, "width").and_then(|v| parse_html_dimension(&v));
509                let height = attr(&dom.attrs, "height").and_then(|v| parse_html_dimension(&v));
510                Some(NodeType::Image {
511                    source,
512                    alt,
513                    width,
514                    height,
515                    placement: FloatPlacement::Here,
516                })
517            }
518            "ul" => Some(NodeType::List {
519                list_type: ListType::Unordered,
520                ordered: false,
521                start: None,
522            }),
523            "ol" => Some(NodeType::List {
524                list_type: ListType::Ordered,
525                ordered: true,
526                start: attr(&dom.attrs, "start").and_then(|s| s.parse::<u32>().ok()),
527            }),
528            "li" => Some(NodeType::ListItem),
529            "dl" => Some(NodeType::List {
530                list_type: ListType::Description,
531                ordered: false,
532                start: None,
533            }),
534            "dt" => Some(NodeType::Bold),
535            "dd" => Some(NodeType::Group),
536            "colgroup" | "col" => None,
537            "blockquote" => Some(NodeType::BlockQuote),
538            "pre" => Some(NodeType::CodeBlock {
539                language: None,
540                content: String::new(),
541            }),
542            "table" => Some(NodeType::Table {
543                col_specs: Vec::new(),
544                num_cols: 0,
545                caption: None,
546                column_widths: Vec::new(),
547                header_row: false,
548            }),
549            "tr" => {
550                let is_header = dom.children.iter().any(|c| c.tag == "th");
551                Some(NodeType::TableRow { is_header })
552            }
553            "th" | "td" => {
554                let colspan = attr(&dom.attrs, "colspan")
555                    .and_then(|v| v.parse::<u8>().ok())
556                    .unwrap_or(1);
557                let rowspan = attr(&dom.attrs, "rowspan")
558                    .and_then(|v| v.parse::<u8>().ok())
559                    .unwrap_or(1);
560                Some(NodeType::TableCell { colspan, rowspan })
561            }
562            "hr" => Some(NodeType::ThematicBreak),
563            "br" => Some(NodeType::LineBreak),
564            "div" => Some(NodeType::Group),
565            "span" => {
566                if let Some(class) = attr(&dom.attrs, "class") {
567                    let first_class = class.split_whitespace().next().unwrap_or("").to_string();
568                    if !first_class.is_empty() {
569                        Some(NodeType::Styled {
570                            style_name: first_class,
571                        })
572                    } else {
573                        Some(NodeType::Group)
574                    }
575                } else {
576                    Some(NodeType::Group)
577                }
578            }
579            "sup" => Some(NodeType::Group),
580            "sub" => Some(NodeType::Group),
581            "math" | "mtext" => Some(NodeType::MathInline {
582                content: dom.text.clone().unwrap_or_default(),
583            }),
584            "figcaption" => Some(NodeType::Caption),
585            "figure" => Some(NodeType::Figure {
586                placement: FloatPlacement::Here,
587            }),
588            _ => Some(NodeType::Group),
589        };
590
591        let node_type = node_type?;
592
593        let id = self.alloc_id();
594        let mut node = Node::new(id, node_type);
595
596        if let Some(pid) = parent_id {
597            node.parent_id = Some(pid);
598        }
599
600        if let Some(label) = attr(&dom.attrs, "id") {
601            node.label = Some(label);
602        }
603
604        if let Some(class) = attr(&dom.attrs, "class") {
605            let first = class.split_whitespace().next().unwrap_or("").to_string();
606            if !first.is_empty() && node.label.is_none() {
607                node.style = Some(first);
608            }
609        }
610
611        let emitted_id = self.module.body.push(node);
612
613        if let Some(pid) = parent_id
614            && let Some(parent) = self.module.body.get_mut(pid)
615        {
616            parent.add_child(emitted_id);
617        }
618
619        if tag == "pre" {
620            if let Some(inner) = dom.children.first()
621                && inner.tag == "code"
622            {
623                let lang = attr(&inner.attrs, "class")
624                    .and_then(|c| {
625                        c.split_whitespace()
626                            .find(|cls| cls.starts_with("language-"))
627                            .map(|cls| cls.strip_prefix("language-").unwrap_or(cls).to_string())
628                    })
629                    .or_else(|| attr(&inner.attrs, "lang"));
630                if let Some(pre_node) = self.module.body.get_mut(emitted_id) {
631                    pre_node.node_type = NodeType::CodeBlock {
632                        language: lang,
633                        content: String::new(),
634                    };
635                }
636
637                if let Some(text) = &inner.text {
638                    let text_id = self.alloc_id();
639                    self.module.body.push(
640                        Node::new(
641                            text_id,
642                            NodeType::Text {
643                                content: text.clone(),
644                            },
645                        )
646                        .with_parent(emitted_id),
647                    );
648                    if let Some(pre_node) = self.module.body.get_mut(emitted_id) {
649                        pre_node.add_child(text_id);
650                    }
651                }
652                return Some(emitted_id);
653            }
654
655            if let Some(text) = &dom.text {
656                let text_id = self.alloc_id();
657                self.module.body.push(
658                    Node::new(
659                        text_id,
660                        NodeType::Text {
661                            content: text.clone(),
662                        },
663                    )
664                    .with_parent(emitted_id),
665                );
666                if let Some(pre_node) = self.module.body.get_mut(emitted_id) {
667                    pre_node.add_child(text_id);
668                }
669            }
670            return Some(emitted_id);
671        }
672
673        if tag == "img" {
674            return Some(emitted_id);
675        }
676
677        if tag == "table" {
678            let mut col_widths = Vec::new();
679            for child in &dom.children {
680                if child.tag == "colgroup" {
681                    for col in &child.children {
682                        if col.tag == "col" {
683                            if let Some(w) =
684                                attr(&col.attrs, "width").and_then(|v| v.parse::<f64>().ok())
685                            {
686                                col_widths.push(w);
687                            } else if let Some(style) = attr(&col.attrs, "style")
688                                && let Some(w) = extract_width_from_style(&style)
689                            {
690                                col_widths.push(w);
691                            }
692                        }
693                    }
694                } else if child.tag == "col" {
695                    if let Some(w) = attr(&child.attrs, "width").and_then(|v| v.parse::<f64>().ok())
696                    {
697                        col_widths.push(w);
698                    } else if let Some(style) = attr(&child.attrs, "style")
699                        && let Some(w) = extract_width_from_style(&style)
700                    {
701                        col_widths.push(w);
702                    }
703                }
704            }
705            if !col_widths.is_empty()
706                && let Some(table_node) = self.module.body.get_mut(emitted_id)
707                && let NodeType::Table {
708                    column_widths,
709                    num_cols,
710                    ..
711                } = &mut table_node.node_type
712            {
713                *num_cols = col_widths.len();
714                *column_widths = col_widths;
715            }
716        }
717
718        if tag == "br" || tag == "hr" {
719            return Some(emitted_id);
720        }
721
722        if tag == "math" || tag == "mtext" {
723            return Some(emitted_id);
724        }
725
726        if let Some(text) = &dom.text
727            && !text.is_empty()
728        {
729            let text_id = self.alloc_id();
730            self.module.body.push(
731                Node::new(
732                    text_id,
733                    NodeType::Text {
734                        content: text.clone(),
735                    },
736                )
737                .with_parent(emitted_id),
738            );
739            if let Some(this_node) = self.module.body.get_mut(emitted_id) {
740                this_node.add_child(text_id);
741            }
742        }
743
744        for child in &dom.children {
745            self.convert_node(child, Some(emitted_id));
746        }
747
748        Some(emitted_id)
749    }
750}

752pub fn parse_html(html: &str) -> SIRModuleV2 {
753    let tokens = tokenize(html);
754    let dom = build_dom(&tokens);
755    let mut converter = Converter::new();
756    converter.convert_dom(&dom);
757    converter.module
758}
