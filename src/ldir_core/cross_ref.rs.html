1//! Cross-reference resolution for document labels and citations.
2
3/// A resolved reference target with its location and context.
4#[derive(Debug, Clone, PartialEq, Eq)]
5pub struct ResolvedRef {
6    /// Label or identifier being referenced.
7    pub label: String,
8    /// Page number where the target is found (1-indexed, 0 if unknown).
9    pub page: u32,
10    /// Section number if heading (e.g., "2.3.1").
11    pub section: Option<String>,
12    /// Reference type: internal link, external URL, bibliography.
13    pub ref_type: RefType,
14}
15
16#[derive(Debug, Clone, Copy, PartialEq, Eq)]
17pub enum RefType {
18    Internal,
19    External,
20    Bibliography,
21    Equation,
22    Figure,
23    Table,
24}
25
26/// A collection of all label definitions in a document.
27#[derive(Debug, Clone, Default, PartialEq, Eq)]
28pub struct LabelRegistry {
29    entries: Vec<LabelEntry>,
30}
31
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub struct LabelEntry {
34    pub label: String,
35    pub page: u32,
36    pub section: Option<String>,
37    pub node_type: String,
38}
39
40impl LabelRegistry {
41    pub fn new() -> Self {
42        Self::default()
43    }
44
45    /// Register a label definition.
46    pub fn register(&mut self, label: String, page: u32, section: Option<String>, node_type: &str) {
47        if !self.entries.iter().any(|e| e.label == label) {
48            self.entries.push(LabelEntry {
49                label,
50                page,
51                section,
52                node_type: node_type.to_string(),
53            });
54        }
55    }
56
57    /// Look up a label by name.
58    pub fn lookup(&self, label: &str) -> Option<&LabelEntry> {
59        self.entries.iter().find(|e| e.label == label)
60    }
61
62    /// Check if all labels referenced in refs have definitions.
63    pub fn unresolved_refs<'a>(&self, refs: impl Iterator<Item = &'a str>) -> Vec<&'a str> {
64        refs.filter(|r| self.lookup(r).is_none()).collect()
65    }
66
67    /// Return count of registered labels.
68    pub fn len(&self) -> usize {
69        self.entries.len()
70    }
71
72    /// Check if registry is empty.
73    pub fn is_empty(&self) -> bool {
74        self.entries.is_empty()
75    }
76}
77
78/// Resolve a reference string (e.g., "\\ref{fig:sunset}") to a resolved ref.
79pub fn resolve_ref(raw: &str) -> ResolvedRef {
80    let label = raw.trim_start_matches("\\ref{").trim_end_matches('}');
81    let (label, ref_type) = if label.ends_with(":url") {
82        (label.trim_end_matches(":url"), RefType::External)
83    } else if label.ends_with(":bib") {
84        (label.trim_end_matches(":bib"), RefType::Bibliography)
85    } else if label.starts_with("eq:") {
86        (label, RefType::Equation)
87    } else if label.starts_with("fig:") {
88        (label, RefType::Figure)
89    } else if label.starts_with("tbl:") {
90        (label, RefType::Table)
91    } else {
92        (label, RefType::Internal)
93    };
94
95    ResolvedRef {
96        label: label.to_string(),
97        page: 0,
98        section: None,
99        ref_type,
100    }
101}

103#[cfg(test)]
104mod tests {
105    use super::*;
106
107    #[test]
108    fn test_register_and_lookup() {
109        let mut reg = LabelRegistry::new();
110        reg.register("fig:sunset".into(), 1, None, "Figure");
111        reg.register("eq:euler".into(), 2, Some("2.1".into()), "Equation");
112        assert!(reg.lookup("fig:sunset").is_some());
113        assert!(reg.lookup("eq:euler").is_some());
114        assert!(reg.lookup("missing").is_none());
115        assert_eq!(reg.len(), 2);
116    }
117
118    #[test]
119    fn test_no_duplicate_registration() {
120        let mut reg = LabelRegistry::new();
121        reg.register("intro".into(), 1, None, "Heading");
122        reg.register("intro".into(), 2, None, "Heading");
123        assert_eq!(reg.len(), 1);
124    }
125
126    #[test]
127    fn test_unresolved_refs_detection() {
128        let mut reg = LabelRegistry::new();
129        reg.register("exists".into(), 1, None, "Paragraph");
130        let refs = ["missing", "exists", "also_missing"];
131        let unresolved = reg.unresolved_refs(refs.iter().copied());
132        assert_eq!(unresolved.len(), 2);
133    }
