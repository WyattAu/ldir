1//! Math layout engine (Phase 2 + Phase 6A: Advanced Math Environments).
2//!
3//! Parses math content strings into tokens, builds a layout tree,
4//! and produces positioned glyph commands for display and inline math.
5//!
6//! Supports: subscripts (`_`), superscripts (`^`), fractions (`\frac{}{}`),
7//! radicals (`\sqrt{}`), big operators (`\sum`, `\int`, `\prod`),
8//! binary operators, delimiters, `\begin{cases}`, matrix environments,
9//! and stretchy delimiters (`\left`/`\right`).

106fn parse_braced_arg(chars: &[char], pos: &mut usize) -> String {
107    if *pos >= chars.len() || chars[*pos] != '{' {
108        return String::new();
109    }
110    *pos += 1;
111    let start = *pos;
112    let mut depth = 1usize;
113    while *pos < chars.len() && depth > 0 {
114        if chars[*pos] == '{' {
115            depth += 1;
116        } else if chars[*pos] == '}' {
117            depth -= 1;
118        }
119        if depth > 0 {
120            *pos += 1;
121        }
122    }
123    let result: String = chars[start..*pos].iter().collect();
124    if *pos < chars.len() {
125        *pos += 1;
126    }
127    result
128}

130fn parse_single_arg(chars: &[char], pos: &mut usize) -> String {
131    if *pos >= chars.len() {
132        return String::new();
133    }
134    if chars[*pos] == '{' {
135        return parse_braced_arg(chars, pos);
136    }
137    let ch = chars[*pos];
138    *pos += 1;
139    ch.to_string()
140}

142fn tokenize_math(content: &str) -> Vec<MathToken> {
143    let chars: Vec<char> = content.chars().collect();
144    let mut tokens = Vec::new();
145    let mut pos = 0;
146    let len = chars.len();
147
148    while pos < len {
149        if chars[pos].is_whitespace() {
150            pos += 1;
151            continue;
152        }
153
154        if chars[pos] == '\\' {
155            if pos + 1 < len && chars[pos + 1] == '\\' {
156                tokens.push(MathToken::RowSeparator);
157                pos += 2;
158                continue;
159            }
160            let cmd_start = pos + 1;
161            pos = cmd_start;
162            while pos < len && chars[pos].is_alphabetic() {
163                pos += 1;
164            }
165            let cmd: String = chars[cmd_start..pos].iter().collect();
166
167            match cmd.as_str() {
168                "frac" => {
169                    let num = parse_braced_arg(&chars, &mut pos);
170                    let den = parse_braced_arg(&chars, &mut pos);
171                    tokens.push(MathToken::Fraction(num, den));
172                }
173                "sqrt" => {
174                    let inner = parse_braced_arg(&chars, &mut pos);
175                    if inner.is_empty() {
176                        if pos < len {
177                            let ch = chars[pos];
178                            pos += 1;
179                            tokens.push(MathToken::Radical(Box::new(MathToken::Text(
180                                ch.to_string(),
181                            ))));
182                        }
183                    } else {
184                        let inner_tokens = tokenize_math(&inner);
185                        if inner_tokens.len() == 1 {
186                            if let Some(token) = inner_tokens.into_iter().next() {
187                                tokens.push(MathToken::Radical(Box::new(token)));
188                            } else {
189                                tokens.push(MathToken::Radical(Box::new(MathToken::Text(inner))));
190                            }
191                        } else {
192                            tokens.push(MathToken::Radical(Box::new(MathToken::Text(inner))));
193                        }
194                    }
195                }
196                "sum" | "int" | "prod" => {
197                    tokens.push(MathToken::BigOp(cmd));
198                }
199                "left" => {
200                    if pos < len {
201                        let delim = chars[pos];
202                        pos += 1;
203                        tokens.push(MathToken::LeftDelim(delim));
204                    }
205                }
206                "right" => {
207                    if pos < len {
208                        let delim = chars[pos];
209                        pos += 1;
210                        tokens.push(MathToken::RightDelim(delim));
211                    }
212                }
213                "begin" => {
214                    if pos < len && chars[pos] == '{' {
215                        pos += 1;
216                        let env_start = pos;
217                        while pos < len && chars[pos] != '}' {
218                            pos += 1;
219                        }
220                        let env: String = chars[env_start..pos].iter().collect();
221                        if pos < len {
222                            pos += 1;
223                        }
224                        match env.as_str() {
225                            "cases" => tokens.push(MathToken::BeginCases),
226                            "matrix" => tokens.push(MathToken::BeginMatrix {
227                                left: None,
228                                right: None,
229                            }),
230                            "pmatrix" => tokens.push(MathToken::BeginMatrix {
231                                left: Some('('),
232                                right: Some(')'),
233                            }),
234                            "bmatrix" => tokens.push(MathToken::BeginMatrix {
235                                left: Some('['),
236                                right: Some(']'),
237                            }),
238                            "vmatrix" => tokens.push(MathToken::BeginMatrix {
239                                left: Some('|'),
240                                right: Some('|'),
241                            }),
242                            "Vmatrix" => tokens.push(MathToken::BeginMatrix {
243                                left: Some('\u{2016}'),
244                                right: Some('\u{2016}'),
245                            }),
246                            _ => {
247                                tokens.push(MathToken::Text(format!("\\begin{{{env}}}")));
248                            }
249                        }
250                    } else {
251                        tokens.push(MathToken::Text("\\begin".to_string()));
252                    }
253                }
254                "end" => {
255                    if pos < len && chars[pos] == '{' {
256                        pos += 1;
257                        let env_start = pos;
258                        while pos < len && chars[pos] != '}' {
259                            pos += 1;
260                        }
261                        let env: String = chars[env_start..pos].iter().collect();
262                        if pos < len {
263                            pos += 1;
264                        }
265                        match env.as_str() {
266                            "cases" => tokens.push(MathToken::EndCases),
267                            "matrix" | "pmatrix" | "bmatrix" | "vmatrix" | "Vmatrix" => {
268                                tokens.push(MathToken::EndMatrix)
269                            }
270                            _ => {
271                                tokens.push(MathToken::Text(format!("\\end{{{env}}}")));
272                            }
273                        }
274                    } else {
275                        tokens.push(MathToken::Text("\\end".to_string()));
276                    }
277                }
278                "text" => {
279                    let content = parse_braced_arg(&chars, &mut pos);
280                    tokens.push(MathToken::Text(content));
281                }
282                "pm" => tokens.push(MathToken::BinOp("±".to_string())),
283                "times" => tokens.push(MathToken::BinOp("×".to_string())),
284                "div" => tokens.push(MathToken::BinOp("÷".to_string())),
285                "leq" => tokens.push(MathToken::BinOp("≤".to_string())),
286                "geq" => tokens.push(MathToken::BinOp("≥".to_string())),
287                "neq" => tokens.push(MathToken::BinOp("≠".to_string())),
288                "approx" => tokens.push(MathToken::BinOp("≈".to_string())),
289                "infty" => tokens.push(MathToken::Text("∞".to_string())),
290                _ => {
291                    let display = match cmd.as_str() {
292                        "alpha" => "α",
293                        "beta" => "β",
294                        "gamma" => "γ",
295                        "delta" => "δ",
296                        "epsilon" => "ε",
297                        "zeta" => "ζ",
298                        "eta" => "η",
299                        "theta" => "θ",
300                        "iota" => "ι",
301                        "kappa" => "κ",
302                        "lambda" => "λ",
303                        "mu" => "μ",
304                        "nu" => "ν",
305                        "xi" => "ξ",
306                        "pi" => "π",
307                        "rho" => "ρ",
308                        "sigma" => "σ",
309                        "tau" => "τ",
310                        "upsilon" => "υ",
311                        "phi" => "φ",
312                        "chi" => "χ",
313                        "psi" => "ψ",
314                        "omega" => "ω",
315                        "Gamma" => "Γ",
316                        "Delta" => "Δ",
317                        "Theta" => "Θ",
318                        "Lambda" => "Λ",
319                        "Xi" => "Ξ",
320                        "Pi" => "Π",
321                        "Sigma" => "Σ",
322                        "Phi" => "Φ",
323                        "Psi" => "Ψ",
324                        "Omega" => "Ω",
325                        other => other,
326                    };
327                    if display == cmd.as_str() {
328                        tokens.push(MathToken::Text(format!("\\{}", cmd)));
329                    } else {
330                        tokens.push(MathToken::Text(display.to_string()));
331                    }
332                }
333            }
334        } else if chars[pos] == '^' {
335            pos += 1;
336            let sup_arg = parse_single_arg(&chars, &mut pos);
337            tokens.push(MathToken::Superscript(sup_arg));
338        } else if chars[pos] == '_' {
339            pos += 1;
340            let sub_arg = parse_single_arg(&chars, &mut pos);
341            tokens.push(MathToken::Subscript(sub_arg));
342        } else if chars[pos] == '(' {
343            tokens.push(MathToken::LeftParen);
344            pos += 1;
345        } else if chars[pos] == ')' {
346            tokens.push(MathToken::RightParen);
347            pos += 1;
348        } else if chars[pos] == '[' {
349            tokens.push(MathToken::LeftBracket);
350            pos += 1;
351        } else if chars[pos] == ']' {
352            tokens.push(MathToken::RightBracket);
353            pos += 1;
354        } else if chars[pos] == '&' {
355            tokens.push(MathToken::Ampersand);
356            pos += 1;
357        } else if "+-×÷±=<>≤≥≠≈".contains(chars[pos]) {
358            tokens.push(MathToken::BinOp(chars[pos].to_string()));
359            pos += 1;
360        } else {
361            let start = pos;
362            while pos < len
363                && !chars[pos].is_whitespace()
364                && chars[pos] != '^'
365                && chars[pos] != '_'
366                && chars[pos] != '\\'
367                && chars[pos] != '('
368                && chars[pos] != ')'
369                && chars[pos] != '['
370                && chars[pos] != ']'
371                && chars[pos] != '&'
372                && !"+-×÷±=<>≤≥≠≈".contains(chars[pos])
373            {
374                pos += 1;
375            }
376            let text: String = chars[start..pos].iter().collect();
377            if !text.is_empty() {
378                tokens.push(MathToken::Text(text));
379            }
380        }
381    }
382
383    tokens
384}

386fn shape_text_run(
387    text: &str,
388    font_size: Fp266,
389    font_data: Option<&[u8]>,
390) -> (Vec<PositionedGlyph>, MathBox) {
391    if text.is_empty() {
392        return (Vec::new(), MathBox::zero());
393    }
394
395    let raw_glyphs: Vec<ShapedGlyph> = if let Some(data) = font_data {
396        crate::shaping::shape_text(data, text, font_size).glyphs
397    } else {
398        crate::shaping::fast_path::shape_unicode_basic(&[], text, font_size, 0).glyphs
399    };
400
401    let mut cursor_x = Fp266::ZERO;
402    let mut glyphs = Vec::with_capacity(raw_glyphs.len());
403    for g in &raw_glyphs {
404        let x = cursor_x + g.x_offset;
405        let y = g.y_offset;
406        glyphs.push(PositionedGlyph {
407            glyph_id: g.glyph_id as i32,
408            x,
409            y,
410            advance: g.advance,
411        });
412        cursor_x += g.advance;
413    }
414
415    let total_width = cursor_x;
416    let height = Fp266::from_frac(font_size.to_int() * 8, 10);
417    let depth = Fp266::from_frac(font_size.to_int() * 2, 10);
418
419    (
420        glyphs,
421        MathBox {
422            width: total_width,
423            height,
424            depth,
425        },
426    )
427}

458    fn emit_text(&mut self, text: &str) -> MathBox {
459        let (mut glyphs, box_info) = shape_text_run(text, self.font_size, self.font_data_ref());
460        for g in &mut glyphs {
461            g.x += self.cursor_x;
462            g.y = self.base_y;
463        }
464        let width = box_info.width;
465        self.cursor_x += width;
466        self.glyphs.extend(glyphs);
467        self.pending_base = text.to_string();
468        box_info
469    }

471    fn emit_superscript(&mut self, base: &str, sup: &str) -> MathBox {
472        let base_box = self.emit_text(base);
473        let sup_size = Fp266::from_frac(self.font_size.to_int() * 7, 10);
474        let saved_font_size = self.font_size;
475        let saved_x = self.cursor_x;
476
477        self.font_size = sup_size;
478        let sup_y = self.base_y + Fp266::from_frac(saved_font_size.to_int() * 6, 10);
479        let saved_base_y = self.base_y;
480        self.base_y = sup_y;
481        self.cursor_x = saved_x + base_box.width + Fp266::from_int(1);
482
483        let sup_box = self.emit_text(sup);
484
485        self.font_size = saved_font_size;
486        self.base_y = saved_base_y;
487        self.cursor_x = saved_x;
488
489        let width = base_box.width + Fp266::from_int(1) + sup_box.width;
490        let height = base_box.height + Fp266::from_frac(saved_font_size.to_int() * 4, 10);
491
492        MathBox {
493            width,
494            height,
495            depth: base_box.depth,
496        }
497    }

527    fn emit_fraction(&mut self, numerator: &str, denominator: &str) -> MathBox {
528        let saved_x = self.cursor_x;
529        let saved_base_y = self.base_y;
530
531        let rule_thickness = Fp266::from_frac(self.font_size.to_int(), 16);
532        let rule_gap = Fp266::from_frac(self.font_size.to_int(), 6);
533
534        let num_size = Fp266::from_frac(self.font_size.to_int() * 8, 10);
535        let den_size = Fp266::from_frac(self.font_size.to_int() * 8, 10);
536        let saved_font_size = self.font_size;
537
538        self.font_size = num_size;
539        let (num_glyphs, num_box) = shape_text_run(numerator, num_size, self.font_data_ref());
540        self.font_size = den_size;
541        let (den_glyphs, den_box) = shape_text_run(denominator, den_size, self.font_data_ref());
542
543        self.font_size = saved_font_size;
544
545        let content_width = num_box.width.max(den_box.width);
546        let total_width = content_width + Fp266::from_int(8);
547
548        let num_x_offset = (total_width - num_box.width).div(Fp266::from_int(2));
549        let den_x_offset = (total_width - den_box.width).div(Fp266::from_int(2));
550
551        let rule_y = saved_base_y - rule_thickness;
552        let num_y = rule_y + rule_gap + num_box.height;
553        let den_y = rule_y - rule_gap - den_box.height + den_box.depth;
554
555        for mut g in num_glyphs {
556            g.x += saved_x + num_x_offset;
557            g.y = num_y;
558            self.glyphs.push(g);
559        }
560
561        for mut g in den_glyphs {
562            g.x += saved_x + den_x_offset;
563            g.y = den_y;
564            self.glyphs.push(g);
565        }
566
567        let rule_x =
568            saved_x + (total_width - num_box.width.max(den_box.width)).div(Fp266::from_int(2));
569        let rule_width = num_box.width.max(den_box.width);
570        self.glyphs.push(PositionedGlyph {
571            glyph_id: -1,
572            x: rule_x,
573            y: rule_y,
574            advance: rule_width,
575        });
576
577        self.cursor_x = saved_x + total_width;
578
579        let height = num_box.height + rule_gap + rule_thickness;
580        let depth = den_box.height - den_box.depth + rule_gap + rule_thickness;
581
582        MathBox {
583            width: total_width,
584            height,
585            depth,
586        }
587    }

589    fn emit_radical(&mut self, inner: &MathToken) -> MathBox {
590        let inner_box = self.layout_token(inner);
591        let inner_width = inner_box.width;
592        let rule_thickness = Fp266::from_frac(self.font_size.to_int(), 16);
593        let padding = Fp266::from_int(4);
594
595        let sqrt_symbol = "√";
596        let (mut sqrt_glyphs, sqrt_box) =
597            shape_text_run(sqrt_symbol, self.font_size, self.font_data_ref());
598
599        let total_width = sqrt_box.width + padding + inner_width;
600        let overbar_width = inner_width + Fp266::from_int(2);
601
602        for g in &mut sqrt_glyphs {
603            g.x += self.cursor_x;
604            g.y = self.base_y;
605        }
606        self.glyphs.extend(sqrt_glyphs);
607
608        let overbar_x = self.cursor_x + sqrt_box.width + padding - Fp266::from_int(1);
609        let overbar_y = self.base_y - inner_box.height - rule_thickness;
610
611        self.glyphs.push(PositionedGlyph {
612            glyph_id: -1,
613            x: overbar_x,
614            y: overbar_y,
615            advance: overbar_width,
616        });
617
618        self.cursor_x += total_width;
619
620        let height = inner_box.height + Fp266::from_int(4);
621        MathBox {
622            width: total_width,
623            height,
624            depth: inner_box.depth,
625        }
626    }

629        let display = match op {
630            "sum" => "∑",
631            "int" => "∫",
632            "prod" => "∏",
633            _ => op,

637        let saved_font_size = self.font_size;
638        self.font_size = big_size;

641        for g in &mut glyphs {
642            g.x += self.cursor_x;
643            g.y = self.base_y - Fp266::from_frac(saved_font_size.to_int(), 4);
644        }
645
646        self.font_size = saved_font_size;
647        let width = box_info.width;
648        self.cursor_x += width;
649        self.glyphs.extend(glyphs);
650
651        let height = Fp266::from_frac(saved_font_size.to_int(), 2);
652        let depth = Fp266::from_frac(saved_font_size.to_int(), 2);
653
654        MathBox {
655            width,
656            height,
657            depth,
658        }

683        total_height: Fp266,
684        x: Fp266,
685        y: Fp266,
686    ) -> MathBox {
687        let saved_font_size = self.font_size;
688        let saved_cursor_x = self.cursor_x;
689        let saved_base_y = self.base_y;
690
691        self.font_size = total_height;
692        self.cursor_x = x;
693        self.base_y = y;
694
695        let box_info = self.emit_text(ch);
696
697        self.font_size = saved_font_size;
698        self.cursor_x = saved_cursor_x;
699        self.base_y = saved_base_y;
700
701        box_info
702    }

704    fn layout_token(&mut self, token: &MathToken) -> MathBox {
705        match token {
706            MathToken::Text(text) => self.emit_text(text),
707            MathToken::Superscript(sup) => {
708                if !self.pending_base.is_empty() {
709                    let base = self.pending_base.clone();
710                    self.pending_base.clear();
711                    self.emit_superscript(&base, sup)
712                } else {
713                    self.emit_text(sup)
714                }
715            }
716            MathToken::Subscript(sub) => {
717                if !self.pending_base.is_empty() {
718                    let base = self.pending_base.clone();
719                    self.pending_base.clear();
720                    self.emit_subscript(&base, sub)
721                } else {
722                    self.emit_text(sub)
723                }
724            }
725            MathToken::Fraction(num, den) => self.emit_fraction(num, den),
726            MathToken::Radical(inner) => self.emit_radical(inner),
727            MathToken::BigOp(op) => self.emit_big_op(op),
728            MathToken::BinOp(op) => self.emit_binop(op),
729            MathToken::LeftParen => self.emit_delimiter("("),
730            MathToken::RightParen => self.emit_delimiter(")"),
731            MathToken::LeftBracket => self.emit_delimiter("["),
732            MathToken::RightBracket => self.emit_delimiter("]"),
733            MathToken::BeginCases
734            | MathToken::EndCases
735            | MathToken::BeginMatrix { .. }
736            | MathToken::EndMatrix
737            | MathToken::LeftDelim(_)
738            | MathToken::RightDelim(_)
739            | MathToken::Ampersand
740            | MathToken::RowSeparator => MathBox::zero(),
741        }
742    }

744    fn layout_node(&mut self, node: &MathNode) -> MathBox {
745        match node {
746            MathNode::Text(text) => {
747                self.pending_base.clear();
748                self.emit_text(text)
749            }
750            MathNode::Superscript { base, sup } => {
751                self.pending_base.clear();
752                self.emit_superscript(base, sup)
753            }
754            MathNode::Subscript { base, sub } => {
755                self.pending_base.clear();
756                self.emit_subscript(base, sub)
757            }
758            MathNode::Fraction { num, den } => {
759                self.pending_base.clear();
760                self.emit_fraction(num, den)
761            }
762            MathNode::Radical { inner } => {
763                self.pending_base.clear();
764                self.emit_radical(inner)
765            }
766            MathNode::BigOp(op) => {
767                self.pending_base.clear();
768                self.emit_big_op(op)
769            }
770            MathNode::BinOp(op) => {
771                self.pending_base.clear();
772                self.emit_binop(op)
773            }
774            MathNode::Delimiter(ch) => {
775                self.pending_base.clear();
776                self.emit_delimiter(&ch.to_string())
777            }
778            MathNode::Cases { rows } => {
779                self.pending_base.clear();
780                self.layout_cases(rows)
781            }
782            MathNode::Matrix { rows, delimiters } => {
783                self.pending_base.clear();
784                self.layout_matrix(rows, *delimiters)
785            }
786            MathNode::StretchyGroup {
787                left,
788                content,
789                right,
790            } => {
791                self.pending_base.clear();
792                self.layout_stretchy_group(*left, content, *right)
793            }
794        }
795    }

797    fn layout_nodes(&mut self, nodes: &[MathNode]) -> MathBox {
798        let mut max_height = Fp266::ZERO;
799        let mut max_depth = Fp266::ZERO;
800
801        for node in nodes {
802            let box_info = self.layout_node(node);
803            max_height = max_height.max(box_info.height);
804            max_depth = max_depth.max(box_info.depth);
805        }
806
807        MathBox {
808            width: self.cursor_x,
809            height: max_height,
810            depth: max_depth,
811        }
812    }

814    fn layout_cases(&mut self, rows: &[Vec<Vec<MathNode>>]) -> MathBox {
815        if rows.is_empty() {
816            return MathBox::zero();
817        }
818
819        let saved_cursor_x = self.cursor_x;
820        let saved_base_y = self.base_y;
821        let line_height = Fp266::from_frac(self.font_size.to_int() * 14, 10);
822        let col_gap = Fp266::from_frac(self.font_size.to_int(), 3);
823
824        let mut cell_boxes: Vec<Vec<MathBox>> = Vec::new();
825        let mut max_col_widths: Vec<Fp266> = Vec::new();
826        let mut max_row_height = Fp266::ZERO;
827        let mut max_row_depth = Fp266::ZERO;
828
829        for row in rows {
830            let mut row_measures = Vec::new();
831            for cell_nodes in row {
832                let mut ms = self.measure_state();
833                let box_info = ms.layout_nodes(cell_nodes);
834                row_measures.push(box_info);
835            }
836            while max_col_widths.len() < row_measures.len() {
837                max_col_widths.push(Fp266::ZERO);
838            }
839            for (i, m) in row_measures.iter().enumerate() {
840                max_col_widths[i] = max_col_widths[i].max(m.width);
841            }
842            let rh = row_measures
843                .iter()
844                .map(|m| m.height)
845                .max()
846                .unwrap_or(Fp266::ZERO);
847            let rd = row_measures
848                .iter()
849                .map(|m| m.depth)
850                .max()
851                .unwrap_or(Fp266::ZERO);
852            max_row_height = max_row_height.max(rh);
853            max_row_depth = max_row_depth.max(rd);
854            cell_boxes.push(row_measures);
855        }
856
857        let brace_width = Fp266::from_frac(self.font_size.to_int(), 2);
858        let content_start_x = saved_cursor_x + brace_width + col_gap;
859
860        let total_content_width: Fp266 = max_col_widths
861            .iter()
862            .enumerate()
863            .map(|(i, w)| if i > 0 { *w + col_gap } else { *w })
864            .fold(Fp266::ZERO, |acc, v| acc + v);
865
866        for (row_idx, row) in rows.iter().enumerate() {
867            self.base_y = saved_base_y + Fp266::from_int(row_idx as i32).mul(line_height);
868            let mut cell_x = content_start_x;
869
870            for (col_idx, cell_nodes) in row.iter().enumerate() {
871                self.cursor_x = cell_x;
872                self.layout_nodes(cell_nodes);
873                if col_idx < max_col_widths.len() {
874                    cell_x += max_col_widths[col_idx] + col_gap;
875                }
876            }
877        }
878
879        let total_height = if rows.len() > 1 {
880            Fp266::from_int((rows.len() - 1) as i32).mul(line_height)
881        } else {
882            Fp266::ZERO
883        };
884
885        let brace_total = max_row_height + total_height + max_row_depth;
886        let brace_y = saved_base_y + total_height.div(Fp266::from_int(2));
887        self.emit_scaled_delimiter("{", brace_total, saved_cursor_x, brace_y);
888
889        self.cursor_x = content_start_x + total_content_width;
890
891        MathBox {
892            width: brace_width + col_gap + total_content_width,
893            height: max_row_height + total_height.div(Fp266::from_int(2)),
894            depth: max_row_depth + total_height.div(Fp266::from_int(2)),
895        }
896    }

898    fn layout_matrix(
899        &mut self,
900        rows: &[Vec<Vec<MathNode>>],
901        delimiters: (Option<char>, Option<char>),
902    ) -> MathBox {
903        if rows.is_empty() {
904            return MathBox::zero();
905        }
906
907        let saved_cursor_x = self.cursor_x;
908        let saved_base_y = self.base_y;
909        let line_height = Fp266::from_frac(self.font_size.to_int() * 14, 10);
910        let col_gap = Fp266::from_frac(self.font_size.to_int(), 3);
911        let delim_pad = Fp266::from_frac(self.font_size.to_int(), 2);
912
913        let mut cell_boxes: Vec<Vec<MathBox>> = Vec::new();
914        let mut max_col_widths: Vec<Fp266> = Vec::new();
915        let mut max_row_height = Fp266::ZERO;
916        let mut max_row_depth = Fp266::ZERO;
917
918        for row in rows {
919            let mut row_measures = Vec::new();
920            for cell_nodes in row {
921                let mut ms = self.measure_state();
922                let box_info = ms.layout_nodes(cell_nodes);
923                row_measures.push(box_info);
924            }
925            while max_col_widths.len() < row_measures.len() {
926                max_col_widths.push(Fp266::ZERO);
927            }
928            for (i, m) in row_measures.iter().enumerate() {
929                max_col_widths[i] = max_col_widths[i].max(m.width);
930            }
931            let rh = row_measures
932                .iter()
933                .map(|m| m.height)
934                .max()
935                .unwrap_or(Fp266::ZERO);
936            let rd = row_measures
937                .iter()
938                .map(|m| m.depth)
939                .max()
940                .unwrap_or(Fp266::ZERO);
941            max_row_height = max_row_height.max(rh);
942            max_row_depth = max_row_depth.max(rd);
943            cell_boxes.push(row_measures);
944        }
945
946        let content_start_x = saved_cursor_x + delimiters.0.map_or(Fp266::ZERO, |_| delim_pad);
947        let total_content_width: Fp266 = max_col_widths
948            .iter()
949            .enumerate()
950            .map(|(i, w)| if i > 0 { *w + col_gap } else { *w })
951            .fold(Fp266::ZERO, |acc, v| acc + v);
952
953        for (row_idx, row) in rows.iter().enumerate() {
954            self.base_y = saved_base_y + Fp266::from_int(row_idx as i32).mul(line_height);
955            let mut cell_x = content_start_x;
956
957            for (col_idx, cell_nodes) in row.iter().enumerate() {
958                let col_center_offset = if col_idx < max_col_widths.len()
959                    && row_idx < cell_boxes.len()
960                    && col_idx < cell_boxes[row_idx].len()
961                {
962                    (max_col_widths[col_idx] - cell_boxes[row_idx][col_idx].width)
963                        .div(Fp266::from_int(2))
964                } else {
965                    Fp266::ZERO
966                };
967                self.cursor_x = cell_x + col_center_offset;
968                self.layout_nodes(cell_nodes);
969                if col_idx < max_col_widths.len() {
970                    cell_x += max_col_widths[col_idx] + col_gap;
971                }
972            }
973        }
974
975        let total_height = if rows.len() > 1 {
976            Fp266::from_int((rows.len() - 1) as i32).mul(line_height)
977        } else {
978            Fp266::ZERO
979        };
980
981        let delim_total = max_row_height + total_height + max_row_depth;
982        let delim_y = saved_base_y + total_height.div(Fp266::from_int(2));
983
984        if let Some(left_ch) = delimiters.0 {
985            self.emit_scaled_delimiter(&left_ch.to_string(), delim_total, saved_cursor_x, delim_y);
986        }
987
988        if let Some(right_ch) = delimiters.1 {
989            let right_x = content_start_x + total_content_width + delim_pad;
990            self.emit_scaled_delimiter(&right_ch.to_string(), delim_total, right_x, delim_y);
991        }
992
993        let left_w = delimiters.0.map_or(Fp266::ZERO, |_| delim_pad);
994        let right_w = delimiters.1.map_or(Fp266::ZERO, |_| delim_pad);
995        self.cursor_x = content_start_x + total_content_width + right_w;
996
997        MathBox {
998            width: left_w + total_content_width + right_w,
999            height: max_row_height + total_height.div(Fp266::from_int(2)),
1000            depth: max_row_depth + total_height.div(Fp266::from_int(2)),
1001        }
1002    }

1004    fn layout_stretchy_group(
1005        &mut self,
1006        left: Option<char>,
1007        content: &[MathNode],
1008        right: Option<char>,
1009    ) -> MathBox {
1010        let saved_cursor_x = self.cursor_x;
1011        let saved_base_y = self.base_y;
1012        let delim_pad = Fp266::from_frac(self.font_size.to_int(), 2);
1013
1014        let content_start_x = saved_cursor_x + left.map_or(Fp266::ZERO, |_| delim_pad);
1015        self.cursor_x = content_start_x;
1016        let content_box = self.layout_nodes(content);
1017        let content_width = content_box.width;
1018
1019        let delim_total = content_box.height + content_box.depth;
1020        let delim_y = saved_base_y;
1021
1022        if let Some(left_ch) = left {
1023            self.emit_scaled_delimiter(&left_ch.to_string(), delim_total, saved_cursor_x, delim_y);
1024        }
1025
1026        if let Some(right_ch) = right {
1027            let right_x = content_start_x + content_width + delim_pad;
1028            self.emit_scaled_delimiter(&right_ch.to_string(), delim_total, right_x, delim_y);
1029        }
1030
1031        let left_w = left.map_or(Fp266::ZERO, |_| delim_pad);
1032        let right_w = right.map_or(Fp266::ZERO, |_| delim_pad);
1033        self.cursor_x = content_start_x + content_width + right_w;
1034
1035        MathBox {
1036            width: left_w + content_width + right_w,
1037            height: content_box.height,
1038            depth: content_box.depth,
1039        }
1040    }

1043struct MathTokenParser {
1044    tokens: Vec<MathToken>,
1045    pos: usize,
1046}
1047
1048impl MathTokenParser {
1049    fn new(tokens: Vec<MathToken>) -> Self {
1050        Self { tokens, pos: 0 }
1051    }
1052
1053    fn parse_expression(&mut self) -> Vec<MathNode> {
1054        let mut nodes = Vec::new();
1055        while self.pos < self.tokens.len() && !self.is_stop_token() {
1056            if let Some(node) = self.parse_node() {
1057                nodes.push(node);
1058            }
1059        }
1060        nodes
1061    }
1062
1063    fn is_stop_token(&self) -> bool {
1064        if self.pos >= self.tokens.len() {
1065            return true;
1066        }
1067        matches!(
1068            self.tokens[self.pos],
1069            MathToken::EndCases
1070                | MathToken::EndMatrix
1071                | MathToken::RightDelim(_)
1072                | MathToken::Ampersand
1073                | MathToken::RowSeparator
1074        )
1075    }
1076
1077    fn parse_node(&mut self) -> Option<MathNode> {
1078        if self.is_stop_token() {
1079            return None;
1080        }
1081
1082        let token = self.tokens.get(self.pos)?.clone();
1083        self.pos += 1;
1084
1085        match token {
1086            MathToken::Text(text) => {
1087                if self.pos < self.tokens.len() {
1088                    match &self.tokens[self.pos] {
1089                        MathToken::Subscript(sub) => {
1090                            let sub = sub.clone();
1091                            self.pos += 1;
1092                            return Some(MathNode::Subscript { base: text, sub });
1093                        }
1094                        MathToken::Superscript(sup) => {
1095                            let sup = sup.clone();
1096                            self.pos += 1;
1097                            return Some(MathNode::Superscript { base: text, sup });
1098                        }
1099                        _ => {}
1100                    }
1101                }
1102                Some(MathNode::Text(text))
1103            }
1104            MathToken::Subscript(sub) => Some(MathNode::Text(sub)),
1105            MathToken::Superscript(sup) => Some(MathNode::Text(sup)),
1106            MathToken::Fraction(num, den) => Some(MathNode::Fraction { num, den }),
1107            MathToken::Radical(inner) => Some(MathNode::Radical { inner }),
1108            MathToken::BigOp(op) => Some(MathNode::BigOp(op)),
1109            MathToken::BinOp(op) => Some(MathNode::BinOp(op)),
1110            MathToken::LeftParen => Some(MathNode::Delimiter('(')),
1111            MathToken::RightParen => Some(MathNode::Delimiter(')')),
1112            MathToken::LeftBracket => Some(MathNode::Delimiter('[')),
1113            MathToken::RightBracket => Some(MathNode::Delimiter(']')),
1114            MathToken::BeginCases => {
1115                let rows = self.parse_rows_until_cases();
1116                Some(MathNode::Cases { rows })
1117            }
1118            MathToken::BeginMatrix { left, right } => {
1119                let rows = self.parse_rows_until_matrix();
1120                Some(MathNode::Matrix {
1121                    rows,
1122                    delimiters: (left, right),
1123                })
1124            }
1125            MathToken::LeftDelim(d) => {
1126                let left_char = if d == '.' { None } else { Some(d) };
1127                let content = self.parse_expression();
1128                let right_char = if self.pos < self.tokens.len() {
1129                    if let MathToken::RightDelim(rd) = &self.tokens[self.pos] {
1130                        let r = if *rd == '.' { None } else { Some(*rd) };
1131                        self.pos += 1;
1132                        r
1133                    } else {
1134                        None
1135                    }
1136                } else {
1137                    None
1138                };
1139                Some(MathNode::StretchyGroup {
1140                    left: left_char,
1141                    content,
1142                    right: right_char,
1143                })
1144            }
1145            MathToken::EndCases
1146            | MathToken::EndMatrix
1147            | MathToken::RightDelim(_)
1148            | MathToken::Ampersand
1149            | MathToken::RowSeparator => None,
1150        }
1151    }
1152
1153    fn parse_rows_until_cases(&mut self) -> Vec<Vec<Vec<MathNode>>> {
1154        let mut rows = Vec::new();
1155        rows.push(self.parse_row());
1156        while self.pos < self.tokens.len() {
1157            match &self.tokens[self.pos] {
1158                MathToken::RowSeparator => {
1159                    self.pos += 1;
1160                    rows.push(self.parse_row());
1161                }
1162                MathToken::EndCases => {
1163                    self.pos += 1;
1164                    break;
1165                }
1166                _ => break,
1167            }
1168        }
1169        rows
1170    }
1171
1172    fn parse_rows_until_matrix(&mut self) -> Vec<Vec<Vec<MathNode>>> {
1173        let mut rows = Vec::new();
1174        rows.push(self.parse_row());
1175        while self.pos < self.tokens.len() {
1176            match &self.tokens[self.pos] {
1177                MathToken::RowSeparator => {
1178                    self.pos += 1;
1179                    rows.push(self.parse_row());
1180                }
1181                MathToken::EndMatrix => {
1182                    self.pos += 1;
1183                    break;
1184                }
1185                _ => break,
1186            }
1187        }
1188        rows
1189    }
1190
1191    fn parse_row(&mut self) -> Vec<Vec<MathNode>> {
1192        let mut cells = Vec::new();
1193        cells.push(self.parse_expression());
1194        while self.pos < self.tokens.len() {
1195            if matches!(self.tokens[self.pos], MathToken::Ampersand) {
1196                self.pos += 1;
1197                cells.push(self.parse_expression());
1198            } else {
1199                break;
1200            }
1201        }
1202        cells
1203    }
1204}

1206/// Result of laying out a math expression.
1207pub struct MathLayoutResult {
1208    /// Positioned glyphs (glyph_id == -1 means a horizontal rule).
1209    pub glyphs: Vec<PositionedGlyph>,
1210    /// Total width of the math expression.
1211    pub width: Fp266,
1212    /// Height above the baseline.
1213    pub height: Fp266,
1214    /// Depth below the baseline.
1215    pub depth: Fp266,
1216}
1217
1218/// Layout a math expression string into positioned glyph commands.
1219pub fn layout_math(
1220    content: &str,
1221    font_data: Option<&[u8]>,
1222    font_size: Fp266,
1223    base_y: Fp266,
1224) -> MathLayoutResult {
1225    let tokens = tokenize_math(content);
1226    if tokens.is_empty() {
1227        return MathLayoutResult {
1228            glyphs: Vec::new(),
1229            width: Fp266::ZERO,
1230            height: Fp266::ZERO,
1231            depth: Fp266::ZERO,
1232        };
1233    }
1234
1235    let mut parser = MathTokenParser::new(tokens);
1236    let nodes = parser.parse_expression();
1237    if nodes.is_empty() {
1238        return MathLayoutResult {
1239            glyphs: Vec::new(),
1240            width: Fp266::ZERO,
1241            height: Fp266::ZERO,
1242            depth: Fp266::ZERO,
1243        };
1244    }
1245
1246    let font_data_owned = font_data.map(|d| Arc::new(d.to_vec()));
1247    let mut state = LayoutState::new(font_size, base_y, font_data_owned);
1248
1249    let box_info = state.layout_nodes(&nodes);
1250
1251    MathLayoutResult {
1252        width: box_info.width,
1253        height: box_info.height,
1254        depth: box_info.depth,
1255        glyphs: state.glyphs,
1256    }
1257}
