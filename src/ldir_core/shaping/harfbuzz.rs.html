1//! HarfBuzz text shaping integration (Phase A-2).
2//!
3//! Provides real font-aware text shaping via HarfBuzz FFI.
4//! All unsafe code is confined to this module.
5//!
6//! # Safety
7//!
8//! All `unsafe` blocks in this module call HarfBuzz C FFI functions via
9//! `harfbuzz_sys`. Each call follows HarfBuzz's documented API contracts:
10//!
11//! - `hb_blob_create`: `font_data` pointer is valid for the duration of the
12//!   call (HB_MEMORY_MODE_READONLY does not take ownership). Null destroy
13//!   function is correct since we do not allocate the data.
14//! - `hb_face_create`/`hb_font_create`: take ownership of blob/face refs;
15//!   correctly destroyed via `hb_*_destroy` at end of function.
16//! - `hb_buffer_add_utf8`: text pointer and length are valid; text is not
17//!   mutated during shaping.
18//! - `hb_buffer_get_glyph_infos`/`hb_buffer_get_glyph_positions`: return
19//!   pointers valid until buffer is destroyed. We iterate within bounds
20//!   (`0..glyph_count`) before destroying the buffer.
21//!
22//! Resource cleanup is guaranteed: blob, face, font, and buffer are all
23//! destroyed in a single `unsafe` block at the end of `shape_harfbuzz`,
24//! regardless of intermediate results.
25

34/// OpenType feature to enable/disable during shaping.
35#[derive(Debug, Clone, Copy, PartialEq, Eq)]
36pub struct Feature {
37    pub tag: [u8; 4],
38    pub value: u32,
39    pub start: u32,
40    pub end: u32,
41}
42
43impl Feature {
44    pub fn enable(tag: &[u8; 4]) -> Self {
45        Self {
46            tag: *tag,
47            value: 1,
48            start: 0,
49            end: u32::MAX,
50        }
51    }
52
53    pub fn disable(tag: &[u8; 4]) -> Self {
54        Self {
55            tag: *tag,
56            value: 0,
57            start: 0,
58            end: u32::MAX,
59        }
60    }
61}

70/// Shapes text using HarfBuzz with optional script, language, direction, and feature overrides.
71pub fn shape_harfbuzz(
72    font_data: &[u8],
73    text: &str,
74    font_size: Fp266,
75    script: Option<hb::hb_script_t>,
76    language: Option<&str>,
77    direction: Option<hb::hb_direction_t>,
78    features: Option<&[Feature]>,
79) -> ShapedRun {
80    if text.is_empty() {
81        return ShapedRun::empty();
82    }
83
84    let blob = unsafe {
85        hb::hb_blob_create(
86            font_data.as_ptr() as *const i8,
87            font_data.len() as u32,
88            hb::HB_MEMORY_MODE_READONLY,
89            std::ptr::null_mut(),
90            None,
91        )
92    };
93
94    let face = unsafe { hb::hb_face_create(blob, 0) };
95    let font = unsafe { hb::hb_font_create(face) };
96
97    let (scale, ptem) = match ttf_parser::Face::parse(font_data, 0) {
98        Ok(tp_face) => {
99            let upem = tp_face.units_per_em();
100            let design_size_pt = font_size.to_f64();
101            let s = (design_size_pt * upem as f64 / 72.0) as i32;
102            (s, design_size_pt as f32)
103        }
104        Err(_) => {
105            tracing::warn!(
106                "harfbuzz: failed to parse font with ttf_parser, falling back to raw scale"
107            );
108            (font_size.raw() as i32, font_size.to_f64() as f32)
109        }
110    };
111    unsafe {
112        hb::hb_font_set_scale(font, scale, scale);
113        hb::hb_font_set_ptem(font, ptem);
114    }
115
116    unsafe { hb::hb_ot_font_set_funcs(font) };
117
118    let buffer = unsafe { hb::hb_buffer_create() };
119
120    unsafe {
121        hb::hb_buffer_add_utf8(
122            buffer,
123            text.as_ptr() as *const i8,
124            text.len() as i32,
125            0,
126            text.len() as i32,
127        );
128    }
129
130    if let Some(script) = script {
131        unsafe { hb::hb_buffer_set_script(buffer, script) };
132    }
133    if let Some(lang) = language {
134        let lang_cstr = std::ffi::CString::new(lang).unwrap_or_default();
135        unsafe {
136            hb::hb_buffer_set_language(buffer, hb::hb_language_from_string(lang_cstr.as_ptr(), -1))
137        };
138    }
139    if let Some(dir) = direction {
140        unsafe { hb::hb_buffer_set_direction(buffer, dir) };
141    } else {
142        unsafe { hb::hb_buffer_guess_segment_properties(buffer) };
143    }
144
145    let hb_features: Vec<hb::hb_feature_t> = features
146        .map(|fs| {
147            fs.iter()
148                .map(|f| hb::hb_feature_t {
149                    tag: u32::from_be_bytes(f.tag),
150                    value: f.value,
151                    start: f.start,
152                    end: f.end,
153                })
154                .collect()
155        })
156        .unwrap_or_default();
157
158    let features_ptr = if hb_features.is_empty() {
159        std::ptr::null()
160    } else {
161        hb_features.as_ptr()
162    };
163
164    unsafe {
165        hb::hb_shape(font, buffer, features_ptr, hb_features.len() as u32);
166    }
167
168    let glyph_count = unsafe { hb::hb_buffer_get_length(buffer) } as usize;
169    let glyph_infos = unsafe { hb::hb_buffer_get_glyph_infos(buffer, std::ptr::null_mut()) };
170    let glyph_positions =
171        unsafe { hb::hb_buffer_get_glyph_positions(buffer, std::ptr::null_mut()) };
172
173    let mut glyphs = Vec::with_capacity(glyph_count);
174    let mut total_advance = Fp266::ZERO;
175
176    for i in 0..glyph_count {
177        let info = unsafe { *glyph_infos.add(i) };
178        let pos = unsafe { *glyph_positions.add(i) };
179
180        let x_offset = Fp266::from_raw(pos.x_offset as i64);
181        let y_offset = Fp266::from_raw(pos.y_offset as i64);
182        let advance = Fp266::from_raw(pos.x_advance as i64);
183
184        total_advance += advance;
185
186        glyphs.push(ShapedGlyph {
187            glyph_id: info.codepoint,
188            x_offset,
189            y_offset,
190            advance,
191            cluster_id: info.cluster,
192        });
193    }
194
195    unsafe {
196        hb::hb_buffer_destroy(buffer);
197        hb::hb_font_destroy(font);
198        hb::hb_face_destroy(face);
199        hb::hb_blob_destroy(blob);
200    }
201
202    ShapedRun {
203        glyphs,
204        total_advance,
205    }
206}

216    #[test]
217    fn shape_empty_text() {
218        let data = test_font_data();
219        let run = shape_harfbuzz(&data, "", Fp266::from_int(12), None, None, None, None);
220        assert!(run.glyphs.is_empty());
221        assert!(run.total_advance.is_zero());
222    }
