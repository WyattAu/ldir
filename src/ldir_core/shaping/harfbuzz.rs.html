1//! HarfBuzz text shaping integration (Phase A-2).
2//!
3//! Provides real font-aware text shaping via HarfBuzz FFI.
4//! All unsafe code is confined to this module.
5//!
6//! # Safety
7//!
8//! All `unsafe` blocks in this module call HarfBuzz C FFI functions via
9//! `harfbuzz_sys`. Each call follows HarfBuzz's documented API contracts:
10//!
11//! - `hb_blob_create`: `font_data` pointer is valid for the duration of the
12//!   call (HB_MEMORY_MODE_READONLY does not take ownership). Null destroy
13//!   function is correct since we do not allocate the data.
14//! - `hb_face_create`/`hb_font_create`: take ownership of blob/face refs;
15//!   correctly destroyed via `hb_*_destroy` at end of function.
16//! - `hb_buffer_add_utf8`: text pointer and length are valid; text is not
17//!   mutated during shaping.
18//! - `hb_buffer_get_glyph_infos`/`hb_buffer_get_glyph_positions`: return
19//!   pointers valid until buffer is destroyed. We iterate within bounds
20//!   (`0..glyph_count`) before destroying the buffer.
21//!
22//! Resource cleanup is guaranteed: blob, face, font, and buffer are all
23//! destroyed in a single `unsafe` block at the end of `shape_harfbuzz`,
24//! regardless of intermediate results.
25

33/// OpenType feature to enable/disable during shaping.
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub struct Feature {
36    pub tag: [u8; 4],
37    pub value: u32,
38    pub start: u32,
39    pub end: u32,
40}
41
42impl Feature {
43    pub fn enable(tag: &[u8; 4]) -> Self {
44        Self {
45            tag: *tag,
46            value: 1,
47            start: 0,
48            end: u32::MAX,
49        }
50    }
51
52    pub fn disable(tag: &[u8; 4]) -> Self {
53        Self {
54            tag: *tag,
55            value: 0,
56            start: 0,
57            end: u32::MAX,
58        }
59    }
60
61    pub fn parse_tag(tag_str: &str) -> Option<[u8; 4]> {
62        let bytes = tag_str.as_bytes();
63        if bytes.len() < 2 || bytes.len() > 5 {
64            return None;
65        }
66        let disable = bytes.starts_with(b"-");
67        let inner = if disable { &bytes[1..] } else { bytes };
68        if inner.len() < 2 || inner.len() > 4 || !inner.iter().all(|b| b.is_ascii_alphanumeric()) {
69            return None;
70        }
71        let mut tag = [b' '; 4];
72        let offset = 4usize.saturating_sub(inner.len());
73        tag[offset..].copy_from_slice(inner);
74        Some(tag)
75    }
76
77    pub fn parse_feature_string(s: &str) -> Option<Self> {
78        let s = s.trim();
79        let disable = s.starts_with('-');
80        let inner = if disable { &s[1..] } else { s };
81        let tag = Self::parse_tag(inner)?;
82        if disable {
83            Some(Self::disable(&tag))
84        } else {
85            Some(Self::enable(&tag))
86        }
87    }
88
89    pub fn parse_features(input: &str) -> Vec<Self> {
90        input
91            .split(',')
92            .filter_map(Self::parse_feature_string)
93            .collect()
94    }
95}

128/// Shapes text using HarfBuzz with optional script, language, direction, and feature overrides.
129pub fn shape_harfbuzz(
130    font_data: &[u8],
131    text: &str,
132    font_size: Fp266,
133    script: Option<hb::hb_script_t>,
134    language: Option<&str>,
135    direction: Option<hb::hb_direction_t>,
136    features: Option<&[Feature]>,
137) -> ShapedRun {
138    if text.is_empty() {
139        return ShapedRun::empty();
140    }
141
142    // SAFETY: font_data slice is valid for the call duration; HB_MEMORY_MODE_READONLY
143    // does not take ownership; null destroy function is correct since we don't own the allocation.
144    let blob = unsafe {
145        hb::hb_blob_create(
146            font_data.as_ptr() as *const i8,
147            font_data.len() as u32,
148            hb::HB_MEMORY_MODE_READONLY,
149            std::ptr::null_mut(),
150            None,
151        )
152    };
153
154    // SAFETY: blob is a valid pointer returned by hb_blob_create.
155    let face = unsafe { hb::hb_face_create(blob, 0) };
156    // SAFETY: face is a valid pointer returned by hb_face_create.
157    let font = unsafe { hb::hb_font_create(face) };
158
159    let (scale, ptem) = match ttf_parser::Face::parse(font_data, 0) {
160        Ok(tp_face) => {
161            let upem = tp_face.units_per_em();
162            let design_size_pt = font_size.to_f64();
163            let s = (design_size_pt * upem as f64 / 72.0) as i32;
164            (s, design_size_pt as f32)
165        }
166        Err(_) => {
167            tracing::warn!(
168                "harfbuzz: failed to parse font with ttf_parser, falling back to raw scale"
169            );
170            (font_size.raw() as i32, font_size.to_f64() as f32)
171        }
172    };
173    // SAFETY: font is a valid pointer; scale values are derived from valid font metrics.
174    unsafe {
175        hb::hb_font_set_scale(font, scale, scale);
176        hb::hb_font_set_ptem(font, ptem);
177    }
178
179    // SAFETY: font is a valid pointer; sets OT font funcs for glyph metrics.
180    unsafe { hb::hb_ot_font_set_funcs(font) };
181
182    // SAFETY: returns a valid buffer pointer (never null per HarfBuzz docs).
183    let buffer = unsafe { hb::hb_buffer_create() };
184
185    // SAFETY: buffer is valid, text is valid UTF-8, len matches the string length.
186    unsafe {
187        hb::hb_buffer_add_utf8(
188            buffer,
189            text.as_ptr() as *const i8,
190            text.len() as i32,
191            0,
192            text.len() as i32,
193        );
194    }
195
196    if let Some(script) = script {
197        // SAFETY: buffer is valid, script is a well-known constant.
198        unsafe { hb::hb_buffer_set_script(buffer, script) };
199    }
200    if let Some(lang) = language {
201        let lang_cstr = std::ffi::CString::new(lang).unwrap_or_default();
202        // SAFETY: buffer is valid, lang_cstr is a valid null-terminated C string,
203        // hb_language_from_string handles the pointer safely.
204        unsafe {
205            hb::hb_buffer_set_language(buffer, hb::hb_language_from_string(lang_cstr.as_ptr(), -1))
206        };
207    }
208    if let Some(dir) = direction {
209        // SAFETY: buffer is valid, dir is a well-known constant.
210        unsafe { hb::hb_buffer_set_direction(buffer, dir) };
211    } else {
212        // SAFETY: buffer is valid; guesses script/direction/language from content.
213        unsafe { hb::hb_buffer_guess_segment_properties(buffer) };
214    }
215
216    let hb_features: Vec<hb::hb_feature_t> = features
217        .map(|fs| {
218            fs.iter()
219                .map(|f| hb::hb_feature_t {
220                    tag: u32::from_be_bytes(f.tag),
221                    value: f.value,
222                    start: f.start,
223                    end: f.end,
224                })
225                .collect()
226        })
227        .unwrap_or_default();
228
229    let features_ptr = if hb_features.is_empty() {
230        std::ptr::null()
231    } else {
232        hb_features.as_ptr()
233    };
234
235    // SAFETY: font, buffer, and features_ptr are all valid pointers; len matches the slice.
236    unsafe {
237        hb::hb_shape(font, buffer, features_ptr, hb_features.len() as u32);
238    }
239
240    // SAFETY: buffer was shaped; returns the number of glyphs in the buffer.
241    let glyph_count = unsafe { hb::hb_buffer_get_length(buffer) } as usize;
242    // SAFETY: buffer was shaped; returns valid pointer to glyph info array.
243    let glyph_infos = unsafe { hb::hb_buffer_get_glyph_infos(buffer, std::ptr::null_mut()) };
244    // SAFETY: buffer was shaped; returns valid pointer to glyph position array.
245    let glyph_positions =
246        unsafe { hb::hb_buffer_get_glyph_positions(buffer, std::ptr::null_mut()) };
247
248    let mut glyphs = Vec::with_capacity(glyph_count);
249    let mut total_advance = Fp266::ZERO;
250
251    for i in 0..glyph_count {
252        // SAFETY: i < glyph_count, within bounds of the glyph_infos array.
253        let info = unsafe { *glyph_infos.add(i) };
254        // SAFETY: i < glyph_count, within bounds of the glyph_positions array.
255        let pos = unsafe { *glyph_positions.add(i) };
256
257        let x_offset = Fp266::from_raw(pos.x_offset as i64);
258        let y_offset = Fp266::from_raw(pos.y_offset as i64);
259        let advance = Fp266::from_raw(pos.x_advance as i64);
260
261        total_advance += advance;
262
263        glyphs.push(ShapedGlyph {
264            glyph_id: info.codepoint,
265            x_offset,
266            y_offset,
267            advance,
268            cluster_id: info.cluster,
269        });
270    }
271
272    // SAFETY: buffer, font, face, and blob are all valid pointers owned by us;
273    // each destroy call decrements the refcount and frees when it reaches zero.
274    unsafe {
275        hb::hb_buffer_destroy(buffer);
276        hb::hb_font_destroy(font);
277        hb::hb_face_destroy(face);
278        hb::hb_blob_destroy(blob);
279    }
280
281    ShapedRun {
282        glyphs,
283        total_advance,
284    }
285}
