1//! DEF-004.5 / `hasSingleRoot`: Check that exactly one root node exists.
2//!
3//! Matches Lean 4 predicate:
4//! ```lean
5//! def hasSingleRoot (doc : SIRDocument) : Bool :=
6//!   (doc.filter (fun instr => instr.parent_id == rootSentinel)).length == 1
7//! ```
8
9use ldir_ir::sir::SIRDocument;
10
11use crate::error::{LdirError, ValidationErrorKind};
12
13/// Check that the document has exactly one root node.
14///
15/// Returns errors for:
16/// - **ERR-VALID-006**: No root node found.
17/// - **ERR-VALID-005**: Multiple root nodes found.
18pub fn check(doc: &SIRDocument) -> Vec<LdirError> {
19    let root_count = doc.roots().count();
20
21    match root_count {
22        0 => vec![ValidationErrorKind::NoRoot.into()],
23        1 => Vec::new(),
24        count => vec![ValidationErrorKind::MultipleRoots { count }.into()],
25    }
26}
27
28#[cfg(test)]
29mod tests {
30    use super::*;
31    use ldir_ir::sir::{ROOT_SENTINEL, SIRInstruction, SIROpcode};
32
33    #[test]
34    fn test_single_root_passes() {
35        let mut doc = SIRDocument::new();
36        doc.push(SIRInstruction::new(
37            SIROpcode::PushBlock,
38            0,
39            ROOT_SENTINEL,
40            0,
41        ));
42        doc.push(SIRInstruction::new(SIROpcode::SetContent, 1, 0, 0));
43        assert!(check(&doc).is_empty());
44    }
45
46    #[test]
47    fn test_no_root_fails() {
48        let mut doc = SIRDocument::new();
49        doc.push(SIRInstruction::new(SIROpcode::SetContent, 1, 0, 0));
50        let errors = check(&doc);
51        assert_eq!(errors.len(), 1);
52        match &errors[0].kind {
53            crate::error::ErrorKind::Validation(ValidationErrorKind::NoRoot) => {}
54            other => panic!("expected NoRoot, got {:?}", other),
55        }
56    }
57
58    #[test]
59    fn test_multiple_roots_fails() {
60        let mut doc = SIRDocument::new();
61        doc.push(SIRInstruction::new(
62            SIROpcode::PushBlock,
63            0,
64            ROOT_SENTINEL,
65            0,
66        ));
67        doc.push(SIRInstruction::new(
68            SIROpcode::PushBlock,
69            1,
70            ROOT_SENTINEL,
71            0,
72        ));
73        let errors = check(&doc);
74        assert_eq!(errors.len(), 1);
75        match &errors[0].kind {
76            crate::error::ErrorKind::Validation(ValidationErrorKind::MultipleRoots { count }) => {
77                assert_eq!(*count, 2)
78            }
79            other => panic!("expected MultipleRoots, got {:?}", other),
80        }
81    }
82
83    #[test]
84    fn test_empty_doc_no_root() {
85        let doc = SIRDocument::new();
86        let errors = check(&doc);
87        assert_eq!(errors.len(), 1);
88    }
89}
