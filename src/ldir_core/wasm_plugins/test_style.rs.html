1/// Test plugin: Paragraph style application.
2/// Finds paragraphs matching selectors and applies style properties.
3pub struct ParagraphStylePlugin;
4
5#[derive(Debug, Clone, Default)]
6pub struct StyleOverride {
7    pub font_size: Option<f32>,
8    pub line_height: Option<f32>,
9    pub text_align: Option<TextAlign>,
10    pub color: Option<String>,
11    pub margin_top: Option<f32>,
12    pub margin_bottom: Option<f32>,
13}
14
15#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
16pub enum TextAlign {
17    #[default]
18    Left,
19    Center,
20    Right,
21    Justify,
22}
23
24impl ParagraphStylePlugin {
25    pub fn new() -> Self {
26        Self
27    }
28
29    pub fn apply_style(
30        &self,
31        selector: &str,
32        _style: &StyleOverride,
33        paragraph_count: usize,
34    ) -> Vec<usize> {
35        match selector {
36            "first" if paragraph_count > 0 => vec![0],
37            _ => Vec::new(),
38        }
39    }
40}
41
42impl Default for ParagraphStylePlugin {
43    fn default() -> Self {
44        Self::new()
45    }
46}

48#[cfg(test)]
49mod tests {
50    use super::*;
51
52    #[test]
53    fn test_paragraph_style_plugin_creation() {
54        let plugin = ParagraphStylePlugin::new();
55        let style = StyleOverride {
56            font_size: Some(14.0),
57            ..Default::default()
58        };
59        let matched = plugin.apply_style("first", &style, 5);
60        assert_eq!(matched, vec![0]);
61    }
62
63    #[test]
64    fn test_style_override_fields() {
65        let style = StyleOverride {
66            font_size: Some(12.0),
67            line_height: Some(1.5),
68            text_align: Some(TextAlign::Center),
69            color: Some("#000000".to_string()),
70            margin_top: Some(8.0),
71            margin_bottom: Some(8.0),
72        };
73        assert_eq!(style.font_size, Some(12.0));
74        assert_eq!(style.line_height, Some(1.5));
75        assert_eq!(style.text_align, Some(TextAlign::Center));
76        assert_eq!(style.color.as_deref(), Some("#000000"));
77        assert_eq!(style.margin_top, Some(8.0));
78        assert_eq!(style.margin_bottom, Some(8.0));
79    }
80
81    #[test]
82    fn test_style_override_default() {
83        let style = StyleOverride::default();
84        assert!(style.font_size.is_none());
85        assert!(style.line_height.is_none());
86        assert_eq!(style.text_align, None);
87        assert!(style.color.is_none());
88        assert!(style.margin_top.is_none());
89        assert!(style.margin_bottom.is_none());
90    }
91
92    #[test]
93    fn test_no_match_unknown_selector() {
94        let plugin = ParagraphStylePlugin::new();
95        let style = StyleOverride::default();
96        let matched = plugin.apply_style("unknown", &style, 5);
97        assert!(matched.is_empty());
98    }
99
100    #[test]
101    fn test_no_match_empty_paragraphs() {
102        let plugin = ParagraphStylePlugin::new();
