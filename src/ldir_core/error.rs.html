1//! Error types for LDIR compilation pipeline.
2//!
3//! Structured error hierarchy per BP-IR-COMPILER-001.
4//! Each error carries EntityID and byte offset per REQ-3.3.4.
5
6use std::fmt;
7
8/// Entity identifier type (u32 per REQ-3.1.6).
9pub type EntityId = u32;
10
11/// Byte offset in the input stream.
12pub type ByteOffset = u32;
13
14/// Parse error codes (ERR-PARSE-001..003).
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub enum ParseErrorKind {
17    /// ERR-PARSE-001: Input too short (< 13 bytes minimum instruction size).
18    InputTooShort {
19        /// Actual byte length of the input.
20        len: usize,
21    },
22    /// ERR-PARSE-002: Input not 4-byte aligned.
23    AlignmentError {
24        /// Byte offset of the alignment violation.
25        offset: ByteOffset,
26    },
27    /// ERR-PARSE-003: Invalid opcode byte.
28    InvalidOpcode {
29        /// The invalid byte value encountered.
30        byte: u8,
31        /// Byte offset where the invalid opcode was found.
32        offset: ByteOffset,
33    },
34    /// rkyv deserialization failure.
35    DeserializationError {
36        /// Human-readable error message from the deserializer.
37        message: String,
38    },
39}
40
41/// Validation error codes (ERR-VALID-001..007).
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub enum ValidationErrorKind {
44    /// ERR-VALID-001: Duplicate entity ID.
45    DuplicateEntityId {
46        /// The entity ID that was duplicated.
47        entity_id: EntityId,
48    },
49    /// ERR-VALID-002: Parent reference does not exist.
50    ParentNotFound {
51        /// The entity whose parent is missing.
52        entity_id: EntityId,
53        /// The parent ID that was not found.
54        parent_id: EntityId,
55    },
56    /// ERR-VALID-003: Circular parent chain detected.
57    CircularParentChain {
58        /// The entity at which the cycle was detected.
59        entity_id: EntityId,
60    },
61    /// ERR-VALID-004: Payload region out of bounds.
62    PayloadOutOfBounds {
63        /// The entity with the invalid payload offset.
64        entity_id: EntityId,
65        /// The byte offset that is out of bounds.
66        offset: ByteOffset,
67    },
68    /// ERR-VALID-005: Multiple root nodes.
69    MultipleRoots {
70        /// The number of root nodes found.
71        count: usize,
72    },
73    /// ERR-VALID-006: No root node found.
74    NoRoot,
75    /// ERR-VALID-007: Invalid block nesting.
76    InvalidBlockNesting {
77        /// The entity with invalid block nesting.
78        entity_id: EntityId,
79    },
80}

82/// Compile error codes (ERR-COMP-001..002).
83#[derive(Debug, Clone, PartialEq, Eq)]
84pub enum CompileErrorKind {
85    /// ERR-COMP-001: Stack overflow in compilation.
86    StackOverflow {
87        /// The stack depth at which overflow was detected.
88        depth: usize,
89    },
90    /// ERR-COMP-002: Unsupported instruction in context.
91    UnsupportedInstruction {
92        /// The entity with the unsupported instruction.
93        entity_id: EntityId,
94    },
95}
96
97/// Emit error code (ERR-EMIT-001).
98#[derive(Debug, Clone, PartialEq, Eq)]
99pub enum EmitErrorKind {
100    /// ERR-EMIT-001: G-IR command too large for buffer.
101    BufferOverflow {
102        /// The number of bytes required.
103        required: usize,
104        /// The number of bytes available.
105        available: usize,
106    },
107}

109/// A structured error with location information.
110#[derive(Debug, Clone)]
111pub struct LdirError {
112    /// The specific error kind.
113    pub kind: ErrorKind,
114    /// Optional entity ID associated with this error.
115    pub entity_id: Option<EntityId>,
116    /// Optional byte offset in the input stream.
117    pub byte_offset: Option<ByteOffset>,
118}
119
120/// Top-level error category for LDIR operations.
121#[derive(Debug, Clone, PartialEq, Eq)]
122pub enum ErrorKind {
123    /// A parse-phase error (IF-PARSE-001).
124    Parse(ParseErrorKind),
125    /// A validation-phase error (IF-VALIDATE-001).
126    Validation(ValidationErrorKind),
127    /// A compile-phase error (IF-COMPILE-001).
128    Compile(CompileErrorKind),
129    /// An emit-phase error (IF-EMIT-001).
130    Emit(EmitErrorKind),
131}
132
133impl fmt::Display for LdirError {
134    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
135        let loc = match (self.entity_id, self.byte_offset) {
136            (Some(eid), Some(off)) => format!(" [entity={eid}, offset={off}]"),
137            (Some(eid), None) => format!(" [entity={eid}]"),
138            (None, Some(off)) => format!(" [offset={off}]"),
139            (None, None) => String::new(),
140        };
141        match &self.kind {
142            ErrorKind::Parse(e) => write!(f, "Parse error{loc}: {e:?}"),
143            ErrorKind::Validation(e) => write!(f, "Validation error{loc}: {e:?}"),
144            ErrorKind::Compile(e) => write!(f, "Compile error{loc}: {e:?}"),
145            ErrorKind::Emit(e) => write!(f, "Emit error{loc}: {e:?}"),
146        }
147    }
148}

196impl LdirError {
