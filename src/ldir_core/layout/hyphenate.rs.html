1//! Pattern-based hyphenation for multiple languages.
2//!
3//! Uses a simplified algorithmic hyphenator that splits words at syllable
4//! boundaries using common patterns, affix rules, and user-specified
5//! hyphenation points. Supports English, German, French, Spanish, and
6//! Portuguese.
7
8/// A position within a word where a hyphen can be inserted.
9#[derive(Debug, Clone, PartialEq, Eq)]
10pub struct HyphenPoint {
11    /// Byte offset in the word where a hyphen can be inserted.
12    pub position: usize,
13    /// Quality of this hyphenation point.
14    pub quality: HyphenQuality,
15}
16
17/// How good a hyphenation break point is.
18#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
19pub enum HyphenQuality {
20    /// Natural syllable boundary (best).
21    Excellent,
22    /// Common affix boundary.
23    Good,
24    /// Pattern-based hyphenation (Liou algorithm).
25    Pattern,
26    /// Algorithmic guess (worst acceptable).
27    Acceptable,
28}
29
30#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
31pub enum HyphenationLang {
32    English,
33    German,
34    French,
35    Spanish,
36    Portuguese,
37    /// Use language-agnostic syllable heuristic only
38    Unknown,
39}

71fn lang_config(lang: HyphenationLang) -> LangConfig {
72    match lang {
73        HyphenationLang::English => LangConfig {
74            prefixes: ENGLISH_PREFIXES,
75            suffixes: ENGLISH_SUFFIXES,
76            min_word_len: 5,
77            protect_margin: 3,
78        },
79        HyphenationLang::German => LangConfig {
80            prefixes: GERMAN_PREFIXES,
81            suffixes: GERMAN_SUFFIXES,
82            min_word_len: 5,
83            protect_margin: 2,
84        },
85        HyphenationLang::French => LangConfig {
86            prefixes: FRENCH_PREFIXES,
87            suffixes: FRENCH_SUFFIXES,
88            min_word_len: 6,
89            protect_margin: 3,
90        },
91        HyphenationLang::Spanish => LangConfig {
92            prefixes: SPANISH_PREFIXES,
93            suffixes: SPANISH_SUFFIXES,
94            min_word_len: 5,
95            protect_margin: 3,
96        },
97        HyphenationLang::Portuguese => LangConfig {
98            prefixes: PORTUGUESE_PREFIXES,
99            suffixes: PORTUGUESE_SUFFIXES,
100            min_word_len: 5,
101            protect_margin: 3,
102        },
103        HyphenationLang::Unknown => LangConfig {
104            prefixes: &[],
105            suffixes: &[],
106            min_word_len: 5,
107            protect_margin: 3,
108        },
109    }

115fn is_vowel(c: char, lang: HyphenationLang) -> bool {
116    match c {
117        'a' | 'e' | 'i' | 'o' | 'u' | 'y' | 'A' | 'E' | 'I' | 'O' | 'U' | 'Y' => true,
118        _ => match lang {
119            HyphenationLang::French | HyphenationLang::Spanish | HyphenationLang::Portuguese => {
120                matches!(c, '\u{00E0}'..='\u{00F6}' | '\u{00F8}'..='\u{00FF}')
121            }
122            HyphenationLang::German => {
123                matches!(
124                    c,
125                    '\u{00E4}' | '\u{00F6}' | '\u{00FC}' | '\u{00C4}' | '\u{00D6}' | '\u{00DC}'
126                )
127            }
128            _ => false,
129        },
130    }
131}

133fn char_byte_offset(chars: &[char], char_index: usize) -> usize {
134    chars[..char_index].iter().map(|c| c.len_utf8()).sum()
135}

137fn hyphenate_by_syllables(word: &str, lang: HyphenationLang) -> Vec<(usize, HyphenQuality)> {
138    let cfg = lang_config(lang);
139    let chars: Vec<char> = word.chars().collect();
140    if chars.len() < cfg.min_word_len {
141        return vec![];
142    }
143
144    let mut points: Vec<(usize, HyphenQuality)> = Vec::new();
145
146    let margin = if chars.len() <= 7 {
147        2
148    } else {
149        cfg.protect_margin
150    };
151
152    for i in margin..chars.len() - margin {
153        let prev = chars[i - 1];
154        let curr = chars[i];
155        let next = chars[i + 1];
156
157        let prev_is_vowel = is_vowel(prev, lang);
158        let curr_is_vowel = is_vowel(curr, lang);
159        let next_is_vowel = is_vowel(next, lang);
160        let after_next_vowel = i + 2 < chars.len() && is_vowel(chars[i + 2], lang);
161
162        if prev_is_vowel && !curr_is_vowel && (next_is_vowel || after_next_vowel) {
163            points.push((char_byte_offset(&chars, i), HyphenQuality::Excellent));
164        }
165        if !prev_is_vowel && prev == curr && i < chars.len() - margin {
166            points.push((char_byte_offset(&chars, i), HyphenQuality::Good));
167        }
168    }
169
170    points
171}

173fn hyphenate_by_affixes(word: &str, lang: HyphenationLang) -> Vec<(usize, HyphenQuality)> {
174    let cfg = lang_config(lang);
175    let chars: Vec<char> = word.chars().collect();
176    if chars.len() < cfg.min_word_len {
177        return vec![];
178    }
179
180    let mut points: Vec<(usize, HyphenQuality)> = Vec::new();
181    let margin = if chars.len() <= 7 {
182        2
183    } else {
184        cfg.protect_margin
185    };
186
187    for &prefix in cfg.prefixes {
188        if word.starts_with(prefix) && prefix.len() + 2 < word.len() {
189            let pos = prefix.len();
190            if pos >= margin && pos <= chars.len() - margin {
191                points.push((char_byte_offset(&chars, pos), HyphenQuality::Good));
192            }
193        }
194    }
195
196    for &suffix in cfg.suffixes {
197        if word.ends_with(suffix) && word.len() > suffix.len() + cfg.protect_margin {
198            let pos = chars.len() - suffix.len();
199            if pos >= margin && pos <= chars.len() - margin {
200                points.push((char_byte_offset(&chars, pos), HyphenQuality::Good));
201            }
202        }
203    }
204
205    points
206}

208fn hyphenate_user_specified(word: &str, lang: HyphenationLang) -> Vec<(usize, HyphenQuality)> {
209    let cfg = lang_config(lang);
210    let chars: Vec<char> = word.chars().collect();
211    if chars.len() < cfg.min_word_len {
212        return vec![];
213    }
214
215    let mut points: Vec<(usize, HyphenQuality)> = Vec::new();
216
217    for (i, window) in chars.windows(2).enumerate() {
218        if window[0] == '\\' && window[1] == '-' {
219            let byte_pos = char_byte_offset(&chars, i);
220            if i >= cfg.protect_margin && i < chars.len() - cfg.protect_margin {
221                points.push((byte_pos, HyphenQuality::Excellent));
222            }
223        }
224    }
225
226    points
227}

229/// Hyphenate a word with the given language, returning all valid hyphenation
230/// points.
231///
232/// Handles:
233/// - User-specified hyphenation points (`\-`)
234/// - Common prefixes and suffixes for the given language
235/// - Syllable boundary detection using vowel/consonant patterns
236///
237/// Words shorter than the language's minimum length are never hyphenated.
238/// The first and last characters are always protected according to the
239/// language's margin.
240pub fn hyphenate_word_with_lang(word: &str, lang: HyphenationLang) -> Vec<HyphenPoint> {
241    let cfg = lang_config(lang);
242    if word.len() < cfg.min_word_len
243        || !word
244            .chars()
245            .all(|c| c.is_alphabetic() || c == '\\' || c == '-')
246    {
247        return vec![];
248    }
249
250    let chars: Vec<char> = word.chars().filter(|c| *c != '\\' && *c != '-').collect();
251    if chars.len() < cfg.min_word_len {
252        return vec![];
253    }
254
255    let mut all_points: Vec<(usize, HyphenQuality)> = Vec::new();
256
257    all_points.extend(hyphenate_user_specified(word, lang));
258    all_points.extend(hyphenate_by_affixes(word, lang));
259    all_points.extend(hyphenate_by_syllables(word, lang));
260
261    all_points.sort_by_key(|&(pos, _)| pos);
262    all_points.dedup_by(|a, b| {
263        if a.0 == b.0 {
264            if a.1 < b.1 {
265                *a = *b;
266            }
267            true
268        } else {
269            false
270        }
271    });
272
273    all_points
274        .into_iter()
275        .map(|(position, quality)| HyphenPoint { position, quality })
276        .collect()
277}

279/// Hyphenate a word, returning all valid hyphenation points.
280///
281/// Equivalent to calling [`hyphenate_word_with_lang`] with
282/// [`HyphenationLang::English`].
283///
284/// Handles:
285/// - User-specified hyphenation points (`\-`)
286/// - Common English prefixes and suffixes
287/// - Syllable boundary detection using vowel/consonant patterns
288///
289/// Words shorter than 5 characters are never hyphenated.
290/// The first and last 3 characters are always protected.
291pub fn hyphenate_word(word: &str) -> Vec<HyphenPoint> {
292    hyphenate_word_with_lang(word, HyphenationLang::English)
293}

299const MIN_PATTERN_WORD_LEN: usize = 5;

301/// A compiled Liou pattern trie for hyphenation.
302struct HyphenPatterns {
303    patterns: Vec<(Vec<char>, Vec<u8>)>,
304}
305
306impl HyphenPatterns {
307    fn from_pattern_data(data: &str) -> Self {
308        let mut patterns = Vec::new();
309        for line in data.lines() {
310            let line = line.trim();
311            if line.is_empty() || line.starts_with('%') || line.starts_with('#') {
312                continue;
313            }
314            for token in line.split_whitespace() {
315                if let Some(pair) = parse_liou_pattern_token(token) {
316                    patterns.push(pair);
317                }
318            }
319        }
320        HyphenPatterns { patterns }
321    }
322}

324/// Parse a single TeX hyphenation pattern token.
325///
326/// Format: characters interleaved with digit levels. For example, `a3b2c2`
327/// means: at position 0 (before 'a') level is 0 (implicit), at position 1
328/// (between 'a' and 'b') level is 3, at position 2 (between 'b' and 'c')
329/// level is 2, at position 3 (after 'c') level is 2.
330/// Digits 0-9 encode hyphenation levels. Odd levels indicate a valid break.
331fn parse_liou_pattern_token(token: &str) -> Option<(Vec<char>, Vec<u8>)> {
332    if token.is_empty() {
333        return None;
334    }
335
336    let mut chars = Vec::new();
337    let mut levels = Vec::new();
338
339    let mut digit_buf = String::new();
340
341    for c in token.chars() {
342        if c.is_ascii_digit() {
343            digit_buf.push(c);
344        } else if c == '.' || c.is_ascii_alphabetic() {
345            if !digit_buf.is_empty() {
346                while levels.len() < chars.len() {
347                    levels.push(0);
348                }
349                for d in digit_buf.drain(..) {
350                    levels.push(d.to_digit(10)? as u8);
351                }
352            }
353            chars.push(c);
354        } else {
355            return None;
356        }
357    }
358
359    if !digit_buf.is_empty() {
360        while levels.len() < chars.len() {
361            levels.push(0);
362        }
363        for d in digit_buf.drain(..) {
364            levels.push(d.to_digit(10)? as u8);
365        }
366    }
367
368    // levels should be chars.len() + 1 positions
369    while levels.len() < chars.len() + 1 {
370        levels.push(0);
371    }
372    if levels.len() > chars.len() + 1 {
373        levels.truncate(chars.len() + 1);
374    }
375
376    if chars.is_empty() {
377        None
378    } else {
379        Some((chars, levels))
380    }
381}

429/// Hyphenate a word using Liou pattern matching.
430fn hyphenate_with_patterns(word: &str, patterns: &HyphenPatterns) -> Vec<HyphenPoint> {
431    let chars: Vec<char> = word.chars().collect();
432    if chars.len() < MIN_PATTERN_WORD_LEN {
433        return vec![];
434    }
435
436    let word_lower: Vec<char> = word.chars().map(|c| c.to_ascii_lowercase()).collect();
437
438    let mut levels = vec![0u8; word_lower.len() + 1];
439
440    for (pattern_chars, pattern_levels) in &patterns.patterns {
441        let wlen = word_lower.len();
442
443        let mut char_positions: Vec<(usize, char)> = Vec::new();
444        for (pi, &pc) in pattern_chars.iter().enumerate() {
445            if pc == '.' {
446                continue;
447            }
448            char_positions.push((pi, pc));
449        }
450        if char_positions.is_empty() {
451            continue;
452        }
453        let char_count = char_positions.len();
454        let first_non_dot = char_positions[0].0;
455        let _last_non_dot = char_positions[char_count - 1].0;
456
457        for offset in 0..=(wlen.saturating_sub(char_count)) {
458            let mut matched = true;
459            for &(pi, pc) in &char_positions {
460                let wi = offset + pi - first_non_dot;
461                if wi >= wlen || word_lower[wi] != pc {
462                    matched = false;
463                    break;
464                }
465            }
466            if !matched {
467                continue;
468            }
469
470            let word_offset = offset.saturating_sub(first_non_dot);
471            for (i, &level) in pattern_levels.iter().enumerate() {
472                let pos = word_offset + i;
473                if pos < levels.len() && level > levels[pos] {
474                    levels[pos] = level;
475                }
476            }
477        }
478    }
479
480    let mut points = Vec::new();
481    for i in 1..levels.len() - 1 {
482        if levels[i] > 0 && levels[i] % 2 == 1 {
483            let byte_offset: usize = chars[..i].iter().map(|c| c.len_utf8()).sum();
484            points.push(HyphenPoint {
485                position: byte_offset,
486                quality: HyphenQuality::Pattern,
487            });
488        }
489    }
490    points
491}

493/// Hyphenate a word using Liou patterns if available, otherwise fall back to
494/// the heuristic engine.
495///
496/// This is an opt-in alternative to [`hyphenate_word`]. Callers who want
497/// pattern-based hyphenation should prefer this function.
498pub fn hyphenate_word_with_patterns(word: &str) -> Vec<HyphenPoint> {
499    let embedded = HyphenPatterns::from_pattern_data(EMBEDDED_ENGLISH_PATTERNS);
500    if embedded.patterns.is_empty() {
501        hyphenate_word(word)
502    } else {
503        let pattern_points = hyphenate_with_patterns(word, &embedded);
504        if pattern_points.is_empty() {
505            hyphenate_word(word)
506        } else {
507            pattern_points
508        }
509    }
510}
