1//! Badness and adjustment ratio computation for line breaking.
2//!
3//! References: YP-LAYOUT-KNUTHPLASS-001, DEF-BADNESS, DEF-ADJ-RATIO.
4//!
5//! ## SIMD Batch Processing (U-2)
6//!
7//! `compute_demerits_batch_simd` processes 4 candidate breaks simultaneously
8//! using `f64x4` SIMD. Callers should gather 4 candidates' parameters
9//! into SIMD vectors, call the batch function, then scatter results.
10//!
11//! Note: SIMD intrinsics require `unsafe` blocks. These are isolated to the
12//! batch functions and justified by the performance-critical nature of the
13//! Knuth-Plass DP inner loop.
14
15#![allow(unsafe_code)]
16
17use crate::fp266::Fp266;

19/// Compute the adjustment ratio for a line.
20///
21/// - If gap > 0 (line needs stretching): r = gap / stretch
22/// - If gap < 0 (line needs shrinking):  r = gap / shrink
23/// - If gap = 0: r = 0 (perfect fit)
24///
25/// When stretch or shrink is zero and the corresponding gap direction applies:
26/// - gap > 0 with no stretch: feasible (r = 0, extra space goes to margin)
27/// - gap < 0 with no shrink: infeasible (r = -∞, content overflows)
28///
29/// See YP-LAYOUT-KNUTHPLASS-001 DEF-ADJ-RATIO.
30#[inline]
31pub fn compute_adjustment_ratio(
32    line_width: Fp266,
33    content_width: Fp266,
34    total_stretch: Fp266,
35    total_shrink: Fp266,
36) -> f64 {
37    let gap = (line_width - content_width).to_f64();
38
39    if gap.abs() < 1e-10 {
40        return 0.0;
41    }
42
43    if gap > 0.0 {
44        // Line needs stretching
45        let s = total_stretch.to_f64();
46        if s < 1e-10 {
47            // No stretch available — extra space goes to margin, feasible
48            0.0
49        } else {
50            gap / s
51        }
52    } else {
53        // Line needs shrinking (gap < 0)
54        let s = total_shrink.to_f64();
55        if s < 1e-10 {
56            // No shrink available — content overflows, infeasible
57            f64::NEG_INFINITY
58        } else {
59            gap / s
60        }
61    }
62}

64/// Compute the badness of a line given its adjustment ratio.
65///
66/// b = 100 * |r|^3
67///
68/// A badness of 0 means perfect fit. Higher badness means worse fit.
69/// See YP-LAYOUT-KNUTHPLASS-001 DEF-BADNESS.
70#[inline]
71pub fn compute_badness(r: f64) -> f64 {
72    100.0 * r.abs().powi(3)
73}
74
75/// Compute the demerits for a single line break.
76///
77/// d = (line_penalty + badness)^2 + fitness_penalty * (1 if fitness class changes)
78///
79/// See YP-LAYOUT-KNUTHPLASS-001 DEF-DEMERITS.
80#[inline]
81pub fn compute_demerits(
82    badness: f64,
83    fitness_penalty: f64,
84    line_penalty: f64,
85    fitness_changed: bool,
86) -> f64 {
87    let line_demerit = line_penalty + badness;
88    let fitness_demerit = if fitness_changed {
89        fitness_penalty
90    } else {
91        0.0
92    };
93    line_demerit * line_demerit + fitness_demerit * fitness_demerit
94}
95
96/// Compute badness for 4 ratios simultaneously using SIMD.
97///
98/// Returns `[b0, b1, b2, b3]` where `bi = 100 * |ri|^3`.
99#[inline]
100#[cfg(target_arch = "x86_64")]
101#[allow(dead_code, clippy::unnecessary_cast)]
102#[target_feature(enable = "sse2")]
103pub unsafe fn compute_badness_batch_simd(r0: f64, r1: f64, r2: f64, r3: f64) -> [f64; 4] {
104    let v = [r0, r1, r2, r3];
105    v.map(|ri| 100.0 * ri.abs().powi(3))
106}
107
108/// Compute demerits for 4 candidate breaks simultaneously.
109///
110/// All input arrays must have exactly 4 elements.
111/// Returns `[d0, d1, d2, d3]` where `di` is the demerit for candidate i.
112///
113/// This is a scalar batch (not SIMD) — it processes 4 candidates without
114/// branching to enable future SIMD optimization and to amortize function
115/// call overhead in the DP inner loop.
116#[inline]
117#[allow(dead_code)]
118pub fn compute_demerits_batch(
119    badness: [f64; 4],
120    fitness_penalty: f64,
121    line_penalty: f64,
122    fitness_changed: [bool; 4],
123) -> [f64; 4] {
124    std::array::from_fn(|i| {
125        compute_demerits(
126            badness[i],
127            fitness_penalty,
128            line_penalty,
129            fitness_changed[i],
130        )
131    })
132}

134/// SIMD-accelerated demerits computation for a batch of 4 candidates.
135///
136/// Processes 4 candidates using `f64x4` SIMD instructions when available.
137/// Falls back to scalar on non-x86_64 targets.
138///
139/// # Safety
140///
141/// The `#[target_feature(enable = "sse2")]` attribute means this function
142/// must only be called on CPUs that support SSE2 (all x86_64 CPUs do).
143#[inline]
144#[cfg(target_arch = "x86_64")]
145#[target_feature(enable = "sse2")]
146#[allow(dead_code, clippy::unnecessary_cast)]
147pub unsafe fn compute_demerits_batch_simd(
148    badness: [f64; 4],
149    fitness_penalty: f64,
150    line_penalty: f64,
151    fitness_changed: [bool; 4],
152) -> [f64; 4] {
153    unsafe {
154        use std::arch::x86_64::*;
155
156        // Load badness values — _mm_loadu_pd loads [ptr[0], ptr[1]]
157        let b_lo = _mm_loadu_pd(badness.as_ptr() as *const f64);
158        let b_hi = _mm_loadu_pd(badness.as_ptr().add(2) as *const f64);
159
160        // line_demerit = line_penalty + badness
161        let lp = _mm_set1_pd(line_penalty);
162        let ld_lo = _mm_add_pd(lp, b_lo);
163        let ld_hi = _mm_add_pd(lp, b_hi);
164
165        // line_demerit^2
166        let sq_lo = _mm_mul_pd(ld_lo, ld_lo);
167        let sq_hi = _mm_mul_pd(ld_hi, ld_hi);
168
169        // fitness_demerit = fitness_penalty if changed, else 0
170        // Note: _mm_set_pd(a, b) stores a in high bits, b in low bits.
171        // Memory order is [low, high] = [b, a].
172        let fd_lo = _mm_set_pd(
173            if fitness_changed[1] {
174                fitness_penalty
175            } else {
176                0.0
177            },
178            if fitness_changed[0] {
179                fitness_penalty
180            } else {
181                0.0
182            },
183        );
184        let fd_hi = _mm_set_pd(
185            if fitness_changed[3] {
186                fitness_penalty
187            } else {
188                0.0
189            },
190            if fitness_changed[2] {
191                fitness_penalty
192            } else {
193                0.0
194            },
195        );
196
197        // fitness_demerit^2
198        let fd_sq_lo = _mm_mul_pd(fd_lo, fd_lo);
199        let fd_sq_hi = _mm_mul_pd(fd_hi, fd_hi);
200
201        // total = line_demerit^2 + fitness_demerit^2
202        let total_lo = _mm_add_pd(sq_lo, fd_sq_lo);
203        let total_hi = _mm_add_pd(sq_hi, fd_sq_hi);
204
205        // Store results
206        let mut result = [0.0f64; 4];
207        let p = result.as_mut_ptr() as *mut f64;
208        _mm_storeu_pd(p, total_lo);
209        _mm_storeu_pd(p.add(2), total_hi);
210        result
211    }
212}

214/// Portable fallback for non-x86_64 targets.
215#[cfg(not(target_arch = "x86_64"))]
216#[allow(dead_code)]
217pub fn compute_demerits_batch_simd(
218    badness: [f64; 4],
219    fitness_penalty: f64,
220    line_penalty: f64,
221    fitness_changed: [bool; 4],
222) -> [f64; 4] {
223    compute_demerits_batch(badness, fitness_penalty, line_penalty, fitness_changed)
224}

232        let r = compute_adjustment_ratio(
233            Fp266::from_int(100),

236            Fp266::from_int(10),

300    #[test]
301    fn test_batch_demerits_scalar() {
302        let badness = [0.0, 0.5, 1.0, 0.0];
303        let fitness_changed = [false, true, false, true];
304        let result = compute_demerits_batch(badness, 100.0, 10.0, fitness_changed);
305        // Each element should match the scalar version
306        for i in 0..4 {
307            let expected = compute_demerits(badness[i], 100.0, 10.0, fitness_changed[i]);
308            assert!(
309                (result[i] - expected).abs() < 1e-10,
310                "batch[{}] = {} != scalar = {}",
311                i,
312                result[i],
313                expected
314            );
315        }
316    }

319    fn test_batch_simd_matches_scalar() {
320        let badness = [0.0, 0.5, 1.0, 0.25];
321        let fitness_changed = [false, true, false, true];
322        let scalar = compute_demerits_batch(badness, 100.0, 10.0, fitness_changed);
323        let simd = unsafe { compute_demerits_batch_simd(badness, 100.0, 10.0, fitness_changed) };

335    #[test]
336    fn test_batch_simd_infeasible() {
337        let badness = [f64::INFINITY, f64::NEG_INFINITY, 0.0, 1e6];
338        let fitness_changed = [false, false, false, false];
339        let result = unsafe { compute_demerits_batch_simd(badness, 10.0, 5.0, fitness_changed) };
340        // infinity handling: (5 + inf)^2 = inf, (5 - inf)^2 = inf

343        // Normal values
344        assert!((result[2] - 25.0).abs() < 1e-10); // (5 + 0)^2 = 25
