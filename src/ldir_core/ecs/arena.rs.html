1//! Simple bump allocator for contiguous value storage.
2//!
3//! Provides O(1) allocation and access. Values are stored contiguously
4//! in a `Vec<T>`, supporting cache-linear iteration per
5//! [THM-ECS-CACHE-FRIENDLY].
6//!
7//! # References
8//!
9//! - REQ-4.1.1: Zero dynamic heap allocations during hot layout pass
10//! - CON-ECS-010: Bump allocator for dense array growth
11//! - REQ-4.1.3: 64-byte cache-line alignment for attribute arrays
12
13/// A simple bump allocator that stores values contiguously.
14///
15/// Allocation appends to an internal `Vec<T>`, returning the index.
16/// The arena does not support individual deallocation; use [`Arena::clear`]
17/// to reset the entire arena.
18///
19/// # Examples
20///
21/// ```ignore
22/// use ldir_core::ecs::Arena;
23///
24/// let mut arena: Arena<i32> = Arena::new();
25/// let a = arena.alloc(10);
26/// let b = arena.alloc(20);
27/// assert_eq!(arena.get(a), Some(&10));
28/// assert_eq!(arena.get(b), Some(&20));
29/// assert_eq!(arena.len(), 2);
30/// ```
31#[allow(dead_code)]
32pub struct Arena<T> {
33    /// Contiguous storage for allocated values.
34    /// Per REQ-4.1.3, dense arrays should be 64-byte cache-line aligned
35    /// (achieved via custom allocator in Phase B).
36    data: Vec<T>,
37}
38
39#[allow(dead_code)]
40impl<T> Arena<T> {
41    /// Creates a new empty arena.
42    pub const fn new() -> Self {
43        Self { data: Vec::new() }
44    }
45
46    /// Allocates a value, returning its index.
47    ///
48    /// Runs in O(1) amortized time (Vec push). Per AX-ECS-004,
49    /// allocation occurs only during cold initialization, not the
50    /// hot layout pass.
51    pub fn alloc(&mut self, value: T) -> usize {
52        let index = self.data.len();
53        self.data.push(value);
54        index
55    }
56
57    /// Returns a reference to the value at `index`, or `None` if out of bounds.
58    pub fn get(&self, index: usize) -> Option<&T> {
59        self.data.get(index)
60    }
61
62    /// Returns a mutable reference to the value at `index`, or `None` if out of bounds.
63    pub fn get_mut(&mut self, index: usize) -> Option<&mut T> {
64        self.data.get_mut(index)
65    }
66
67    /// Returns the number of allocated values.
68    pub fn len(&self) -> usize {
69        self.data.len()
70    }
71
72    /// Returns `true` if no values are allocated.
73    pub fn is_empty(&self) -> bool {
74        self.data.is_empty()
75    }
76
77    /// Clears all values, resetting length to zero.
78    ///
79    /// Does not change capacity. Per AX-ECS-004, this is used
80    /// during teardown rather than the hot layout pass.
81    pub fn clear(&mut self) {
82        self.data.clear();
83    }
84
85    /// Returns the underlying slice of allocated values.
86    pub fn as_slice(&self) -> &[T] {
87        &self.data
88    }
89
90    /// Returns the underlying mutable slice of allocated values.
91    pub fn as_mut_slice(&mut self) -> &mut [T] {
92        &mut self.data
93    }
94}
95
96impl<T> Default for Arena<T> {
97    fn default() -> Self {
98        Self::new()
99    }
100}

102#[cfg(test)]
103mod tests {
104    use super::*;
105
106    #[test]
107    fn test_alloc_returns_sequential_indices() {
108        let mut arena: Arena<i32> = Arena::new();
109        assert_eq!(arena.alloc(10), 0);
110        assert_eq!(arena.alloc(20), 1);
111        assert_eq!(arena.alloc(30), 2);
112    }
113
114    #[test]
115    fn test_get_returns_correct_value() {
116        let mut arena: Arena<String> = Arena::new();
117        let idx = arena.alloc(String::from("hello"));
118        assert_eq!(arena.get(idx), Some(&String::from("hello")));
119    }
120
121    #[test]
122    fn test_get_out_of_bounds_returns_none() {
123        let arena: Arena<i32> = Arena::new();
124        assert_eq!(arena.get(0), None);
125    }
126
127    #[test]
128    fn test_get_mut_allows_mutation() {
129        let mut arena: Arena<i32> = Arena::new();
130        let idx = arena.alloc(10);
131        *arena.get_mut(idx).unwrap() = 42;
132        assert_eq!(arena.get(idx), Some(&42));
133    }
134
135    #[test]
136    fn test_len_tracks_count() {
137        let mut arena: Arena<i32> = Arena::new();
138        assert_eq!(arena.len(), 0);
139        assert!(arena.is_empty());
140        arena.alloc(1);
141        assert_eq!(arena.len(), 1);
142        assert!(!arena.is_empty());
143        arena.alloc(2);
144        assert_eq!(arena.len(), 2);
145    }
146
147    #[test]
148    fn test_clear_resets_length() {
149        let mut arena: Arena<i32> = Arena::new();
150        arena.alloc(1);
151        arena.alloc(2);
152        arena.clear();
153        assert_eq!(arena.len(), 0);
154        assert!(arena.is_empty());
155    }
156
157    #[test]
158    fn test_clear_preserves_capacity_for_reuse() {
159        let mut arena: Arena<i32> = Arena::new();
160        arena.alloc(1);
161        arena.alloc(2);
162        arena.clear();
163        // After clear, new allocations should reuse capacity
164        let idx = arena.alloc(3);
165        assert_eq!(arena.get(idx), Some(&3));
166        assert_eq!(arena.len(), 1);
167    }
168
169    #[test]
170    fn test_as_slice_returns_all_values() {
171        let mut arena: Arena<i32> = Arena::new();
172        arena.alloc(10);
173        arena.alloc(20);
174        arena.alloc(30);
175        assert_eq!(arena.as_slice(), &[10, 20, 30]);
176    }
177}
