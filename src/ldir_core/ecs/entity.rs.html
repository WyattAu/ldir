1//! Entity identifier and allocator for the ECS.
2//!
3//! Entities are lightweight 32-bit identifiers carrying a slot index
4//! and a generation counter. The generation detects stale references
5//! after entity recycling.
6//!
7//! # References
8//!
9//! - REQ-3.1.6: 32-bit generation indices as entity identifiers
10//! - DEF-ENTITY: E = (e_id, g_gen)
11//! - AX-ECS-001: Entity uniqueness
12//! - THM-ECS-CAPACITY: Maximum entity count is 2^32 - 1
13//! - ALG-ECS-CREATE: Entity creation algorithm
14
15use std::fmt;
16
17/// Raw entity slot index — a 32-bit value.
18///
19/// Per REQ-3.1.6, entity IDs are 32-bit values.
20/// Slot 0 is reserved as the null/sentinel entity per REQ-3.1.2.
21#[allow(dead_code)]
22pub type EntityId = u32;
23
24/// A versioned entity reference consisting of a slot index and generation counter.
25///
26/// An entity reference is valid iff `generation` matches the allocator's
27/// current generation for the same `index`. This detects stale references
28/// after entity recycling (DEF-ENTITY).
29///
30/// # References
31///
32/// - DEF-ENTITY: E = (e_id, g_gen)
33/// - REQ-3.1.6: 32-bit generation indices as entity identifiers
34/// - THM-ECS-CAPACITY: Maximum entity count is 2^32 - 1
35#[allow(dead_code)]
36#[derive(Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
37pub struct Entity {
38    /// Slot index in the entity allocator.
39    pub index: u32,
40    /// Generation counter for stale-reference detection.
41    /// Per NC-ECS-005: generation counter width is 32 bits.
42    pub generation: u32,
43}
44
45impl Entity {
46    /// Creates a new entity reference with the given slot index and generation.
47    #[allow(dead_code)]
48    pub const fn new(index: u32, generation: u32) -> Self {
49        Self { index, generation }
50    }
51}
52
53impl fmt::Debug for Entity {
54    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
55        f.debug_struct("Entity")
56            .field("index", &self.index)
57            .field("generation", &self.generation)
58            .finish()
59    }
60}
61
62/// Bump allocator that generates unique entity IDs with generation tracking.
63///
64/// Maintains a monotonically increasing counter for fresh allocations
65/// and a free list for recycled slots. Each deallocation increments the
66/// generation counter for the slot, invalidating all outstanding references
67/// to the old generation.
68///
69/// No heap allocation occurs during [`EntityAllocator::allocate`] for recycled slots
70/// (free-list pop). Fresh slots may trigger amortized Vec growth.
71///
72/// # References
73///
74/// - AX-ECS-001: Entity uniqueness
75/// - THM-ECS-CAPACITY: Maximum entity count is 2^32 - 1
76/// - ALG-ECS-CREATE: Entity creation algorithm (lines 3-8)
77/// - ALG-ECS-DESTROY: Entity destruction algorithm (lines 36-38)
78/// - PRE-ECS-002: Entity generation validity check
79#[allow(dead_code)]
80pub struct EntityAllocator {
81    /// Next slot index for fresh allocation.
82    next_slot: u32,
83    /// Generation counter per slot. Incremented on deallocation
84    /// so that stale Entity references (old generation) are detected.
85    generations: Vec<u32>,
86    /// Free list of recycled slots. LIFO order for cache locality.
87    free_list: Vec<u32>,
88}
89
90#[allow(dead_code)]
91impl EntityAllocator {
92    /// Creates a new entity allocator with no live entities.
93    pub const fn new() -> Self {
94        Self {
95            next_slot: 0,
96            generations: Vec::new(),
97            free_list: Vec::new(),
98        }
99    }
100
101    /// Allocates a new entity, returning its versioned ID.
102    ///
103    /// If the free list is non-empty, recycles a slot (generation was
104    /// already bumped on deallocation). Otherwise, bumps `next_slot`.
105    ///
106    /// Per ALG-ECS-CREATE lines 3-8.
107    pub fn allocate(&mut self) -> Entity {
108        if let Some(index) = self.free_list.pop() {
109            let generation = self.generations[index as usize];
110            Entity::new(index, generation)
111        } else {
112            let index = self.next_slot;
113            self.next_slot += 1;
114            self.generations.push(0);
115            Entity::new(index, 0)
116        }
117    }
118
119    /// Returns `true` if the entity reference is still valid.
120    ///
121    /// Checks that the generation matches (PRE-ECS-002) and that
122    /// the entity is not in the free list.
123    pub fn is_alive(&self, entity: Entity) -> bool {
124        if entity.index as usize >= self.generations.len() {
125            return false;
126        }
127        let current_gen = self.generations[entity.index as usize];
128        current_gen == entity.generation
129    }
130
131    /// Deallocates an entity, returning its slot to the free list.
132    ///
133    /// Increments the generation counter for the slot so that any
134    /// outstanding `Entity` references with the old generation become
135    /// stale (detected by [`EntityAllocator::is_alive`]).
136    ///
137    /// Per ALG-ECS-DESTROY lines 36-38.
138    pub fn deallocate(&mut self, entity: Entity) {
139        if !self.is_alive(entity) {
140            return;
141        }
142        self.generations[entity.index as usize] += 1;
143        self.free_list.push(entity.index);
144    }
145
146    /// Returns the total number of slots ever allocated (including recycled).
147    pub fn capacity(&self) -> usize {
148        self.generations.len()
149    }
150
151    /// Returns the number of currently live entities.
152    pub fn alive_count(&self) -> usize {
153        self.generations.len().saturating_sub(self.free_list.len())
154    }
155}
156
157impl Default for EntityAllocator {
158    fn default() -> Self {
159        Self::new()
160    }
161}

167    #[test]
168    fn test_allocate_returns_sequential_entities() {
169        let mut alloc = EntityAllocator::new();
170        let e0 = alloc.allocate();
171        let e1 = alloc.allocate();
172        let e2 = alloc.allocate();
173        assert_eq!(e0.index, 0);
174        assert_eq!(e1.index, 1);
175        assert_eq!(e2.index, 2);
176        assert_eq!(e0.generation, 0);
177        assert_eq!(e1.generation, 0);
178        assert_eq!(e2.generation, 0);
179    }

181    #[test]
182    fn test_fresh_entity_is_alive() {
183        let mut alloc = EntityAllocator::new();
184        let e = alloc.allocate();
185        assert!(alloc.is_alive(e));
186    }

188    #[test]
189    fn test_deallocate_makes_entity_dead() {
190        let mut alloc = EntityAllocator::new();
191        let e = alloc.allocate();
192        alloc.deallocate(e);
193        assert!(!alloc.is_alive(e));
194    }

197    fn test_deallocate_nonexistent_is_noop() {
198        let mut alloc = EntityAllocator::new();
199        let e = Entity::new(999, 0);
200        alloc.deallocate(e); // should not panic
201    }

203    #[test]
204    fn test_recycled_slot_gets_new_generation() {
205        let mut alloc = EntityAllocator::new();
206        let e = alloc.allocate(); // index=0, gen=0
207        alloc.deallocate(e); // gen bumped to 1
208        let e2 = alloc.allocate(); // recycles index=0, gen=1
209        assert_eq!(e2.index, 0);
210        assert_eq!(e2.generation, 1);
211        assert!(!alloc.is_alive(e)); // old ref stale
212        assert!(alloc.is_alive(e2)); // new ref valid
213    }

215    #[test]
216    fn test_double_recycle_increments_generation_twice() {
217        let mut alloc = EntityAllocator::new();
218        let e0 = alloc.allocate(); // index=0, gen=0
219        alloc.deallocate(e0); // gen → 1
220        let e1 = alloc.allocate(); // recycles, gen=1
221        alloc.deallocate(e1); // gen → 2
222        let e2 = alloc.allocate(); // recycles, gen=2
223        assert_eq!(e2.index, 0);
224        assert_eq!(e2.generation, 2);
225    }

229        let mut alloc = EntityAllocator::new();

247    #[test]
248    fn test_stale_generation_is_detected() {
249        let mut alloc = EntityAllocator::new();
250        let original = alloc.allocate(); // index=0, gen=0
251        alloc.deallocate(original);
252        // Allocate enough entities to reuse the slot
253        let recycled = alloc.allocate(); // should reuse index=0
254        assert_eq!(recycled.index, original.index);
255        assert_ne!(recycled.generation, original.generation);
256        assert!(!alloc.is_alive(original));
257        assert!(alloc.is_alive(recycled));
258    }

260    #[test]
261    fn test_free_list_lifo_order() {
262        let mut alloc = EntityAllocator::new();
263        let e0 = alloc.allocate(); // index=0
264        let _e1 = alloc.allocate(); // index=1
265        let e2 = alloc.allocate(); // index=2
266        alloc.deallocate(e0);
267        alloc.deallocate(e2);
268        // LIFO: e2 freed last, so recycled first
269        let recycled1 = alloc.allocate();
270        assert_eq!(recycled1.index, 2);
271        let recycled2 = alloc.allocate();
272        assert_eq!(recycled2.index, 0);
273    }

275    #[test]
276    fn test_fresh_allocation_after_free_list_exhausted() {
277        let mut alloc = EntityAllocator::new();
278        let e0 = alloc.allocate(); // index=0
279        alloc.deallocate(e0); // gen → 1
280        let _recycled = alloc.allocate(); // recycles index=0, gen=1
281        let fresh = alloc.allocate(); // new slot index=1
282        assert_eq!(fresh.index, 1);
283        assert_eq!(fresh.generation, 0);
284    }
