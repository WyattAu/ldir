1//! S-IR document type.
2//!
3//! A `SIRDocument` is an ordered collection of `SIRInstruction` values
4//! representing a tree-structured document with entity-based addressing.
5//!
6//! Matches Lean 4 `abbrev SIRDocument := List SIRInstruction` in
7//! `ProofIRWellformedness.lean` Section 1.
8//!
9//! # Well-Formedness (DEF-004)
10//!
11//! A document is well-formed iff all 6 WF-SIR conditions hold:
12//! 1. **AX-001** (`entityUnique`): All entity IDs are distinct.
13//! 2. **AX-002** (`parentExists`): Every parent reference is valid.
14//! 3. **AX-003** (`isAcyclic`): No circular parent chains.
15//! 4. **AX-004**: Payload offsets are in bounds.
16//! 5. **DEF-004.5** (`hasSingleRoot`): Exactly one root entity.
17//! 6. **DEF-004.6**: Block nesting is properly structured.
18
19use crate::sir::instruction::{EntityId, SIRInstruction};
20use crate::sir::payload::PayloadRegion;
21use crate::sir::v2::SourceSpan;
22
23/// Ordered collection of S-IR instructions representing a document.
24///
25/// The instruction sequence encodes a tree structure via parent references:
26/// each instruction's `parent_id` points to another instruction's `entity_id`
27/// (or `ROOT_SENTINEL` for the single root).
28///
29/// # Examples
30///
31/// ```
32/// use ldir_ir::sir::{SIRDocument, SIRInstruction, SIROpcode, ROOT_SENTINEL};
33///
34/// let mut doc = SIRDocument::new();
35/// doc.push(SIRInstruction::new(SIROpcode::PushBlock, 0, ROOT_SENTINEL, 0));
36/// doc.push(SIRInstruction::new(SIROpcode::SetContent, 1, 0, 0));
37///
38/// assert_eq!(doc.len(), 2);
39/// assert_eq!(doc.entity_ids().collect::<Vec<_>>(), vec![0, 1]);
40/// ```
41#[derive(Debug, Clone, PartialEq, Eq, Hash)]
42pub struct SIRDocument {
43    instructions: Vec<SIRInstruction>,
44    /// Contiguous payload data referenced by instruction `payload_offset` fields.
45    payload: PayloadRegion,
46    /// Footnote entries: (number, text).
47    pub footnotes: Vec<(u32, String)>,
48    /// Source spans for instructions, indexed by instruction position.
49    pub source_spans: Vec<Option<SourceSpan>>,
50}
51
52impl SIRDocument {
53    /// Create a new empty S-IR document.
54    #[inline]
55    pub fn new() -> Self {
56        Self {
57            instructions: Vec::new(),
58            payload: PayloadRegion::new(),
59            footnotes: Vec::new(),
60            source_spans: Vec::new(),
61        }
62    }
63
64    /// Create a new S-IR document with pre-allocated capacity.
65    pub fn with_capacity(capacity: usize) -> Self {
66        Self {
67            instructions: Vec::with_capacity(capacity),
68            payload: PayloadRegion::new(),
69            footnotes: Vec::new(),
70            source_spans: Vec::new(),
71        }
72    }
73
74    /// Number of instructions in the document.
75    #[inline]
76    pub fn len(&self) -> usize {
77        self.instructions.len()
78    }
79
80    /// Check if the document has no instructions.
81    #[inline]
82    pub fn is_empty(&self) -> bool {
83        self.instructions.is_empty()
84    }
85
86    /// Get a reference to an instruction by index.
87    #[inline]
88    pub fn get(&self, index: usize) -> Option<&SIRInstruction> {
89        self.instructions.get(index)
90    }
91
92    /// Get a mutable reference to an instruction by index.
93    #[inline]
94    pub fn get_mut(&mut self, index: usize) -> Option<&mut SIRInstruction> {
95        self.instructions.get_mut(index)
96    }
97
98    /// Push an instruction onto the document.
99    #[inline]
100    pub fn push(&mut self, instruction: SIRInstruction) {
101        self.instructions.push(instruction);
102    }
103
104    /// Iterate over entity IDs in document order.
105    ///
106    /// Used by `entityUnique` (DEF-004 cond. 1) to check for duplicates.
107    pub fn entity_ids(&self) -> impl Iterator<Item = EntityId> + '_ {
108        self.instructions.iter().map(|i| i.entity_id())
109    }
110
111    /// Iterate over instructions in document order.
112    #[inline]
113    pub fn iter(&self) -> impl Iterator<Item = &SIRInstruction> {
114        self.instructions.iter()
115    }
116
117    /// Iterate over instructions mutably.
118    #[inline]
119    pub fn iter_mut(&mut self) -> impl Iterator<Item = &mut SIRInstruction> {
120        self.instructions.iter_mut()
121    }
122
123    /// Find an instruction by entity ID.
124    ///
125    /// Returns the index of the first instruction with the given entity ID,
126    /// or `None` if not found.
127    ///
128    /// Per AX-001, entity IDs are unique, so at most one match exists
129    /// in a well-formed document.
130    pub fn find_by_entity_id(&self, entity_id: EntityId) -> Option<usize> {
131        self.instructions
132            .iter()
133            .position(|i| i.entity_id() == entity_id)
134    }
135
136    /// Find all root instructions (parent_id == ROOT_SENTINEL).
137    ///
138    /// Per DEF-004 cond. 5, a well-formed document has exactly one root.
139    pub fn roots(&self) -> impl Iterator<Item = &SIRInstruction> + '_ {
140        self.instructions.iter().filter(|i| i.is_root())
141    }
142
143    /// Get the raw instruction slice.
144    #[inline]
145    pub fn as_slice(&self) -> &[SIRInstruction] {
146        &self.instructions
147    }
148
149    /// Get the raw instruction vector.
150    #[inline]
151    pub fn into_vec(self) -> Vec<SIRInstruction> {
152        self.instructions
153    }
154
155    /// Clear all instructions.
156    pub fn clear(&mut self) {
157        self.instructions.clear();
158    }
159
160    /// Reserve capacity for additional instructions.
161    pub fn reserve(&mut self, additional: usize) {
162        self.instructions.reserve(additional);
163    }
164
165    /// Push an instruction and associate payload data with it.
166    ///
167    /// The payload bytes are appended to the internal payload region, and
168    /// the instruction's `payload_offset` is set to the start of the appended data.
169    pub fn push_with_payload(&mut self, mut instruction: SIRInstruction, payload_data: &[u8]) {
170        let offset = self.payload.append(payload_data);
171        instruction = SIRInstruction::new(
172            instruction.opcode(),
173            instruction.entity_id(),
174            instruction.parent_id(),
175            offset,
176        );
177        self.instructions.push(instruction);
178    }
179
180    /// Get a reference to the payload region.
181    #[inline]
182    pub fn payload(&self) -> &PayloadRegion {
183        &self.payload
184    }
185
186    /// Get a mutable reference to the payload region.
187    #[inline]
188    pub fn payload_mut(&mut self) -> &mut PayloadRegion {
189        &mut self.payload
190    }
191
192    /// Look up the payload string for a given instruction via its `payload_offset`.
193    ///
194    /// Returns the text from `payload_offset` to the next NUL byte or end of payload.
195    /// Returns `None` if the offset is out of bounds or the bytes are not valid UTF-8.
196    pub fn payload_text(&self, instruction: &SIRInstruction) -> Option<&str> {
197        let bytes = self.payload.get_until_nul(instruction.payload_offset())?;
198        std::str::from_utf8(bytes).ok()
199    }
200
201    /// Serialize this document to rkyv bytes (instructions only).
202    ///
203    /// Note: Payload region is NOT included. Use [`Self::to_bytes_with_payload`]
204    /// for full round-trip fidelity including payload data.
205    pub fn to_bytes(&self) -> Vec<u8> {
206        // INVARIANT: SIRInstruction is a plain enum-of-primitives with no
207        // custom Serialize impl, so rkyv serialization cannot fail for
208        // well-formed data. Panicking here is intentional — a failure
209        // indicates a bug in instruction construction.
210        #[allow(clippy::expect_used)]
211        rkyv::to_bytes::<rkyv::rancor::Error>(&self.instructions)
212            .expect("rkyv serialization of SIR instructions should not fail")
213            .into_vec()
214    }
215
216    /// Deserialize an `S-IR document` from rkyv bytes (instructions only).
217    ///
218    /// The payload region will be empty. Use [`Self::from_bytes_with_payload`]
219    /// for full round-trip fidelity.
220    pub fn from_bytes(bytes: &[u8]) -> Result<Self, rkyv::rancor::Error> {
221        let instructions: Vec<SIRInstruction> = rkyv::from_bytes::<_, rkyv::rancor::Error>(bytes)?;
222        Ok(Self {
223            instructions,
224            payload: PayloadRegion::new(),
225            footnotes: Vec::new(),
226            source_spans: Vec::new(),
227        })
228    }
229
230    /// Serialize document with payload as a combined byte buffer.
231    ///
232    /// Format: `[u32 payload_len][payload bytes][rkyv instruction bytes]`
233    pub fn to_bytes_with_payload(&self) -> Vec<u8> {
234        let payload_bytes = self.payload.as_bytes();
235        // INVARIANT: SIRInstruction is a plain enum-of-primitives; rkyv
236        // serialization cannot fail for well-formed data.
237        #[allow(clippy::expect_used)]
238        let instr_bytes = rkyv::to_bytes::<rkyv::rancor::Error>(&self.instructions)
239            .expect("rkyv serialization of SIR instructions should not fail")
240            .into_vec();
241        let payload_len = payload_bytes.len() as u32;
242        let mut out = Vec::with_capacity(4 + payload_bytes.len() + instr_bytes.len());
243        out.extend_from_slice(&payload_len.to_le_bytes());
244        out.extend_from_slice(payload_bytes);
245        out.extend_from_slice(&instr_bytes);
246        out
247    }
248
249    /// Deserialize document from combined byte buffer.
250    pub fn from_bytes_with_payload(bytes: &[u8]) -> Result<Self, Box<dyn std::error::Error>> {
251        if bytes.len() < 4 {
252            return Err("buffer too short for payload header".into());
253        }
254        let payload_len = u32::from_le_bytes([bytes[0], bytes[1], bytes[2], bytes[3]]) as usize;
255        if bytes.len() < 4 + payload_len {
256            return Err("buffer too short for payload data".into());
257        }
258        let payload = PayloadRegion::from_bytes(bytes[4..4 + payload_len].to_vec());
259        let instructions: Vec<SIRInstruction> =
260            rkyv::from_bytes::<_, rkyv::rancor::Error>(&bytes[4 + payload_len..])?;
261        Ok(Self {
262            instructions,
263            payload,
264            footnotes: Vec::new(),
265            source_spans: Vec::new(),
266        })
267    }
268}

402#[cfg(test)]
403mod proptests {
404    use super::*;
405    use crate::sir::ROOT_SENTINEL;
406    use crate::sir::opcode::SIROpcode;
407    use proptest::prelude::*;
408
409    proptest! {
410        #[test]
411        fn roundtrip_document_instructions(ref instrs in proptest::collection::vec(any::<u8>(), 0..100)) {
412            let mut doc = SIRDocument::new();
413            for &b in instrs {
414                let opcode = SIROpcode::from_u8(b % 5).unwrap_or(SIROpcode::PushBlock);
415                let entity_id = b as u32;
416                let parent_id = if b % 3 == 0 { ROOT_SENTINEL } else { (b as u32).saturating_sub(1) };
417                doc.push(SIRInstruction::new(opcode, entity_id, parent_id, 0));
418            }
419            let _ = doc.len();
420        }
421    }
422
423    proptest! {
424        #[test]
425        fn payload_roundtrip(ref content in "[a-zA-Z0-9 ]{0,1000}") {
426            let mut doc = SIRDocument::new();
427            doc.push(SIRInstruction::new(SIROpcode::PushBlock, 0, ROOT_SENTINEL, 0));
428            doc.push_with_payload(
429                SIRInstruction::new(SIROpcode::SetContent, 1, 0, 0),
430                content.as_bytes(),
431            );
432            let text = doc.payload_text(doc.get(1).unwrap());
433            assert_eq!(text, Some(content.as_str()));
434        }
435    }
