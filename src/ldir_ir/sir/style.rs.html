1//! S-IR style definitions.
2//!
3//! Style modifiers for inline text formatting (bold, italic, monospace, etc.)
4//! are encoded as bitflags in the `ApplyStyle` instruction's payload.
5//!
6//! ## Encoding
7//!
8//! The `ApplyStyle` instruction's `payload_offset` field stores a packed u32:
9//! - Bits 0-7: Style modifier flags (bold, italic, mono, etc.)
10//! - Bits 8-15: Reserved for future use (font size, color index)
11//!
12//! ## Push/Pop Convention
13//!
14//! Styles are applied using `ApplyStyle` instructions with a direction flag:
15//! - Bit 7 set → push style onto stack (enter span)
16//! - Bit 7 clear → pop style from stack (exit span)
17
18/// Style modifier bitflags for inline text formatting.
19///
20/// Encoded in the lower byte of the `ApplyStyle` payload.
21#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
22#[repr(transparent)]
23pub struct StyleModifier(pub u8);
24
25impl StyleModifier {
26    /// Push style onto the stack (enter styled span).
27    /// When clear, this is a pop operation (exit styled span).
28    pub const PUSH: u8 = 1 << 7;
29    /// Bold text (**bold**).
30    pub const BOLD: u8 = 1 << 0;
31    /// Italic text (*italic*).
32    pub const ITALIC: u8 = 1 << 1;
33    /// Monospace text (`code`).
34    pub const MONO: u8 = 1 << 2;
35    /// Underlined text.
36    pub const UNDERLINE: u8 = 1 << 3;
37    /// Strikethrough text (~~strike~~).
38    pub const STRIKE: u8 = 1 << 4;
39    /// Small caps.
40    pub const SMALL_CAPS: u8 = 1 << 5;
41
42    /// No modifiers.
43    pub const EMPTY: StyleModifier = StyleModifier(0);
44
45    /// Bold style.
46    pub const BOLD_STYLE: StyleModifier = StyleModifier(Self::BOLD);
47    /// Italic style.
48    pub const ITALIC_STYLE: StyleModifier = StyleModifier(Self::ITALIC);
49    /// Monospace (inline code) style.
50    pub const MONO_STYLE: StyleModifier = StyleModifier(Self::MONO);
51    /// Strikethrough style (~~strike~~).
52    pub const STRIKE_STYLE: StyleModifier = StyleModifier(Self::STRIKE);
53    /// Bold + italic combined.
54    pub const BOLD_ITALIC: StyleModifier = StyleModifier(Self::BOLD | Self::ITALIC);
55
56    /// Check if a specific flag is set.
57    pub const fn contains(&self, flag: u8) -> bool {
58        self.0 & flag != 0
59    }
60
61    /// Encode a style push operation.
62    pub fn push(modifiers: StyleModifier) -> u32 {
63        (Self::PUSH | modifiers.0) as u32
64    }
65
66    /// Encode a style pop operation.
67    pub fn pop() -> u32 {
68        0u32
69    }
70
71    /// Decode style modifiers from a packed u32.
72    pub fn from_packed(packed: u32) -> (StyleModifier, bool) {
73        let byte = packed as u8;
74        let is_push = byte & Self::PUSH != 0;
75        let modifiers = StyleModifier(byte & !Self::PUSH);
76        (modifiers, is_push)
77    }
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83
84    #[test]
85    fn test_style_push_bold() {
86        let packed = StyleModifier::push(StyleModifier::BOLD_STYLE);
87        let (mods, is_push) = StyleModifier::from_packed(packed);
88        assert!(is_push);
89        assert!(mods.contains(StyleModifier::BOLD));
90    }
91
92    #[test]
93    fn test_style_push_bold_italic() {
94        let packed = StyleModifier::push(StyleModifier::BOLD_ITALIC);
95        let (mods, is_push) = StyleModifier::from_packed(packed);
96        assert!(is_push);
97        assert!(mods.contains(StyleModifier::BOLD));
98        assert!(mods.contains(StyleModifier::ITALIC));
99    }
100
101    #[test]
102    fn test_style_pop() {
103        let packed = StyleModifier::pop();
104        let (_mods, is_push) = StyleModifier::from_packed(packed);
105        assert!(!is_push);
106    }
107
108    #[test]
109    fn test_style_mixed() {
110        let packed = StyleModifier::push(StyleModifier(
111            StyleModifier::MONO | StyleModifier::UNDERLINE,
112        ));
113        let (mods, is_push) = StyleModifier::from_packed(packed);
114        assert!(is_push);
115        assert!(mods.contains(StyleModifier::MONO));
116        assert!(mods.contains(StyleModifier::UNDERLINE));
117        assert!(!mods.contains(StyleModifier::BOLD));
118    }
119
120    #[test]
121    fn test_empty_style() {
122        assert!(!StyleModifier::EMPTY.contains(StyleModifier::BOLD));
123    }
124}
