1//! S-IR opcode definitions (REQ-3.1.3).
2//!
3//! Maps to the Lean 4 `SIROpcode` inductive type in
4//! `ProofIRWellformedness.lean` Section 1.
5//!
6//! Each opcode is assigned a 1-byte discriminant for the wire format.
7
8/// Block type parameter for `PushBlock` instructions.
9///
10/// Matches Lean 4 `BlockType` inductive:
11/// `document | paragraph | heading | list | math | code`.
12#[derive(
13    Debug, Clone, Copy, PartialEq, Eq, Hash, rkyv::Archive, rkyv::Serialize, rkyv::Deserialize,
14)]
15#[repr(u8)]
16#[rkyv(attr(derive(Debug, Clone, Copy, PartialEq, Eq, Hash)))]
17pub enum BlockType {
18    /// Top-level document root node.
19    Document = 0x00,
20    /// Paragraph block.
21    Paragraph = 0x01,
22    /// Heading block (with level in payload).
23    Heading = 0x02,
24    /// Ordered or unordered list block.
25    List = 0x03,
26    /// Mathematical expression block (deferred to post-MVP, REQ-1.2.3).
27    Math = 0x04,
28    /// Code / verbatim block.
29    Code = 0x05,
30    /// Blockquote block.
31    BlockQuote = 0x06,
32    /// Thematic break (horizontal rule).
33    ThematicBreak = 0x07,
34    /// Image block (embedded or linked image).
35    Image = 0x08,
36    /// Table block (grid of cells).
37    Table = 0x09,
38    /// Table row within a table.
39    TableRow = 0x0a,
40    /// Table cell within a table row.
41    TableCell = 0x0b,
42    /// Footnote inline reference.
43    Footnote = 0x0c,
44    /// Footnote definition block (container for all footnotes).
45    FootnoteBlock = 0x0d,
46    /// Figure float container.
47    Figure = 0x0e,
48}
49
50impl BlockType {
51    /// Try to convert a raw byte to a `BlockType`.
52    ///
53    /// Returns `None` for bytes outside the valid discriminant range.
54    pub fn from_u8(byte: u8) -> Option<Self> {
55        match byte {
56            0x00 => Some(Self::Document),
57            0x01 => Some(Self::Paragraph),
58            0x02 => Some(Self::Heading),
59            0x03 => Some(Self::List),
60            0x04 => Some(Self::Math),
61            0x05 => Some(Self::Code),
62            0x06 => Some(Self::BlockQuote),
63            0x07 => Some(Self::ThematicBreak),
64            0x08 => Some(Self::Image),
65            0x09 => Some(Self::Table),
66            0x0a => Some(Self::TableRow),
67            0x0b => Some(Self::TableCell),
68            0x0c => Some(Self::Footnote),
69            0x0d => Some(Self::FootnoteBlock),
70            0x0e => Some(Self::Figure),
71            _ => None,
72        }
73    }
74}

76/// S-IR operation discriminator (1 byte, REQ-3.1.3).
77///
78/// Wire format assigns each variant a fixed byte value:
79/// - `0x00`..`0x05`: `PushBlock` with `BlockType` encoded as lower nibble
80/// - `0x06`..`0x09`: Parameterless opcodes
81///
82/// Matches Lean 4 `SIROpcode` inductive in `ProofIRWellformedness.lean`:
83/// `pushBlock | setContent | applyStyle | insertMath | linkData`.
84#[derive(
85    Debug, Clone, Copy, PartialEq, Eq, Hash, rkyv::Archive, rkyv::Serialize, rkyv::Deserialize,
86)]
87#[repr(u8)]
88#[rkyv(attr(derive(Debug, Clone, Copy, PartialEq, Eq, Hash)))]
89pub enum SIROpcode {
90    /// Push a structural block onto the document tree.
91    /// Carries a `BlockType` parameter in the payload region.
92    ///
93    /// Maps to Lean 4: `SIROpcode.pushBlock(bt : BlockType)`.
94    PushBlock = 0x00,
95
96    /// Set text content for the current entity.
97    /// Payload contains a UTF-8 text blob (REQ-3.1.4).
98    ///
99    /// Maps to Lean 4: `SIROpcode.setContent`.
100    SetContent = 0x01,
101
102    /// Apply a style (font, size, color) to the current entity.
103    /// Payload contains style parameters.
104    ///
105    /// Maps to Lean 4: `SIROpcode.applyStyle`.
106    ApplyStyle = 0x02,
107
108    /// Insert a mathematical expression placeholder.
109    /// Deferred to post-MVP per REQ-1.2.3.
110    /// Payload contains MathML or TeX reference.
111    ///
112    /// Maps to Lean 4: `SIROpcode.insertMath`.
113    InsertMath = 0x03,
114
115    /// Attach a hyperlink or data reference.
116    /// Payload contains the link target pointer.
117    ///
118    /// Maps to Lean 4: `SIROpcode.linkData`.
119    LinkData = 0x04,
120}
121
122impl SIROpcode {
123    /// Try to convert a raw byte to a `SIROpcode`.
124    ///
125    /// Returns `None` for bytes outside the valid discriminant range.
126    pub fn from_u8(byte: u8) -> Option<Self> {
127        match byte {
128            0x00 => Some(Self::PushBlock),
129            0x01 => Some(Self::SetContent),
130            0x02 => Some(Self::ApplyStyle),
131            0x03 => Some(Self::InsertMath),
132            0x04 => Some(Self::LinkData),
133            _ => None,
134        }
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    #[test]
143    fn test_opcode_discriminants() {
144        assert_eq!(SIROpcode::PushBlock as u8, 0x00);
145        assert_eq!(SIROpcode::SetContent as u8, 0x01);
146        assert_eq!(SIROpcode::ApplyStyle as u8, 0x02);
147        assert_eq!(SIROpcode::InsertMath as u8, 0x03);
148        assert_eq!(SIROpcode::LinkData as u8, 0x04);
149    }
150
151    #[test]
152    fn test_block_type_discriminants() {
153        assert_eq!(BlockType::Document as u8, 0x00);
154        assert_eq!(BlockType::Code as u8, 0x05);
155        assert_eq!(BlockType::Image as u8, 0x08);
156        assert_eq!(BlockType::Table as u8, 0x09);
157    }
158
159    #[test]
160    fn test_opcode_from_u8_roundtrip() {
161        for byte in 0..=255u8 {
162            if let Some(op) = SIROpcode::from_u8(byte) {
163                assert_eq!(op as u8, byte);
164            }
165        }
166    }
167
168    #[test]
169    fn test_block_type_from_u8_roundtrip() {
170        for byte in 0..=255u8 {
171            if let Some(bt) = BlockType::from_u8(byte) {
172                assert_eq!(bt as u8, byte);
173            }
174        }
175    }
