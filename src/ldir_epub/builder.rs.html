94    #[must_use = "building EPUB can fail; check the result"]
95    pub fn build(&self, module: &SIRModuleV2) -> Result<Vec<u8>, EpubError> {
96        let title = module.metadata.title.as_deref().unwrap_or("Untitled");
97        let author = module.metadata.author.as_deref().unwrap_or("Unknown");
98        let lang = &module.metadata.language;
99        let uid = format!("urn:uuid:{}", uuid_simple());
100
101        let html_options = HtmlOptions {
102            include_toc: self.options.include_toc,
103            include_styles: false,
104            math_format: MathFormat::LaTeX,
105            indent: 2,
106        };
107        let content_html = HtmlRenderer::with_options(html_options).render(module);
108
109        let body_content = extract_body(&content_html);
110
111        let chapter_xhtml = format_xhtml(title, &body_content);
112        let css = self.options.css.clone().unwrap_or_else(default_css);
113        let opf = format_opf(
114            title,
115            author,
116            lang,
117            &uid,
118            &self.overlays,
119            &self.narrator,
120            &self.total_duration,
121        );
122        let toc_xhtml = format_nav_xhtml(title, module);
123        let container_xml = format_container_xml();
124        let toc_ncx = format_toc_ncx(title, &uid);
125
126        let mut zip = SimpleZip::new();
127        let mut mimetype_bytes = b"application/epub+zip".to_vec();
128        mimetype_bytes.push(0);
129        zip.add_file("mimetype", &mimetype_bytes, true);
130        zip.add_file("META-INF/container.xml", container_xml.as_bytes(), false);
131        zip.add_file("OEBPS/content.opf", opf.as_bytes(), false);
132        zip.add_file("OEBPS/toc.ncx", toc_ncx.as_bytes(), false);
133        zip.add_file("OEBPS/toc.xhtml", toc_xhtml.as_bytes(), false);
134        zip.add_file("OEBPS/style.css", css.as_bytes(), false);
135        zip.add_file("OEBPS/chapter1.xhtml", chapter_xhtml.as_bytes(), false);
136
137        for overlay in &self.overlays {
138            let smil = format_smil(overlay);
139            let path = format!("OEBPS/overlays/{}.smil", overlay.body_id);
140            zip.add_file(&path, smil.as_bytes(), false);
141        }
142
143        zip.finish().map_err(EpubError::BuildError)
144    }
145}

147fn extract_body(html: &str) -> String {
148    if let Some(start) = html.find("<body>") {
149        let start = start + "<body>".len();
150        if let Some(end) = html.find("</body>") {
151            return html[start..end].to_string();
152        }
153    }
154    html.to_string()
155}

192fn format_opf(
193    title: &str,
194    author: &str,
195    lang: &str,
196    uid: &str,
197    overlays: &[MediaOverlay],
198    narrator: &Option<String>,
199    total_duration: &Option<Duration>,
200) -> String {
201    let mut manifest_extra = String::new();
202    let mut spine_extra = String::new();
203    let mut meta_extra = String::new();
204
205    for overlay in overlays {
206        let item_id = format!("{}-mo", overlay.body_id);
207        let href = format!("overlays/{}.smil", overlay.body_id);
208        manifest_extra.push_str(&format!(
209            "  <item id=\"{item_id}\" href=\"{href}\" media-type=\"application/smil+xml\" properties=\"media-overlay\"/>\n",
210            item_id = escape_xml(&item_id),
211            href = escape_xml(&href),
212        ));
213        spine_extra.push_str(&format!(
214            "  <itemref idref=\"{body_id}\" media-overlay=\"{item_id}\"/>\n",
215            body_id = escape_xml(&overlay.body_id),
216            item_id = escape_xml(&item_id),
217        ));
218    }
219
220    if let Some(name) = narrator {
221        meta_extra.push_str(&format!(
222            "  <meta property=\"media:narrator\">{}</meta>\n",
223            escape_xml(name),
224        ));
225    }
226    if let Some(dur) = total_duration {
227        meta_extra.push_str(&format!(
228            "  <meta property=\"media:duration\">{}</meta>\n",
229            format_duration(*dur),
230        ));
231    }
232
233    format!(
234        r#"<?xml version="1.0" encoding="UTF-8"?>
235<package xmlns="http://www.idpf.org/2007/opf" version="3.0" unique-identifier="uid">
236<metadata xmlns:dc="http://purl.org/dc/elements/1.1/">
237  <dc:identifier id="uid">{uid}</dc:identifier>
238  <dc:title>{title}</dc:title>
239  <dc:language>{lang}</dc:language>
240  <dc:creator>{author}</dc:creator>
241  <dc:date>{date_only}</dc:date>
242  <meta property="dcterms:modified">{date_now}</meta>
243  <meta property="a11y:validate">true</meta>
244  <meta property="rendition:layout">reflowable</meta>
245{meta_extra}</metadata>
246<manifest>
247  <item id="chapter1" href="chapter1.xhtml" media-type="application/xhtml+xml" properties="scripted"/>
248  <item id="toc" href="toc.xhtml" media-type="application/xhtml+xml" properties="nav"/>
249  <item id="ncx" href="toc.ncx" media-type="application/x-dtbncx+xml"/>
250  <item id="style" href="style.css" media-type="text/css"/>
251{manifest_extra}</manifest>
252<spine toc="ncx">
253  <itemref idref="chapter1"/>
254{spine_extra}</spine>
255</package>"#,
256        title = escape_xml(title),
257        author = escape_xml(author),
258        lang = lang,
259        uid = uid,
260        date_only = date_now_str().split('T').next().unwrap_or(""),
261        date_now = date_now_str(),
262        meta_extra = meta_extra,
263        manifest_extra = manifest_extra,
264        spine_extra = spine_extra,
265    )
266}

303fn build_nested_toc(module: &SIRModuleV2) -> String {
304    let mut entries: Vec<TocEntry> = Vec::new();
305    for node in module.headings() {
306        if let Some(level) = node.heading_level() {
307            let text = module.body.collect_text(node.id);
308            let fallback_id = format!("heading-{}", node.id);
309            let id = node.label.as_deref().unwrap_or(&fallback_id);
310            entries.push(TocEntry {
311                id: id.to_string(),
312                text,
313                level,
314            });
315        }
316    }
317    format_nested_entries(&entries, 1)
318}
319
320fn format_nested_entries(entries: &[TocEntry], min_level: u8) -> String {
321    let mut result = String::new();
322    let mut current_level = min_level;
323    for entry in entries {
324        while entry.level > current_level {
325            result.push_str("    ".repeat(current_level as usize).as_str());
326            result.push_str("<ol>\n");
327            current_level += 1;
328        }
329        while entry.level < current_level && current_level > min_level {
330            current_level -= 1;
331            result.push_str("    ".repeat(current_level as usize).as_str());
332            result.push_str("</ol>\n");
333        }
334        result.push_str("    ".repeat(current_level as usize).as_str());
335        result.push_str(&format!(
336            "<li><a href=\"chapter1.xhtml#{}\">{}</a></li>\n",
337            entry.id,
338            escape_xml(&entry.text)
339        ));
340    }
341    while current_level > min_level {
342        current_level -= 1;
343        result.push_str("    ".repeat(current_level as usize).as_str());
344        result.push_str("</ol>\n");
345    }
346    result
347}

379fn format_smil(overlay: &MediaOverlay) -> String {
380    let mut par_elements = String::new();
381    for param in &overlay.params {
382        let clip_begin = format_smil_timecode(param.clip_begin);
383        let clip_end = format_smil_timecode(param.clip_end);
384        par_elements.push_str(&format!(
385            "    <par>\n      <text src=\"{}\"/>\n      <audio src=\"{}\" clipBegin=\"{}\" clipEnd=\"{}\"/>\n    </par>\n",
386            escape_xml(&param.text_ref),
387            escape_xml(&param.audio_ref),
388            clip_begin,
389            clip_end,
390        ));
391    }
392
393    format!(
394        r#"<?xml version="1.0" encoding="UTF-8"?>
395<smil xmlns="http://www.w3.org/ns/SMIL" xmlns:epub="http://www.idpf.org/2007/ops" version="3.0">
396<body>
397<seq id="{body_id}" epub:textref="{body_id}.xhtml">
398{par_elements}</seq>
399</body>
400</smil>"#,
401        body_id = escape_xml(&overlay.body_id),
402        par_elements = par_elements,
403    )
404}

406fn format_smil_timecode(seconds: Option<f32>) -> String {
407    match seconds {
408        Some(s) => {
409            let total_ms = (s * 1000.0).round() as u64;
410            let ms = total_ms % 1000;
411            let total_s = total_ms / 1000;
412            let s_part = total_s % 60;
413            let total_m = total_s / 60;
414            let m_part = total_m % 60;
415            let h_part = total_m / 60;
416            format!("{:02}:{:02}:{:02}.{:03}", h_part, m_part, s_part, ms)
417        }
418        None => "0:00:00.000".to_string(),
419    }
420}

422fn format_duration(duration: Duration) -> String {
423    let total_secs = duration.as_secs_f32();
424    format_smil_timecode(Some(total_secs))
425}

427fn escape_xml(s: &str) -> String {
428    let mut out = String::with_capacity(s.len());
429    for c in s.chars() {
430        match c {
431            '&' => out.push_str("&amp;"),
432            '<' => out.push_str("&lt;"),
433            '>' => out.push_str("&gt;"),
434            '"' => out.push_str("&quot;"),
435            _ => out.push(c),
436        }
437    }
438    out
439}

441fn uuid_simple() -> String {
442    use std::time::{SystemTime, UNIX_EPOCH};
443    let secs = SystemTime::now()
444        .duration_since(UNIX_EPOCH)
445        .map(|d| d.as_secs())
446        .unwrap_or(0);
447    format!("{:08x}-{:04x}-{:04x}-{:04x}-{:012x}", secs, 0, 0, 0, 0)
448}

450fn date_now_str() -> String {
451    use std::time::{SystemTime, UNIX_EPOCH};
452    let d = SystemTime::now()
453        .duration_since(UNIX_EPOCH)
454        .map(|d| d.as_secs())
455        .unwrap_or(0);
456    let secs = d % 86400;
457    let days_since_epoch = d / 86400;
458    let (year, month, day) = days_to_date(days_since_epoch as i64);
459    format!(
460        "{}-{:02}-{:02}T{:02}:{:02}:{:02}Z",
461        year,
462        month,
463        day,
464        secs / 3600,
465        (secs % 3600) / 60,
466        secs % 60
467    )
468}

470fn days_to_date(days: i64) -> (i64, i64, i64) {
471    let year = 1970 + days / 365;
472    let remaining = days % 365;
473    let month = (remaining / 30) + 1;
474    let day = (remaining % 30) + 1;
475    (year.min(2100), month.min(12), day.min(31))
476}

478struct SimpleZip {
479    files: Vec<(String, Vec<u8>, bool)>,
480}
481
482impl SimpleZip {
483    fn new() -> Self {
484        Self { files: Vec::new() }
485    }
486
487    fn add_file(&mut self, path: &str, data: &[u8], stored: bool) {
488        self.files.push((path.to_string(), data.to_vec(), stored));
489    }
490
491    fn finish(&self) -> Result<Vec<u8>, String> {
492        let mut buf = Vec::new();
493        let mut central_offsets: Vec<(u64, u32, u32, String)> = Vec::new();
494
495        for (path, data, stored) in &self.files {
496            let offset = buf.len() as u64;
497            let crc = crc32(data);
498            let size = data.len() as u32;
499
500            buf.extend_from_slice(&[0x50, 0x4B, 0x03, 0x04]);
501            buf.extend_from_slice(&[20, 0]);
502            buf.extend_from_slice(&[0, 0]);
503            buf.extend_from_slice(if *stored { &[0, 0] } else { &[8, 0] });
504            buf.extend_from_slice(&[0, 0]);
505            buf.extend_from_slice(&[0, 0]);
506            buf.extend_from_slice(&crc.to_le_bytes());
507            buf.extend_from_slice(&size.to_le_bytes());
508            buf.extend_from_slice(&size.to_le_bytes());
509            let name_bytes = path.as_bytes();
510            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
511            buf.extend_from_slice(&[0u8; 10]);
512            buf.extend_from_slice(name_bytes);
513            buf.extend_from_slice(data);
514
515            central_offsets.push((offset, size, crc, path.clone()));
516        }
517
518        let central_start = buf.len() as u64;
519
520        for (offset, size, crc, name) in &central_offsets {
521            buf.extend_from_slice(&[0x50, 0x4B, 0x01, 0x02]);
522            buf.extend_from_slice(&[20, 0]);
523            buf.extend_from_slice(&[20, 0]);
524            buf.extend_from_slice(&[0, 0]);
525            buf.extend_from_slice(&[0, 0]);
526            buf.extend_from_slice(&[0, 0]);
527            buf.extend_from_slice(&[0, 0]);
528            buf.extend_from_slice(&crc.to_le_bytes());
529            buf.extend_from_slice(&size.to_le_bytes());
530            buf.extend_from_slice(&size.to_le_bytes());
531            let name_bytes = name.as_bytes();
532            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
533            buf.extend_from_slice(&[0u8; 10]);
534            buf.extend_from_slice(&[0, 0, 0]);
535            buf.extend_from_slice(&(*offset as u32).to_le_bytes());
536            buf.extend_from_slice(name_bytes);
537        }
538
539        let central_end = buf.len() as u64;
540        let central_count = central_offsets.len() as u16;
541
542        buf.extend_from_slice(&[0x50, 0x4B, 0x05, 0x06]);
543        buf.extend_from_slice(&[0, 0]);
544        buf.extend_from_slice(&[0, 0]);
545        buf.extend_from_slice(&central_count.to_le_bytes());
546        buf.extend_from_slice(&central_count.to_le_bytes());
547        buf.extend_from_slice(&(central_start as u32).to_le_bytes());
548        buf.extend_from_slice(&(central_end as u32).to_le_bytes());
549        buf.extend_from_slice(&[0u8; 2]);
550
551        Ok(buf)
552    }
553}

555fn crc32(data: &[u8]) -> u32 {
556    let mut crc: u32 = 0xFFFFFFFF;
557    for &byte in data {
558        crc ^= byte as u32;
559        for _ in 0..8 {
560            if crc & 1 != 0 {
561                crc = (crc >> 1) ^ 0xEDB88320;
562            } else {
563                crc >>= 1;
564            }
565        }
566    }
567    !crc
568}

773    fn test_smil_generation() {
774        let overlay = MediaOverlay {
775            body_id: "chapter1".into(),
776            params: vec![
777                OverlayParam {
778                    text_ref: "chapter1.xhtml#p1".into(),
779                    audio_ref: "audio/chapter1.mp3".into(),
780                    clip_begin: Some(0.0),
781                    clip_end: Some(5.25),
782                },
783                OverlayParam {
784                    text_ref: "chapter1.xhtml#p2".into(),
785                    audio_ref: "audio/chapter1.mp3".into(),
786                    clip_begin: Some(5.25),
787                    clip_end: Some(12.75),
788                },
789            ],
790        };
791        let smil = format_smil(&overlay);
792        assert!(smil.contains("xmlns=\"http://www.w3.org/ns/SMIL\""));
793        assert!(smil.contains("epub:textref=\"chapter1.xhtml\""));
794        assert!(smil.contains("<seq id=\"chapter1\""));
795        assert!(smil.contains("<text src=\"chapter1.xhtml#p1\"/>"));
796        assert!(smil.contains("<audio src=\"audio/chapter1.mp3\""));
797        assert!(smil.contains("clipBegin=\"00:00:00.000\""));
798        assert!(smil.contains("clipEnd=\"00:00:05.250\""));
799        assert!(smil.contains("clipBegin=\"00:00:05.250\""));
800        assert!(smil.contains("clipEnd=\"00:00:12.750\""));
801    }

826        Ok(())
