142    #[must_use = "building EPUB can fail; check the result"]
143    pub fn build(&self, module: &SIRModuleV2) -> Result<Vec<u8>, EpubError> {
144        let title = module.metadata.title.as_deref().unwrap_or("Untitled");
145        let author = module.metadata.author.as_deref().unwrap_or("Unknown");
146        let lang = &module.metadata.language;
147        let uid = format!("urn:uuid:{}", uuid_simple());
148
149        let html_options = HtmlOptions {
150            include_toc: self.options.include_toc,
151            include_styles: false,
152            math_format: MathFormat::LaTeX,
153            indent: 2,
154        };
155        let content_html = HtmlRenderer::with_options(html_options).render(module);
156
157        let body_content = extract_body(&content_html);
158
159        let chapter_xhtml = format_xhtml(title, &body_content);
160        let css = self.options.css.clone().unwrap_or_else(default_css);
161        let opf = format_opf(
162            title,
163            author,
164            lang,
165            &uid,
166            &self.overlays,
167            &self.narrator,
168            &self.total_duration,
169        );
170        let toc_xhtml = format_nav_xhtml(title, module, &self.landmarks);
171        let container_xml = format_container_xml();
172        let toc_ncx = generate_ncx(title, &uid, module);
173
174        let mut zip = SimpleZip::new();
175        let mut mimetype_bytes = b"application/epub+zip".to_vec();
176        mimetype_bytes.push(0);
177        zip.add_file("mimetype", &mimetype_bytes, true);
178        zip.add_file("META-INF/container.xml", container_xml.as_bytes(), false);
179        zip.add_file("OEBPS/content.opf", opf.as_bytes(), false);
180        zip.add_file("OEBPS/toc.ncx", toc_ncx.as_bytes(), false);
181        zip.add_file("OEBPS/toc.xhtml", toc_xhtml.as_bytes(), false);
182        zip.add_file("OEBPS/style.css", css.as_bytes(), false);
183        zip.add_file("OEBPS/chapter1.xhtml", chapter_xhtml.as_bytes(), false);
184
185        for overlay in &self.overlays {
186            let smil = format_smil(overlay);
187            let path = format!("OEBPS/overlays/{}.smil", overlay.body_id);
188            zip.add_file(&path, smil.as_bytes(), false);
189        }
190
191        zip.finish().map_err(EpubError::BuildError)
192    }
193}

195fn extract_body(html: &str) -> String {
196    if let Some(start) = html.find("<body>") {
197        let start = start + "<body>".len();
198        if let Some(end) = html.find("</body>") {
199            return html[start..end].to_string();
200        }
201    }
202    html.to_string()
203}

240fn format_opf(
241    title: &str,
242    author: &str,
243    lang: &str,
244    uid: &str,
245    overlays: &[MediaOverlay],
246    narrator: &Option<String>,
247    total_duration: &Option<Duration>,
248) -> String {
249    let mut manifest_extra = String::new();
250    let mut spine_extra = String::new();
251    let mut meta_extra = String::new();
252
253    for overlay in overlays {
254        let item_id = format!("{}-mo", overlay.body_id);
255        let href = format!("overlays/{}.smil", overlay.body_id);
256        manifest_extra.push_str(&format!(
257            "  <item id=\"{item_id}\" href=\"{href}\" media-type=\"application/smil+xml\" properties=\"media-overlay\"/>\n",
258            item_id = escape_xml(&item_id),
259            href = escape_xml(&href),
260        ));
261        spine_extra.push_str(&format!(
262            "  <itemref idref=\"{body_id}\" media-overlay=\"{item_id}\"/>\n",
263            body_id = escape_xml(&overlay.body_id),
264            item_id = escape_xml(&item_id),
265        ));
266    }
267
268    if let Some(name) = narrator {
269        meta_extra.push_str(&format!(
270            "  <meta property=\"media:narrator\">{}</meta>\n",
271            escape_xml(name),
272        ));
273    }
274    if let Some(dur) = total_duration {
275        meta_extra.push_str(&format!(
276            "  <meta property=\"media:duration\">{}</meta>\n",
277            format_duration(*dur),
278        ));
279    }
280
281    format!(
282        r#"<?xml version="1.0" encoding="UTF-8"?>
283<package xmlns="http://www.idpf.org/2007/opf" version="3.0" unique-identifier="uid">
284<metadata xmlns:dc="http://purl.org/dc/elements/1.1/">
285  <dc:identifier id="uid">{uid}</dc:identifier>
286  <dc:title>{title}</dc:title>
287  <dc:language>{lang}</dc:language>
288  <dc:creator>{author}</dc:creator>
289  <dc:date>{date_only}</dc:date>
290  <meta property="dcterms:modified">{date_now}</meta>
291  <meta property="a11y:validate">true</meta>
292  <meta property="rendition:layout">reflowable</meta>
293{meta_extra}</metadata>
294<manifest>
295  <item id="chapter1" href="chapter1.xhtml" media-type="application/xhtml+xml" properties="scripted"/>
296  <item id="toc" href="toc.xhtml" media-type="application/xhtml+xml" properties="nav"/>
297  <item id="ncx" href="toc.ncx" media-type="application/x-dtbncx+xml"/>
298  <item id="style" href="style.css" media-type="text/css"/>
299{manifest_extra}</manifest>
300<spine toc="ncx">
301  <itemref idref="chapter1"/>
302{spine_extra}</spine>
303</package>"#,
304        title = escape_xml(title),
305        author = escape_xml(author),
306        lang = lang,
307        uid = uid,
308        date_only = date_now_str().split('T').next().unwrap_or(""),
309        date_now = date_now_str(),
310        meta_extra = meta_extra,
311        manifest_extra = manifest_extra,
312        spine_extra = spine_extra,
313    )
314}

366fn build_nested_toc(module: &SIRModuleV2) -> String {
367    let mut entries: Vec<TocEntry> = Vec::new();
368    for node in module.headings() {
369        if let Some(level) = node.heading_level() {
370            let text = module.body.collect_text(node.id);
371            let fallback_id = format!("heading-{}", node.id);
372            let id = node.label.as_deref().unwrap_or(&fallback_id);
373            entries.push(TocEntry {
374                id: id.to_string(),
375                text,
376                level,
377            });
378        }
379    }
380    format_nested_entries(&entries, 1)
381}
382
383fn format_nested_entries(entries: &[TocEntry], min_level: u8) -> String {
384    let mut result = String::new();
385    let mut current_level = min_level;
386    for entry in entries {
387        while entry.level > current_level {
388            result.push_str("    ".repeat(current_level as usize).as_str());
389            result.push_str("<ol>\n");
390            current_level += 1;
391        }
392        while entry.level < current_level && current_level > min_level {
393            current_level -= 1;
394            result.push_str("    ".repeat(current_level as usize).as_str());
395            result.push_str("</ol>\n");
396        }
397        result.push_str("    ".repeat(current_level as usize).as_str());
398        result.push_str(&format!(
399            "<li><a href=\"chapter1.xhtml#{}\">{}</a></li>\n",
400            entry.id,
401            escape_xml(&entry.text)
402        ));
403    }
404    while current_level > min_level {
405        current_level -= 1;
406        result.push_str("    ".repeat(current_level as usize).as_str());
407        result.push_str("</ol>\n");
408    }
409    result
410}

412fn generate_ncx(title: &str, uid: &str, module: &SIRModuleV2) -> String {
413    let mut nav_points = String::new();
414    let mut play_order: u32 = 1;
415
416    let mut entries: Vec<TocEntry> = Vec::new();
417    for node in module.headings() {
418        if let Some(_level) = node.heading_level() {
419            let text = module.body.collect_text(node.id);
420            let fallback_id = format!("heading-{}", node.id);
421            let id = node.label.as_deref().unwrap_or(&fallback_id);
422            entries.push(TocEntry {
423                id: id.to_string(),
424                text,
425                level: _level,
426            });
427        }
428    }
429
430    for entry in &entries {
431        nav_points.push_str(&format!(
432            "  <navPoint id=\"navpoint-{play_order}\" playOrder=\"{play_order}\">\n",
433            play_order = play_order
434        ));
435        nav_points.push_str(&format!(
436            "    <navLabel><text>{}</text></navLabel>\n",
437            escape_xml(&entry.text)
438        ));
439        nav_points.push_str(&format!(
440            "    <content src=\"chapter1.xhtml#{}\"/>\n",
441            entry.id
442        ));
443        nav_points.push_str("  </navPoint>\n");
444        play_order += 1;
445    }
446
447    if nav_points.is_empty() {
448        nav_points = format!(
449            "  <navPoint id=\"navpoint-1\" playOrder=\"1\">\n    <navLabel><text>{}</text></navLabel>\n    <content src=\"chapter1.xhtml\"/>\n  </navPoint>\n",
450            escape_xml(title)
451        );
452    }
453
454    let depth = entries.iter().map(|e| e.level as u32).max().unwrap_or(1);
455
456    format!(
457        r#"<?xml version="1.0" encoding="UTF-8"?>
458<ncx xmlns="http://www.daisy.org/z3986/2005/ncx/" version="2005-1">
459<head>
460<meta name="dtb:uid" content="{uid}"/>
461<meta name="dtb:depth" content="{depth}"/>
462<meta name="dtb:totalPageCount" content="0"/>
463<meta name="dtb:maxPageNumber" content="0"/>
464</head>
465<docTitle><text>{title}</text></docTitle>
466<navMap>
467{nav_points}</navMap>
468</ncx>"#,
469        title = escape_xml(title),
470        uid = uid,
471        depth = depth,
472        nav_points = nav_points,
473    )
474}

486fn format_smil(overlay: &MediaOverlay) -> String {
487    let mut par_elements = String::new();
488    for param in &overlay.params {
489        let clip_begin = format_smil_timecode(param.clip_begin);
490        let clip_end = format_smil_timecode(param.clip_end);
491        par_elements.push_str(&format!(
492            "    <par>\n      <text src=\"{}\"/>\n      <audio src=\"{}\" clipBegin=\"{}\" clipEnd=\"{}\"/>\n    </par>\n",
493            escape_xml(&param.text_ref),
494            escape_xml(&param.audio_ref),
495            clip_begin,
496            clip_end,
497        ));
498    }
499
500    format!(
501        r#"<?xml version="1.0" encoding="UTF-8"?>
502<smil xmlns="http://www.w3.org/ns/SMIL" xmlns:epub="http://www.idpf.org/2007/ops" version="3.0">
503<body>
504<seq id="{body_id}" epub:textref="{body_id}.xhtml">
505{par_elements}</seq>
506</body>
507</smil>"#,
508        body_id = escape_xml(&overlay.body_id),
509        par_elements = par_elements,
510    )
511}

513fn format_smil_timecode(seconds: Option<f32>) -> String {
514    match seconds {
515        Some(s) => {
516            let total_ms = (s * 1000.0).round() as u64;
517            let ms = total_ms % 1000;
518            let total_s = total_ms / 1000;
519            let s_part = total_s % 60;
520            let total_m = total_s / 60;
521            let m_part = total_m % 60;
522            let h_part = total_m / 60;
523            format!("{:02}:{:02}:{:02}.{:03}", h_part, m_part, s_part, ms)
524        }
525        None => "0:00:00.000".to_string(),
526    }
527}
528
529fn format_duration(duration: Duration) -> String {
530    let total_secs = duration.as_secs_f32();
531    format_smil_timecode(Some(total_secs))
532}

534fn escape_xml(s: &str) -> String {
535    let mut out = String::with_capacity(s.len());
536    for c in s.chars() {
537        match c {
538            '&' => out.push_str("&amp;"),
539            '<' => out.push_str("&lt;"),
540            '>' => out.push_str("&gt;"),
541            '"' => out.push_str("&quot;"),
542            _ => out.push(c),
543        }
544    }
545    out
546}

548fn uuid_simple() -> String {
549    use std::time::{SystemTime, UNIX_EPOCH};
550    let secs = SystemTime::now()
551        .duration_since(UNIX_EPOCH)
552        .map(|d| d.as_secs())
553        .unwrap_or(0);
554    format!("{:08x}-{:04x}-{:04x}-{:04x}-{:012x}", secs, 0, 0, 0, 0)
555}

557fn date_now_str() -> String {
558    use std::time::{SystemTime, UNIX_EPOCH};
559    let d = SystemTime::now()
560        .duration_since(UNIX_EPOCH)
561        .map(|d| d.as_secs())
562        .unwrap_or(0);
563    let secs = d % 86400;
564    let days_since_epoch = d / 86400;
565    let (year, month, day) = days_to_date(days_since_epoch as i64);
566    format!(
567        "{}-{:02}-{:02}T{:02}:{:02}:{:02}Z",
568        year,
569        month,
570        day,
571        secs / 3600,
572        (secs % 3600) / 60,
573        secs % 60
574    )
575}

577fn days_to_date(days: i64) -> (i64, i64, i64) {
578    let year = 1970 + days / 365;
579    let remaining = days % 365;
580    let month = (remaining / 30) + 1;
581    let day = (remaining % 30) + 1;
582    (year.min(2100), month.min(12), day.min(31))
583}

585struct SimpleZip {
586    files: Vec<(String, Vec<u8>, bool)>,
587}
588
589impl SimpleZip {
590    fn new() -> Self {
591        Self { files: Vec::new() }
592    }
593
594    fn add_file(&mut self, path: &str, data: &[u8], stored: bool) {
595        self.files.push((path.to_string(), data.to_vec(), stored));
596    }
597
598    fn finish(&self) -> Result<Vec<u8>, String> {
599        let mut buf = Vec::new();
600        let mut central_offsets: Vec<(u64, u32, u32, String)> = Vec::new();
601
602        for (path, data, stored) in &self.files {
603            let offset = buf.len() as u64;
604            let crc = crc32(data);
605            let size = data.len() as u32;
606
607            buf.extend_from_slice(&[0x50, 0x4B, 0x03, 0x04]);
608            buf.extend_from_slice(&[20, 0]);
609            buf.extend_from_slice(&[0, 0]);
610            buf.extend_from_slice(if *stored { &[0, 0] } else { &[8, 0] });
611            buf.extend_from_slice(&[0, 0]);
612            buf.extend_from_slice(&[0, 0]);
613            buf.extend_from_slice(&crc.to_le_bytes());
614            buf.extend_from_slice(&size.to_le_bytes());
615            buf.extend_from_slice(&size.to_le_bytes());
616            let name_bytes = path.as_bytes();
617            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
618            buf.extend_from_slice(&[0u8; 10]);
619            buf.extend_from_slice(name_bytes);
620            buf.extend_from_slice(data);
621
622            central_offsets.push((offset, size, crc, path.clone()));
623        }
624
625        let central_start = buf.len() as u64;
626
627        for (offset, size, crc, name) in &central_offsets {
628            buf.extend_from_slice(&[0x50, 0x4B, 0x01, 0x02]);
629            buf.extend_from_slice(&[20, 0]);
630            buf.extend_from_slice(&[20, 0]);
631            buf.extend_from_slice(&[0, 0]);
632            buf.extend_from_slice(&[0, 0]);
633            buf.extend_from_slice(&[0, 0]);
634            buf.extend_from_slice(&[0, 0]);
635            buf.extend_from_slice(&crc.to_le_bytes());
636            buf.extend_from_slice(&size.to_le_bytes());
637            buf.extend_from_slice(&size.to_le_bytes());
638            let name_bytes = name.as_bytes();
639            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
640            buf.extend_from_slice(&[0u8; 10]);
641            buf.extend_from_slice(&[0, 0, 0]);
642            buf.extend_from_slice(&(*offset as u32).to_le_bytes());
643            buf.extend_from_slice(name_bytes);
644        }
645
646        let central_end = buf.len() as u64;
647        let central_count = central_offsets.len() as u16;
648
649        buf.extend_from_slice(&[0x50, 0x4B, 0x05, 0x06]);
650        buf.extend_from_slice(&[0, 0]);
651        buf.extend_from_slice(&[0, 0]);
652        buf.extend_from_slice(&central_count.to_le_bytes());
653        buf.extend_from_slice(&central_count.to_le_bytes());
654        buf.extend_from_slice(&(central_start as u32).to_le_bytes());
655        buf.extend_from_slice(&(central_end as u32).to_le_bytes());
656        buf.extend_from_slice(&[0u8; 2]);
657
658        Ok(buf)
659    }
660}

662fn crc32(data: &[u8]) -> u32 {
663    let mut crc: u32 = 0xFFFFFFFF;
664    for &byte in data {
665        crc ^= byte as u32;
666        for _ in 0..8 {
667            if crc & 1 != 0 {
668                crc = (crc >> 1) ^ 0xEDB88320;
669            } else {
670                crc >>= 1;
671            }
672        }
673    }
674    !crc
675}
