52    #[must_use = "building DOCX can fail; check the result"]
53    pub fn build_with_base_dir(
54        &self,
55        module: &SIRModuleV2,
56        base_dir: &str,
57    ) -> Result<Vec<u8>, DocxError> {
58        let image_refs = self.collect_image_refs(module);
59
60        let image_rid_map: HashMap<u32, &ImageRef> = image_refs
61            .iter()
62            .filter(|img| img.is_available(base_dir))
63            .map(|img| (img.node_id, img))
64            .collect();
65
66        let shared_image_map: Arc<HashMap<u32, &ImageRef>> = Arc::new(image_rid_map);
67
68        let document_xml = self.render_document_with_images(module, &shared_image_map);
69
70        let mut content_types_overrides = String::from(
71            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
72<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">
73  <Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>
74  <Default Extension="xml" ContentType="application/xml"/>
75  <Override PartName="/word/document.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml"/>
76  <Override PartName="/word/styles.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.styles+xml"/>
77  <Override PartName="/word/numbering.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.numbering+xml"/>
78  <Override PartName="/docProps/core.xml" ContentType="application/vnd.openxmlformats-package.core-properties+xml"/>"#,
79        );
80
81        let mut document_rels = String::from(
82            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
83<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
84  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>
85  <Relationship Id="rId2" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/numbering" Target="numbering.xml"/>"#,
86        );
87
88        let rels = r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
89<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
90  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="word/document.xml"/>
91  <Relationship Id="rId2" Type="http://schemas.openxmlformats.org/package/2006/relationships/metadata/core-properties" Target="docProps/core.xml"/>
92</Relationships>"#;
93
94        let styles_xml = self.render_styles();
95        let numbering_xml = self.render_numbering();
96        let core_xml = self.render_core_properties(module);
97
98        let mut zip = SimpleZip::new();
99
100        for &img in shared_image_map.values() {
101            let full_path = Path::new(base_dir).join(&img.source);
102            let data = std::fs::read(&full_path).map_err(|e| {
103                DocxError::BuildError(format!("failed to read image {}: {e}", img.source))
104            })?;
105            zip.add_file(&img.zip_path, &data, false);
106
107            content_types_overrides.push_str(&format!(
108                "\n  <Override PartName=\"/{}\" ContentType=\"{}\"/>",
109                img.zip_path, img.content_type
110            ));
111
112            document_rels.push_str(&format!(
113                "\n  <Relationship Id=\"{}\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/image\" Target=\"{}\"/>",
114                img.r_id, &img.zip_path[5..]
115            ));
116        }
117
118        content_types_overrides.push_str("\n</Types>");
119        document_rels.push_str("\n</Relationships>");
120
121        zip.add_file(
122            "[Content_Types].xml",
123            content_types_overrides.as_bytes(),
124            false,
125        );
126        zip.add_file("_rels/.rels", rels.as_bytes(), false);
127        zip.add_file("word/document.xml", document_xml.as_bytes(), false);
128        zip.add_file(
129            "word/_rels/document.xml.rels",
130            document_rels.as_bytes(),
131            false,
132        );
133        zip.add_file("word/styles.xml", styles_xml.as_bytes(), false);
134        zip.add_file("word/numbering.xml", numbering_xml.as_bytes(), false);
135        zip.add_file("docProps/core.xml", core_xml.as_bytes(), false);
136
137        zip.finish().map_err(DocxError::BuildError)
138    }

140    fn collect_image_refs(&self, module: &SIRModuleV2) -> Vec<ImageRef> {
141        let mut refs = Vec::new();
142        let mut counter = 0u32;
143        for &root_id in module.body.roots() {
144            if let Some(root) = module.body.get(root_id) {
145                self.collect_images_walk(module, root, &mut refs, &mut counter);
146            }
147        }
148        refs
149    }

151    fn collect_images_walk(
152        &self,
153        module: &SIRModuleV2,
154        node: &Node,
155        out: &mut Vec<ImageRef>,
156        counter: &mut u32,
157    ) {
158        if let NodeType::Image { source, alt, .. } = &node.node_type {
159            *counter += 1;
160            let ext = guess_image_extension(source);
161            let content_type = extension_to_mime(&ext);
162            let r_id = format!("rId{}", *counter + 2);
163            let zip_path = format!("word/media/image{}.{}", *counter, ext);
164
165            out.push(ImageRef {
166                r_id,
167                source: source.clone(),
168                content_type,
169                alt_text: alt.clone(),
170                zip_path,
171                node_id: node.id,
172            });
173        }
174
175        for &child_id in &node.child_ids {
176            if let Some(child) = module.body.get(child_id) {
177                self.collect_images_walk(module, child, out, counter);
178            }
179        }
180    }

182    fn render_document_with_images(
183        &self,
184        module: &SIRModuleV2,
185        image_map: &HashMap<u32, &ImageRef>,
186    ) -> String {
187        let mut body = String::new();
188        for &root_id in module.body.roots() {
189            if let Some(root) = module.body.get(root_id) {
190                self.render_node_with_images(&mut body, module, root, image_map);
191            }
192        }
193
194        format!(
195            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>

197            xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"
198            xmlns:a="http://schemas.openxmlformats.org/drawingml/2006/main"
199            xmlns:wp="http://schemas.openxmlformats.org/drawingml/2006/wordprocessingDrawing"
200            xmlns:pic="http://schemas.openxmlformats.org/drawingml/2006/picture">
201<w:body>
202{body}
203</w:body>
204</w:document>"#
205        )

208    fn render_node_with_images(
209        &self,
210        out: &mut String,
211        module: &SIRModuleV2,
212        node: &Node,
213        image_map: &HashMap<u32, &ImageRef>,
214    ) {
215        match &node.node_type {
216            NodeType::Document => {
217                for &child_id in &node.child_ids {
218                    if let Some(child) = module.body.get(child_id) {
219                        self.render_node_with_images(out, module, child, image_map);
220                    }
221                }
222            }
223
224            NodeType::Part => {
225                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Title\"/></w:pPr>");
226                for &child_id in &node.child_ids {
227                    if let Some(child) = module.body.get(child_id) {
228                        self.render_inline_with_images(out, module, child, image_map);
229                    }
230                }
231                out.push_str("</w:p>");
232            }
233
234            NodeType::Chapter
235            | NodeType::Section
236            | NodeType::Subsection
237            | NodeType::Subsubsection => {
238                let style = match &node.node_type {
239                    NodeType::Chapter => "Heading1",
240                    NodeType::Section => "Heading2",
241                    NodeType::Subsection => "Heading3",
242                    NodeType::Subsubsection => "Heading4",
243                    _ => "Heading2",
244                };
245                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"");
246                out.push_str(style);
247                out.push_str("\"/></w:pPr>");
248                for &child_id in &node.child_ids {
249                    if let Some(child) = module.body.get(child_id) {
250                        self.render_inline_with_images(out, module, child, image_map);
251                    }
252                }
253                out.push_str("</w:p>");
254            }
255
256            NodeType::Paragraph => {
257                out.push_str("<w:p>");
258                for &child_id in &node.child_ids {
259                    if let Some(child) = module.body.get(child_id) {
260                        self.render_inline_with_images(out, module, child, image_map);
261                    }
262                }
263                out.push_str("</w:p>");
264            }
265
266            NodeType::List { ordered, .. } => {
267                for &child_id in &node.child_ids {
268                    if let Some(child) = module.body.get(child_id) {
269                        self.render_list_item(out, module, child, *ordered);
270                    }
271                }
272            }
273
274            NodeType::ListItem => {
275                self.render_list_item(out, module, node, false);
276            }
277
278            NodeType::BlockQuote => {
279                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Quote\"/></w:pPr>");
280                for &child_id in &node.child_ids {
281                    if let Some(child) = module.body.get(child_id) {
282                        self.render_inline_with_images(out, module, child, image_map);
283                    }
284                }
285                out.push_str("</w:p>");
286            }
287
288            NodeType::CodeBlock { language, .. } => {
289                out.push_str("<w:p><w:pPr><w:shd w:val=\"clear\" w:color=\"auto\" w:fill=\"F5F5F5\"/></w:pPr>");
290                if let Some(lang) = language {
291                    out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/><w:color w:val=\"606060\"/></w:rPr><w:t xml:space=\"preserve\">");
292                    out.push_str(&format!("[{}]\n", lang));
293                    out.push_str("</w:t></w:r>");
294                }
295                let text = module.body.collect_text(node.id);
296                out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/></w:rPr><w:t xml:space=\"preserve\">");
297                out.push_str(&escape_xml(&text));
298                out.push_str("</w:t></w:r></w:p>");
299            }
300
301            NodeType::MathBlock { .. } => {
302                let text = module.body.collect_text(node.id);
303                out.push_str("<w:p><w:r><w:rPr><w:i/><w:color w:val=\"0000CC\"/></w:rPr><w:t xml:space=\"preserve\">");
304                out.push_str(&escape_xml(&text));
305                out.push_str("</w:t></w:r></w:p>");
306            }
307
308            NodeType::Table { .. } => {
309                out.push_str("<w:tbl><w:tblPr><w:tblBorders>");
310                out.push_str(
311                    "<w:top w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
312                );
313                out.push_str(
314                    "<w:left w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
315                );
316                out.push_str(
317                    "<w:bottom w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
318                );
319                out.push_str(
320                    "<w:right w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
321                );
322                out.push_str(
323                    "<w:insideH w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
324                );
325                out.push_str(
326                    "<w:insideV w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
327                );
328                out.push_str("</w:tblBorders></w:tblPr>");
329                for &child_id in &node.child_ids {
330                    if let Some(child) = module.body.get(child_id) {
331                        self.render_table_row(out, module, child);
332                    }
333                }
334                out.push_str("</w:tbl>");
335            }
336
337            NodeType::ThematicBreak => {
338                out.push_str("<w:p><w:pPr><w:pBdr><w:bottom w:val=\"single\" w:sz=\"6\" w:space=\"1\" w:color=\"999999\"/></w:pBdr></w:pPr></w:p>");
339            }
340
341            NodeType::FootnoteBlock => {
342                for node in module.body.iter() {
343                    if let NodeType::Footnote { content } = &node.node_type {
344                        out.push_str("<w:p><w:r><w:rPr><w:sz w:val=\"16\"/></w:rPr><w:t>");
345                        out.push_str(&escape_xml(content));
346                        out.push_str("</w:t></w:r></w:p>");
347                    }
348                }
349            }
350
351            NodeType::Figure { .. } => {
352                out.push_str("<w:p>");
353                for &child_id in &node.child_ids {
354                    if let Some(child) = module.body.get(child_id) {
355                        if let NodeType::Image { .. } = &child.node_type {
356                            if let Some(img_ref) = image_map.get(&child.id) {
357                                emit_inline_image(out, img_ref);
358                            } else if let NodeType::Image { alt, .. } = &child.node_type {
359                                out.push_str("<w:r><w:t>");
360                                out.push_str(&escape_xml(alt));
361                                out.push_str(" [image not embedded]</w:t></w:r>");
362                            }
363                        } else {
364                            self.render_node_with_images(out, module, child, image_map);
365                        }
366                    }
367                }
368                out.push_str("</w:p>");
369            }
370
371            NodeType::Caption => {
372                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Caption\"/></w:pPr>");
373                for &child_id in &node.child_ids {
374                    if let Some(child) = module.body.get(child_id) {
375                        self.render_inline_with_images(out, module, child, image_map);
376                    }
377                }
378                out.push_str("</w:p>");
379            }
380
381            NodeType::PageBreak => {
382                out.push_str("<w:p><w:r><w:br w:type=\"page\"/></w:r></w:p>");
383            }
384
385            NodeType::TableOfContents { .. } => {
386                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"TOCHeading\"/></w:pPr><w:r><w:t>Table of Contents</w:t></w:r></w:p>");
387                out.push_str("<w:p><w:r><w:fldChar w:fldCharType=\"begin\"/></w:r><w:r><w:instrText xml:space=\"preserve\"> TOC \\o \"1-3\" \\h \\z \\u </w:instrText><w:r><w:fldChar w:fldCharType=\"separate\"/></w:r><w:r><w:t>[Update this field in Word to generate TOC]</w:t></w:r><w:r><w:fldChar w:fldCharType=\"end\"/></w:r></w:p>");
388            }
389
390            NodeType::Citation { keys, .. } => {
391                out.push_str("<w:r><w:rPr><w:vertAlign w:val=\"superscript\"/><w:sz w:val=\"16\"/></w:rPr><w:t>");
392                out.push_str(&escape_xml(&keys.join(", ")));
393                out.push_str("</w:t></w:r>");
394            }
395
396            NodeType::Reference { label } => {
397                out.push_str("<w:r><w:rPr><w:color w:val=\"0066CC\"/></w:rPr><w:t>");
398                out.push_str(&escape_xml(&format!("[ref: {}]", label)));
399                out.push_str("</w:t></w:r>");
400            }
401
402            _ => {
403                for &child_id in &node.child_ids {
404                    if let Some(child) = module.body.get(child_id) {
405                        self.render_node_with_images(out, module, child, image_map);
406                    }
407                }
408            }
409        }
410    }

516                out.push_str("</w:t></w:r>");
517            }

520                out.push_str("<w:r><w:br/></w:r>");
521            }
522
523            NodeType::Footnote { content } => {
524                out.push_str("<w:r><w:rPr><w:vertAlign w:val=\"superscript\"/><w:sz w:val=\"16\"/></w:rPr><w:t>");
525                out.push_str(&escape_xml(content));
526                out.push_str("</w:t></w:r>");
527            }
528
529            NodeType::Styled { .. } | NodeType::Group => {
530                for &child_id in &node.child_ids {
531                    if let Some(child) = module.body.get(child_id) {
532                        self.render_inline_with_images(out, module, child, image_map);
533                    }
534                }
535            }
536
537            _ => {
538                let text = module.body.collect_text(node.id);
539                if !text.is_empty() {
540                    out.push_str("<w:r><w:t>");
541                    out.push_str(&escape_xml(&text));
542                    out.push_str("</w:t></w:r>");
543                }
544            }
545        }

548    fn render_inline(&self, out: &mut String, module: &SIRModuleV2, node: &Node) {
549        match &node.node_type {
550            NodeType::Text { content } => {
551                out.push_str("<w:r><w:t>");
552                out.push_str(&escape_xml(content));
553                out.push_str("</w:t></w:r>");

913fn escape_xml(s: &str) -> String {
914    let mut out = String::with_capacity(s.len());
915    for c in s.chars() {
916        match c {
917            '&' => out.push_str("&amp;"),
918            '<' => out.push_str("&lt;"),
919            '>' => out.push_str("&gt;"),
920            '"' => out.push_str("&quot;"),
921            '\'' => out.push_str("&apos;"),
922            _ => out.push(c),
923        }
924    }
925    out
926}

928fn format_iso8601(secs: u64) -> String {
929    let secs = secs as i64;
930    let year = 1970 + secs / 31_536_000;
931    let rem = secs % 31_536_000;
932    let month = 1 + rem / 2_592_000;
933    let rem = rem % 2_592_000;
934    let day = 1 + rem / 86_400;
935    format!("{year:04}-{month:02}-{day:02}T00:00:00Z")
936}

938struct SimpleZip {
939    files: Vec<(String, Vec<u8>, bool)>,
940}
941
942impl SimpleZip {
943    fn new() -> Self {
944        Self { files: Vec::new() }
945    }
946
947    fn add_file(&mut self, path: &str, data: &[u8], stored: bool) {
948        self.files.push((path.to_string(), data.to_vec(), stored));
949    }
950
951    fn finish(&self) -> Result<Vec<u8>, String> {
952        let mut buf = Vec::new();
953        let mut central_offsets: Vec<(u64, u32, u32, String)> = Vec::new();
954
955        for (path, data, stored) in &self.files {
956            let offset = buf.len() as u64;
957            let crc = crc32(data);
958            let size = data.len() as u32;
959
960            buf.extend_from_slice(&[0x50, 0x4B, 0x03, 0x04]);
961            buf.extend_from_slice(&[20, 0]);
962            buf.extend_from_slice(&[0, 0]);
963            buf.extend_from_slice(if *stored { &[0, 0] } else { &[8, 0] });
964            buf.extend_from_slice(&[0, 0]);
965            buf.extend_from_slice(&[0, 0]);
966            buf.extend_from_slice(&crc.to_le_bytes());
967            buf.extend_from_slice(&size.to_le_bytes());
968            buf.extend_from_slice(&size.to_le_bytes());
969            let name_bytes = path.as_bytes();
970            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
971            buf.extend_from_slice(&[0u8; 10]);
972            buf.extend_from_slice(name_bytes);
973            buf.extend_from_slice(data);
974
975            central_offsets.push((offset, size, crc, path.clone()));
976        }
977
978        let central_start = buf.len() as u64;
979
980        for (offset, size, crc, name) in &central_offsets {
981            buf.extend_from_slice(&[0x50, 0x4B, 0x01, 0x02]);
982            buf.extend_from_slice(&[20, 0]);
983            buf.extend_from_slice(&[20, 0]);
984            buf.extend_from_slice(&[0, 0]);
985            buf.extend_from_slice(&[0, 0]);
986            buf.extend_from_slice(&[0, 0]);
987            buf.extend_from_slice(&[0, 0]);
988            buf.extend_from_slice(&crc.to_le_bytes());
989            buf.extend_from_slice(&size.to_le_bytes());
990            buf.extend_from_slice(&size.to_le_bytes());
991            let name_bytes = name.as_bytes();
992            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
993            buf.extend_from_slice(&[0u8; 10]);
994            buf.extend_from_slice(&[0, 0, 0]);
995            buf.extend_from_slice(&(*offset as u32).to_le_bytes());
996            buf.extend_from_slice(name_bytes);
997        }
998
999        let central_end = buf.len() as u64;
1000        let central_count = central_offsets.len() as u16;
1001
1002        buf.extend_from_slice(&[0x50, 0x4B, 0x05, 0x06]);
1003        buf.extend_from_slice(&[0, 0]);
1004        buf.extend_from_slice(&[0, 0]);
1005        buf.extend_from_slice(&central_count.to_le_bytes());
1006        buf.extend_from_slice(&central_count.to_le_bytes());
1007        buf.extend_from_slice(&(central_start as u32).to_le_bytes());
1008        buf.extend_from_slice(&(central_end as u32).to_le_bytes());
1009        buf.extend_from_slice(&[0u8; 2]);
1010
1011        Ok(buf)
1012    }
1013}

1015fn crc32(data: &[u8]) -> u32 {
1016    let mut crc: u32 = 0xFFFFFFFF;
1017    for &byte in data {
1018        crc ^= byte as u32;
1019        for _ in 0..8 {
1020            if crc & 1 != 0 {
1021                crc = (crc >> 1) ^ 0xEDB88320;
1022            } else {
1023                crc >>= 1;
1024            }
1025        }
1026    }
1027    !crc
1028}
