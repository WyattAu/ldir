25    pub fn build(&self, module: &SIRModuleV2) -> Result<Vec<u8>, DocxError> {
26        let document_xml = self.render_document(module);
27
28        let content_types = r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
29<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">
30  <Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>
31  <Default Extension="xml" ContentType="application/xml"/>
32  <Override PartName="/word/document.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml"/>
33  <Override PartName="/word/styles.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.styles+xml"/>
34  <Override PartName="/word/numbering.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.numbering+xml"/>
35  <Override PartName="/docProps/core.xml" ContentType="application/vnd.openxmlformats-package.core-properties+xml"/>
36</Types>"#;
37
38        let rels = r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
39<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
40  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="word/document.xml"/>
41  <Relationship Id="rId2" Type="http://schemas.openxmlformats.org/package/2006/relationships/metadata/core-properties" Target="docProps/core.xml"/>
42</Relationships>"#;
43
44        let document_rels = r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
45<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
46  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>
47  <Relationship Id="rId2" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/numbering" Target="numbering.xml"/>
48</Relationships>"#;
49
50        let styles_xml = self.render_styles();
51        let numbering_xml = self.render_numbering();
52        let core_xml = self.render_core_properties(module);
53
54        let mut zip = SimpleZip::new();
55        zip.add_file("[Content_Types].xml", content_types.as_bytes(), false);
56        zip.add_file("_rels/.rels", rels.as_bytes(), false);
57        zip.add_file("word/document.xml", document_xml.as_bytes(), false);
58        zip.add_file(
59            "word/_rels/document.xml.rels",
60            document_rels.as_bytes(),
61            false,
62        );
63        zip.add_file("word/styles.xml", styles_xml.as_bytes(), false);
64        zip.add_file("word/numbering.xml", numbering_xml.as_bytes(), false);
65        zip.add_file("docProps/core.xml", core_xml.as_bytes(), false);
66
67        zip.finish().map_err(DocxError::BuildError)
68    }

70    fn render_document(&self, module: &SIRModuleV2) -> String {
71        let mut body = String::new();
72        for &root_id in module.body.roots() {
73            if let Some(root) = module.body.get(root_id) {
74                self.render_node(&mut body, module, root);
75            }
76        }
77
78        format!(
79            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
80<w:document xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main">
81<w:body>
82{body}
83</w:body>
84</w:document>"#
85        )
86    }

88    fn render_node(&self, out: &mut String, module: &SIRModuleV2, node: &Node) {
89        match &node.node_type {
90            NodeType::Document => {
91                for &child_id in &node.child_ids {
92                    if let Some(child) = module.body.get(child_id) {
93                        self.render_node(out, module, child);
94                    }
95                }
96            }
97
98            NodeType::Part => {
99                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Title\"/></w:pPr>");
100                for &child_id in &node.child_ids {
101                    if let Some(child) = module.body.get(child_id) {
102                        self.render_inline(out, module, child);
103                    }
104                }
105                out.push_str("</w:p>");
106            }
107
108            NodeType::Chapter
109            | NodeType::Section
110            | NodeType::Subsection
111            | NodeType::Subsubsection => {
112                let style = match &node.node_type {
113                    NodeType::Chapter => "Heading1",
114                    NodeType::Section => "Heading2",
115                    NodeType::Subsection => "Heading3",
116                    NodeType::Subsubsection => "Heading4",
117                    _ => "Heading2",
118                };
119                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"");
120                out.push_str(style);
121                out.push_str("\"/></w:pPr>");
122                for &child_id in &node.child_ids {
123                    if let Some(child) = module.body.get(child_id) {
124                        self.render_inline(out, module, child);
125                    }
126                }
127                out.push_str("</w:p>");
128            }
129
130            NodeType::Paragraph => {
131                out.push_str("<w:p>");
132                for &child_id in &node.child_ids {
133                    if let Some(child) = module.body.get(child_id) {
134                        self.render_inline(out, module, child);
135                    }
136                }
137                out.push_str("</w:p>");
138            }
139
140            NodeType::List { ordered, .. } => {
141                for &child_id in &node.child_ids {
142                    if let Some(child) = module.body.get(child_id) {
143                        self.render_list_item(out, module, child, *ordered);
144                    }
145                }
146            }
147
148            NodeType::ListItem => {
149                self.render_list_item(out, module, node, false);
150            }
151
152            NodeType::BlockQuote => {
153                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Quote\"/></w:pPr>");
154                for &child_id in &node.child_ids {
155                    if let Some(child) = module.body.get(child_id) {
156                        self.render_inline(out, module, child);
157                    }
158                }
159                out.push_str("</w:p>");
160            }
161
162            NodeType::CodeBlock { language, .. } => {
163                out.push_str("<w:p><w:pPr><w:shd w:val=\"clear\" w:color=\"auto\" w:fill=\"F5F5F5\"/></w:pPr>");
164                if let Some(lang) = language {
165                    out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/><w:color w:val=\"606060\"/></w:rPr><w:t xml:space=\"preserve\">");
166                    out.push_str(&format!("[{}]\n", lang));
167                    out.push_str("</w:t></w:r>");
168                }
169                let text = module.body.collect_text(node.id);
170                out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/></w:rPr><w:t xml:space=\"preserve\">");
171                out.push_str(&escape_xml(&text));
172                out.push_str("</w:t></w:r></w:p>");
173            }
174
175            NodeType::MathBlock { .. } => {
176                let text = module.body.collect_text(node.id);
177                out.push_str("<w:p><w:r><w:rPr><w:i/><w:color w:val=\"0000CC\"/></w:rPr><w:t xml:space=\"preserve\">");
178                out.push_str(&escape_xml(&text));
179                out.push_str("</w:t></w:r></w:p>");
180            }
181
182            NodeType::Table { .. } => {
183                out.push_str("<w:tbl><w:tblPr><w:tblBorders>");
184                out.push_str(
185                    "<w:top w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
186                );
187                out.push_str(
188                    "<w:left w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
189                );
190                out.push_str(
191                    "<w:bottom w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
192                );
193                out.push_str(
194                    "<w:right w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
195                );
196                out.push_str(
197                    "<w:insideH w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
198                );
199                out.push_str(
200                    "<w:insideV w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
201                );
202                out.push_str("</w:tblBorders></w:tblPr>");
203                for &child_id in &node.child_ids {
204                    if let Some(child) = module.body.get(child_id) {
205                        self.render_table_row(out, module, child);
206                    }
207                }
208                out.push_str("</w:tbl>");
209            }
210
211            NodeType::ThematicBreak => {
212                out.push_str("<w:p><w:pPr><w:pBdr><w:bottom w:val=\"single\" w:sz=\"6\" w:space=\"1\" w:color=\"999999\"/></w:pBdr></w:pPr></w:p>");
213            }
214
215            NodeType::FootnoteBlock => {
216                for node in module.body.iter() {
217                    if let NodeType::Footnote { content } = &node.node_type {
218                        out.push_str("<w:p><w:r><w:rPr><w:sz w:val=\"16\"/></w:rPr><w:t>");
219                        out.push_str(&escape_xml(content));
220                        out.push_str("</w:t></w:r></w:p>");
221                    }
222                }
223            }
224
225            NodeType::Figure { .. } => {
226                out.push_str("<w:p>");
227                for &child_id in &node.child_ids {
228                    if let Some(child) = module.body.get(child_id) {
229                        match &child.node_type {
230                            NodeType::Image { alt, .. } => {
231                                out.push_str("<w:r><w:t>");
232                                out.push_str(&escape_xml(alt));
233                                out.push_str(" [image not embedded]</w:t></w:r>");
234                            }
235                            _ => self.render_node(out, module, child),
236                        }
237                    }
238                }
239                out.push_str("</w:p>");
240            }
241
242            NodeType::Caption => {
243                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Caption\"/></w:pPr>");
244                for &child_id in &node.child_ids {
245                    if let Some(child) = module.body.get(child_id) {
246                        self.render_inline(out, module, child);
247                    }
248                }
249                out.push_str("</w:p>");
250            }
251
252            NodeType::PageBreak => {
253                out.push_str("<w:p><w:r><w:br w:type=\"page\"/></w:r></w:p>");
254            }
255
256            NodeType::TableOfContents { .. } => {
257                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"TOCHeading\"/></w:pPr><w:r><w:t>Table of Contents</w:t></w:r></w:p>");
258                out.push_str("<w:p><w:r><w:fldChar w:fldCharType=\"begin\"/></w:r><w:r><w:instrText xml:space=\"preserve\"> TOC \\o \"1-3\" \\h \\z \\u </w:instrText><w:r><w:fldChar w:fldCharType=\"separate\"/></w:r><w:r><w:t>[Update this field in Word to generate TOC]</w:t></w:r><w:r><w:fldChar w:fldCharType=\"end\"/></w:r></w:p>");
259            }
260
261            NodeType::Citation { keys, .. } => {
262                out.push_str("<w:r><w:rPr><w:vertAlign w:val=\"superscript\"/><w:sz w:val=\"16\"/></w:rPr><w:t>");
263                out.push_str(&escape_xml(&keys.join(", ")));
264                out.push_str("</w:t></w:r>");
265            }
266
267            NodeType::Reference { label } => {
268                out.push_str("<w:r><w:rPr><w:color w:val=\"0066CC\"/></w:rPr><w:t>");
269                out.push_str(&escape_xml(&format!("[ref: {}]", label)));
270                out.push_str("</w:t></w:r>");
271            }
272
273            _ => {
274                for &child_id in &node.child_ids {
275                    if let Some(child) = module.body.get(child_id) {
276                        self.render_node(out, module, child);
277                    }
278                }
279            }
280        }
281    }

283    fn render_inline(&self, out: &mut String, module: &SIRModuleV2, node: &Node) {
284        match &node.node_type {
285            NodeType::Text { content } => {
286                out.push_str("<w:r><w:t>");
287                out.push_str(&escape_xml(content));
288                out.push_str("</w:t></w:r>");
289            }
290
291            NodeType::Bold => {
292                out.push_str("<w:r><w:rPr><w:b/></w:rPr>");
293                for &child_id in &node.child_ids {
294                    if let Some(child) = module.body.get(child_id) {
295                        self.render_inline(out, module, child);
296                    }
297                }
298                out.push_str("</w:r>");
299            }
300
301            NodeType::Italic => {
302                out.push_str("<w:r><w:rPr><w:i/></w:rPr>");
303                for &child_id in &node.child_ids {
304                    if let Some(child) = module.body.get(child_id) {
305                        self.render_inline(out, module, child);
306                    }
307                }
308                out.push_str("</w:r>");
309            }
310
311            NodeType::Mono => {
312                out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/></w:rPr>");
313                for &child_id in &node.child_ids {
314                    if let Some(child) = module.body.get(child_id) {
315                        self.render_inline(out, module, child);
316                    }
317                }
318                out.push_str("</w:r>");
319            }
320
321            NodeType::Underline => {
322                out.push_str("<w:r><w:rPr><w:u w:val=\"single\"/></w:rPr>");
323                for &child_id in &node.child_ids {
324                    if let Some(child) = module.body.get(child_id) {
325                        self.render_inline(out, module, child);
326                    }
327                }
328                out.push_str("</w:r>");
329            }
330
331            NodeType::Strikethrough => {
332                out.push_str("<w:r><w:rPr><w:strike/></w:rPr>");
333                for &child_id in &node.child_ids {
334                    if let Some(child) = module.body.get(child_id) {
335                        self.render_inline(out, module, child);
336                    }
337                }
338                out.push_str("</w:r>");
339            }
340
341            NodeType::SmallCaps => {
342                out.push_str("<w:r><w:rPr><w:smallCaps/></w:rPr>");
343                for &child_id in &node.child_ids {
344                    if let Some(child) = module.body.get(child_id) {
345                        self.render_inline(out, module, child);
346                    }
347                }
348                out.push_str("</w:r>");
349            }
350
351            NodeType::Link { url, title } => {
352                let text = module.body.collect_text(node.id);
353                out.push_str(
354                    "<w:r><w:rPr><w:color w:val=\"0066CC\"/><w:u w:val=\"single\"/></w:rPr><w:t>",
355                );
356                out.push_str(&escape_xml(&text));
357                out.push_str("</w:t></w:r>");
358                if let Some(t) = title {
359                    out.push_str("<w:r><w:rPr><w:color w:val=\"0066CC\"/></w:rPr><w:t xml:space=\"preserve\"> (");
360                    out.push_str(&escape_xml(t));
361                    out.push_str(")</w:t></w:r>");
362                }
363                out.push_str("<w:r><w:rPr><w:sz w:val=\"16\"/><w:color w:val=\"0066CC\"/></w:rPr><w:t xml:space=\"preserve\"> [");
364                out.push_str(&escape_xml(url));
365                out.push_str("]</w:t></w:r>");
366            }
367
368            NodeType::Image { alt, .. } => {
369                out.push_str("<w:r><w:t>[");
370                out.push_str(&escape_xml(alt));
371                out.push_str("]</w:t></w:r>");
372            }
373
374            NodeType::MathInline { content } => {
375                out.push_str("<w:r><w:rPr><w:i/></w:rPr><w:t>");
376                out.push_str(&escape_xml(content));
377                out.push_str("</w:t></w:r>");
378            }
379
380            NodeType::LineBreak => {
381                out.push_str("<w:r><w:br/></w:r>");
382            }
383
384            NodeType::Footnote { content } => {
385                out.push_str("<w:r><w:rPr><w:vertAlign w:val=\"superscript\"/><w:sz w:val=\"16\"/></w:rPr><w:t>");
386                out.push_str(&escape_xml(content));
387                out.push_str("</w:t></w:r>");
388            }
389
390            NodeType::Styled { .. } | NodeType::Group => {
391                for &child_id in &node.child_ids {
392                    if let Some(child) = module.body.get(child_id) {
393                        self.render_inline(out, module, child);
394                    }
395                }
396            }
397
398            _ => {
399                let text = module.body.collect_text(node.id);
400                if !text.is_empty() {
401                    out.push_str("<w:r><w:t>");
402                    out.push_str(&escape_xml(&text));
403                    out.push_str("</w:t></w:r>");
404                }
405            }
406        }
407    }

409    fn render_list_item(&self, out: &mut String, module: &SIRModuleV2, node: &Node, ordered: bool) {
410        let num_id = if ordered { "2" } else { "1" };
411        out.push_str(&format!(
412            "<w:p><w:pPr><w:pStyle w:val=\"ListParagraph\"/><w:numPr><w:ilvl w:val=\"0\"/><w:numId w:val=\"{num_id}\"/></w:numPr></w:pPr>"
413        ));
414        for &child_id in &node.child_ids {
415            if let Some(child) = module.body.get(child_id) {
416                self.render_inline(out, module, child);
417            }
418        }
419        out.push_str("</w:p>");
420    }

422    fn render_table_row(&self, out: &mut String, module: &SIRModuleV2, node: &Node) {
423        let is_header = matches!(&node.node_type, NodeType::TableRow { is_header: true });
424        out.push_str("<w:tr>");
425        if is_header {
426            out.push_str("<w:trPr><w:tblHeader/></w:trPr>");
427        }
428        for &child_id in &node.child_ids {
429            if let Some(child) = module.body.get(child_id)
430                && matches!(&child.node_type, NodeType::TableCell { .. })
431            {
432                out.push_str("<w:tc><w:p>");
433                if is_header {
434                    out.push_str("<w:pPr><w:jc w:val=\"center\"/></w:pPr>");
435                }
436                for &cell_child_id in &child.child_ids {
437                    if let Some(cell_child) = module.body.get(cell_child_id) {
438                        if is_header {
439                            out.push_str("<w:r><w:rPr><w:b/></w:rPr>");
440                            self.render_inline(out, module, cell_child);
441                            out.push_str("</w:r>");
442                        } else {
443                            self.render_inline(out, module, cell_child);
444                        }
445                    }
446                }
447                out.push_str("</w:p></w:tc>");
448            }
449        }
450        out.push_str("</w:tr>");
451    }

558fn escape_xml(s: &str) -> String {
559    let mut out = String::with_capacity(s.len());
560    for c in s.chars() {
561        match c {
562            '&' => out.push_str("&amp;"),
563            '<' => out.push_str("&lt;"),
564            '>' => out.push_str("&gt;"),
565            '"' => out.push_str("&quot;"),
566            '\'' => out.push_str("&apos;"),
567            _ => out.push(c),
568        }
569    }
570    out
571}

573fn format_iso8601(secs: u64) -> String {
574    let secs = secs as i64;
575    let year = 1970 + secs / 31_536_000;
576    let rem = secs % 31_536_000;
577    let month = 1 + rem / 2_592_000;
578    let rem = rem % 2_592_000;
579    let day = 1 + rem / 86_400;
580    format!("{year:04}-{month:02}-{day:02}T00:00:00Z")
581}

583struct SimpleZip {
584    files: Vec<(String, Vec<u8>, bool)>,
585}
586
587impl SimpleZip {
588    fn new() -> Self {
589        Self { files: Vec::new() }
590    }
591
592    fn add_file(&mut self, path: &str, data: &[u8], stored: bool) {
593        self.files.push((path.to_string(), data.to_vec(), stored));
594    }
595
596    fn finish(&self) -> Result<Vec<u8>, String> {
597        let mut buf = Vec::new();
598        let mut central_offsets: Vec<(u64, u32, u32, String)> = Vec::new();
599
600        for (path, data, stored) in &self.files {
601            let offset = buf.len() as u64;
602            let crc = crc32(data);
603            let size = data.len() as u32;
604
605            buf.extend_from_slice(&[0x50, 0x4B, 0x03, 0x04]);
606            buf.extend_from_slice(&[20, 0]);
607            buf.extend_from_slice(&[0, 0]);
608            buf.extend_from_slice(if *stored { &[0, 0] } else { &[8, 0] });
609            buf.extend_from_slice(&[0, 0]);
610            buf.extend_from_slice(&[0, 0]);
611            buf.extend_from_slice(&crc.to_le_bytes());
612            buf.extend_from_slice(&size.to_le_bytes());
613            buf.extend_from_slice(&size.to_le_bytes());
614            let name_bytes = path.as_bytes();
615            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
616            buf.extend_from_slice(&[0u8; 10]);
617            buf.extend_from_slice(name_bytes);
618            buf.extend_from_slice(data);
619
620            central_offsets.push((offset, size, crc, path.clone()));
621        }
622
623        let central_start = buf.len() as u64;
624
625        for (offset, size, crc, name) in &central_offsets {
626            buf.extend_from_slice(&[0x50, 0x4B, 0x01, 0x02]);
627            buf.extend_from_slice(&[20, 0]);
628            buf.extend_from_slice(&[20, 0]);
629            buf.extend_from_slice(&[0, 0]);
630            buf.extend_from_slice(&[0, 0]);
631            buf.extend_from_slice(&[0, 0]);
632            buf.extend_from_slice(&[0, 0]);
633            buf.extend_from_slice(&crc.to_le_bytes());
634            buf.extend_from_slice(&size.to_le_bytes());
635            buf.extend_from_slice(&size.to_le_bytes());
636            let name_bytes = name.as_bytes();
637            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
638            buf.extend_from_slice(&[0u8; 10]);
639            buf.extend_from_slice(&[0, 0, 0]);
640            buf.extend_from_slice(&(*offset as u32).to_le_bytes());
641            buf.extend_from_slice(name_bytes);
642        }
643
644        let central_end = buf.len() as u64;
645        let central_count = central_offsets.len() as u16;
646
647        buf.extend_from_slice(&[0x50, 0x4B, 0x05, 0x06]);
648        buf.extend_from_slice(&[0, 0]);
649        buf.extend_from_slice(&[0, 0]);
650        buf.extend_from_slice(&central_count.to_le_bytes());
651        buf.extend_from_slice(&central_count.to_le_bytes());
652        buf.extend_from_slice(&(central_start as u32).to_le_bytes());
653        buf.extend_from_slice(&(central_end as u32).to_le_bytes());
654        buf.extend_from_slice(&[0u8; 2]);
655
656        Ok(buf)
657    }
658}
659
660fn crc32(data: &[u8]) -> u32 {
661    let mut crc: u32 = 0xFFFFFFFF;
662    for &byte in data {
663        crc ^= byte as u32;
664        for _ in 0..8 {
665            if crc & 1 != 0 {
666                crc = (crc >> 1) ^ 0xEDB88320;
667            } else {
668                crc >>= 1;
669            }
670        }
671    }
672    !crc
673}
