79    pub fn build_with_base_dir(
80        &self,
81        module: &SIRModuleV2,
82        base_dir: &str,
83    ) -> Result<Vec<u8>, DocxError> {
84        let image_refs = self.collect_image_refs(module);
85
86        let image_rid_map: HashMap<u32, &ImageRef> = image_refs
87            .iter()
88            .filter(|img| img.is_available(base_dir))
89            .map(|img| (img.node_id, img))
90            .collect();
91
92        let shared_image_map: Arc<HashMap<u32, &ImageRef>> = Arc::new(image_rid_map);
93
94        let mut notes = CollectedNotes {
95            footnotes: Vec::new(),
96            endnotes: Vec::new(),
97            comments: Vec::new(),
98        };
99
100        let document_xml =
101            self.render_document_with_images_and_notes(module, &shared_image_map, &mut notes);
102
103        let mut content_types_overrides = String::from(
104            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
105<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">
106  <Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>
107  <Default Extension="xml" ContentType="application/xml"/>
108  <Override PartName="/word/document.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml"/>
109  <Override PartName="/word/styles.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.styles+xml"/>
110  <Override PartName="/word/numbering.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.numbering+xml"/>
111  <Override PartName="/docProps/core.xml" ContentType="application/vnd.openxmlformats-package.core-properties+xml"/>"#,
112        );
113
114        let mut document_rels = String::from(
115            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
116<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
117  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>
118  <Relationship Id="rId2" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/numbering" Target="numbering.xml"/>"#,
119        );
120
121        let rels = r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
122<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
123  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="word/document.xml"/>
124  <Relationship Id="rId2" Type="http://schemas.openxmlformats.org/package/2006/relationships/metadata/core-properties" Target="docProps/core.xml"/>
125</Relationships>"#;
126
127        let styles_xml = self.render_styles();
128        let numbering_xml = self.render_numbering();
129        let core_xml = self.render_core_properties(module);
130
131        let mut zip = SimpleZip::new();
132
133        for &img in shared_image_map.values() {
134            let full_path = Path::new(base_dir).join(&img.source);
135            let data = std::fs::read(&full_path).map_err(|e| {
136                DocxError::BuildError(format!("failed to read image {}: {e}", img.source))
137            })?;
138            zip.add_file(&img.zip_path, &data, false);
139
140            content_types_overrides.push_str(&format!(
141                "\n  <Override PartName=\"/{}\" ContentType=\"{}\"/>",
142                img.zip_path, img.content_type
143            ));
144
145            document_rels.push_str(&format!(
146                "\n  <Relationship Id=\"{}\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/image\" Target=\"{}\"/>",
147                img.r_id, &img.zip_path[5..]
148            ));
149        }
150
151        if !notes.footnotes.is_empty() {
152            let footnotes_xml = render_footnotes_xml(&notes.footnotes);
153            zip.add_file("word/footnotes.xml", footnotes_xml.as_bytes(), false);
154
155            content_types_overrides.push_str(
156                "\n  <Override PartName=\"/word/footnotes.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.wordprocessingml.footnotes+xml\"/>",
157            );
158
159            let rid = "rIdFootnotes";
160            document_rels.push_str(&format!(
161                "\n  <Relationship Id=\"{}\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/footnotes\" Target=\"footnotes.xml\"/>",
162                rid
163            ));
164        }
165
166        if !notes.endnotes.is_empty() {
167            let endnotes_xml = render_endnotes_xml(&notes.endnotes);
168            zip.add_file("word/endnotes.xml", endnotes_xml.as_bytes(), false);
169
170            content_types_overrides.push_str(
171                "\n  <Override PartName=\"/word/endnotes.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.wordprocessingml.endnotes+xml\"/>",
172            );
173
174            let rid = "rIdEndnotes";
175            document_rels.push_str(&format!(
176                "\n  <Relationship Id=\"{}\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/endnotes\" Target=\"endnotes.xml\"/>",
177                rid
178            ));
179        }
180
181        if !notes.comments.is_empty() {
182            let comments_xml = render_comments_xml(&notes.comments);
183            zip.add_file("word/comments.xml", comments_xml.as_bytes(), false);
184
185            content_types_overrides.push_str(
186                "\n  <Override PartName=\"/word/comments.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.wordprocessingml.comments+xml\"/>",
187            );
188
189            let rid = "rIdComments";
190            document_rels.push_str(&format!(
191                "\n  <Relationship Id=\"{}\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/comments\" Target=\"comments.xml\"/>",
192                rid
193            ));
194        }
195
196        content_types_overrides.push_str("\n</Types>");
197        document_rels.push_str("\n</Relationships>");
198
199        zip.add_file(
200            "[Content_Types].xml",
201            content_types_overrides.as_bytes(),
202            false,
203        );
204        zip.add_file("_rels/.rels", rels.as_bytes(), false);
205        zip.add_file("word/document.xml", document_xml.as_bytes(), false);
206        zip.add_file(
207            "word/_rels/document.xml.rels",
208            document_rels.as_bytes(),
209            false,
210        );
211        zip.add_file("word/styles.xml", styles_xml.as_bytes(), false);
212        zip.add_file("word/numbering.xml", numbering_xml.as_bytes(), false);
213        zip.add_file("docProps/core.xml", core_xml.as_bytes(), false);
214
215        zip.finish().map_err(DocxError::BuildError)
216    }

218    fn collect_image_refs(&self, module: &SIRModuleV2) -> Vec<ImageRef> {
219        let mut refs = Vec::new();
220        let mut counter = 0u32;
221        for &root_id in module.body.roots() {
222            if let Some(root) = module.body.get(root_id) {
223                self.collect_images_walk(module, root, &mut refs, &mut counter);
224            }
225        }
226        refs
227    }

229    fn collect_images_walk(
230        &self,
231        module: &SIRModuleV2,
232        node: &Node,
233        out: &mut Vec<ImageRef>,
234        counter: &mut u32,
235    ) {
236        if let NodeType::Image { source, alt, .. } = &node.node_type {
237            *counter += 1;
238            let ext = guess_image_extension(source);
239            let content_type = extension_to_mime(&ext);
240            let r_id = format!("rId{}", *counter + 2);
241            let zip_path = format!("word/media/image{}.{}", *counter, ext);
242
243            out.push(ImageRef {
244                r_id,
245                source: source.clone(),
246                content_type,
247                alt_text: alt.clone(),
248                zip_path,
249                node_id: node.id,
250            });
251        }
252
253        for &child_id in &node.child_ids {
254            if let Some(child) = module.body.get(child_id) {
255                self.collect_images_walk(module, child, out, counter);
256            }
257        }
258    }

261        &self,
262        module: &SIRModuleV2,
263        image_map: &HashMap<u32, &ImageRef>,
264        notes: &mut CollectedNotes,
265    ) -> String {
266        let mut body = String::new();

270            }
271        }
272
273        format!(
274            r#"<?xml version="1.0" encoding="UTF-8" standalone="yes"?>

281{body}
282</w:body>
283</w:document>"#
284        )

287    fn render_node_with_images_and_notes(
288        &self,
289        out: &mut String,
290        module: &SIRModuleV2,
291        node: &Node,
292        image_map: &HashMap<u32, &ImageRef>,
293        notes: &mut CollectedNotes,
294    ) {
295        match &node.node_type {
296            NodeType::Document => {
297                for &child_id in &node.child_ids {
298                    if let Some(child) = module.body.get(child_id) {
299                        self.render_node_with_images_and_notes(
300                            out, module, child, image_map, notes,
301                        );
302                    }
303                }
304            }
305
306            NodeType::Part => {
307                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Title\"/></w:pPr>");
308                for &child_id in &node.child_ids {
309                    if let Some(child) = module.body.get(child_id) {
310                        self.render_inline_with_images_and_notes(
311                            out, module, child, image_map, notes,
312                        );
313                    }
314                }
315                out.push_str("</w:p>");
316            }
317
318            NodeType::Chapter
319            | NodeType::Section
320            | NodeType::Subsection
321            | NodeType::Subsubsection => {
322                let style = match &node.node_type {
323                    NodeType::Chapter => "Heading1",
324                    NodeType::Section => "Heading2",
325                    NodeType::Subsection => "Heading3",
326                    NodeType::Subsubsection => "Heading4",
327                    _ => "Heading2",
328                };
329                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"");
330                out.push_str(style);
331                out.push_str("\"/></w:pPr>");
332                for &child_id in &node.child_ids {
333                    if let Some(child) = module.body.get(child_id) {
334                        self.render_inline_with_images_and_notes(
335                            out, module, child, image_map, notes,
336                        );
337                    }
338                }
339                out.push_str("</w:p>");
340            }
341
342            NodeType::Paragraph => {
343                out.push_str("<w:p>");
344                for &child_id in &node.child_ids {
345                    if let Some(child) = module.body.get(child_id) {
346                        self.render_inline_with_images_and_notes(
347                            out, module, child, image_map, notes,
348                        );
349                    }
350                }
351                out.push_str("</w:p>");
352            }
353
354            NodeType::List { ordered, .. } => {
355                for &child_id in &node.child_ids {
356                    if let Some(child) = module.body.get(child_id) {
357                        self.render_list_item_with_notes(out, module, child, *ordered, notes);
358                    }
359                }
360            }
361
362            NodeType::ListItem => {
363                self.render_list_item_with_notes(out, module, node, false, notes);
364            }
365
366            NodeType::BlockQuote => {
367                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Quote\"/></w:pPr>");
368                for &child_id in &node.child_ids {
369                    if let Some(child) = module.body.get(child_id) {
370                        self.render_inline_with_images_and_notes(
371                            out, module, child, image_map, notes,
372                        );
373                    }
374                }
375                out.push_str("</w:p>");
376            }
377
378            NodeType::CodeBlock { language, .. } => {
379                out.push_str("<w:p><w:pPr><w:shd w:val=\"clear\" w:color=\"auto\" w:fill=\"F5F5F5\"/></w:pPr>");
380                if let Some(lang) = language {
381                    out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/><w:color w:val=\"606060\"/></w:rPr><w:t xml:space=\"preserve\">");
382                    out.push_str(&format!("[{}]\n", lang));
383                    out.push_str("</w:t></w:r>");
384                }
385                let text = module.body.collect_text(node.id);
386                out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/></w:rPr><w:t xml:space=\"preserve\">");
387                out.push_str(&escape_xml(&text));
388                out.push_str("</w:t></w:r></w:p>");
389            }
390
391            NodeType::MathBlock { .. } => {
392                let text = module.body.collect_text(node.id);
393                out.push_str("<w:p><w:r><w:rPr><w:i/><w:color w:val=\"0000CC\"/></w:rPr><w:t xml:space=\"preserve\">");
394                out.push_str(&escape_xml(&text));
395                out.push_str("</w:t></w:r></w:p>");
396            }
397
398            NodeType::Table { .. } => {
399                out.push_str("<w:tbl><w:tblPr><w:tblBorders>");
400                out.push_str(
401                    "<w:top w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
402                );
403                out.push_str(
404                    "<w:left w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
405                );
406                out.push_str(
407                    "<w:bottom w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
408                );
409                out.push_str(
410                    "<w:right w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
411                );
412                out.push_str(
413                    "<w:insideH w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
414                );
415                out.push_str(
416                    "<w:insideV w:val=\"single\" w:sz=\"4\" w:space=\"0\" w:color=\"000000\"/>",
417                );
418                out.push_str("</w:tblBorders></w:tblPr>");
419                for &child_id in &node.child_ids {
420                    if let Some(child) = module.body.get(child_id) {
421                        self.render_table_row(out, module, child);
422                    }
423                }
424                out.push_str("</w:tbl>");
425            }
426
427            NodeType::ThematicBreak => {
428                out.push_str("<w:p><w:pPr><w:pBdr><w:bottom w:val=\"single\" w:sz=\"6\" w:space=\"1\" w:color=\"999999\"/></w:pBdr></w:pPr></w:p>");
429            }
430
431            NodeType::FootnoteBlock => {
432                for node in module.body.iter() {
433                    if let NodeType::Footnote { content } = &node.node_type {
434                        out.push_str("<w:p><w:r><w:rPr><w:sz w:val=\"16\"/></w:rPr><w:t>");
435                        out.push_str(&escape_xml(content));
436                        out.push_str("</w:t></w:r></w:p>");
437                    }
438                }
439            }
440
441            NodeType::Endnote { content } => {
442                let id = notes.endnotes.len() as u32 + 1;
443                notes.endnotes.push(EndnoteEntry {
444                    id,
445                    content: content.clone(),
446                });
447                out.push_str(&format!("<w:r><w:endnoteReference w:id=\"{}\"/></w:r>", id));
448            }
449
450            NodeType::Figure { .. } => {
451                out.push_str("<w:p>");
452                for &child_id in &node.child_ids {
453                    if let Some(child) = module.body.get(child_id) {
454                        if let NodeType::Image { .. } = &child.node_type {
455                            if let Some(img_ref) = image_map.get(&child.id) {
456                                emit_inline_image(out, img_ref);
457                            } else if let NodeType::Image { alt, .. } = &child.node_type {
458                                out.push_str("<w:r><w:t>");
459                                out.push_str(&escape_xml(alt));
460                                out.push_str(" [image not embedded]</w:t></w:r>");
461                            }
462                        } else {
463                            self.render_node_with_images_and_notes(
464                                out, module, child, image_map, notes,
465                            );
466                        }
467                    }
468                }
469                out.push_str("</w:p>");
470            }
471
472            NodeType::Caption => {
473                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"Caption\"/></w:pPr>");
474                for &child_id in &node.child_ids {
475                    if let Some(child) = module.body.get(child_id) {
476                        self.render_inline_with_images_and_notes(
477                            out, module, child, image_map, notes,
478                        );
479                    }
480                }
481                out.push_str("</w:p>");
482            }
483
484            NodeType::PageBreak => {
485                out.push_str("<w:p><w:r><w:br w:type=\"page\"/></w:r></w:p>");
486            }
487
488            NodeType::TableOfContents { .. } => {
489                out.push_str("<w:p><w:pPr><w:pStyle w:val=\"TOCHeading\"/></w:pPr><w:r><w:t>Table of Contents</w:t></w:r></w:p>");
490                out.push_str("<w:p><w:r><w:fldChar w:fldCharType=\"begin\"/></w:r><w:r><w:instrText xml:space=\"preserve\"> TOC \\o \"1-3\" \\h \\z \\u </w:instrText><w:r><w:fldChar w:fldCharType=\"separate\"/></w:r><w:r><w:t>[Update this field in Word to generate TOC]</w:t></w:r><w:r><w:fldChar w:fldCharType=\"end\"/></w:r></w:p>");
491            }
492
493            NodeType::Citation { keys, .. } => {
494                out.push_str("<w:r><w:rPr><w:vertAlign w:val=\"superscript\"/><w:sz w:val=\"16\"/></w:rPr><w:t>");
495                out.push_str(&escape_xml(&keys.join(", ")));
496                out.push_str("</w:t></w:r>");
497            }
498
499            NodeType::Reference { label } => {
500                out.push_str("<w:r><w:rPr><w:color w:val=\"0066CC\"/></w:rPr><w:t>");
501                out.push_str(&escape_xml(&format!("[ref: {}]", label)));
502                out.push_str("</w:t></w:r>");
503            }
504
505            _ => {
506                for &child_id in &node.child_ids {
507                    if let Some(child) = module.body.get(child_id) {
508                        self.render_node_with_images_and_notes(
509                            out, module, child, image_map, notes,
510                        );
511                    }
512                }
513            }
514        }
515    }

542            }
543
544            NodeType::Italic => {
545                out.push_str("<w:r><w:rPr><w:i/></w:rPr>");
546                for &child_id in &node.child_ids {
547                    if let Some(child) = module.body.get(child_id) {
548                        self.render_inline_with_images_and_notes(
549                            out, module, child, image_map, notes,
550                        );
551                    }
552                }
553                out.push_str("</w:r>");
554            }
555
556            NodeType::Mono => {
557                out.push_str("<w:r><w:rPr><w:rFonts w:ascii=\"Courier New\" w:hAnsi=\"Courier New\"/></w:rPr>");
558                for &child_id in &node.child_ids {
559                    if let Some(child) = module.body.get(child_id) {
560                        self.render_inline_with_images_and_notes(
561                            out, module, child, image_map, notes,
562                        );
563                    }
564                }
565                out.push_str("</w:r>");
566            }
567
568            NodeType::Underline => {
569                out.push_str("<w:r><w:rPr><w:u w:val=\"single\"/></w:rPr>");
570                for &child_id in &node.child_ids {
571                    if let Some(child) = module.body.get(child_id) {
572                        self.render_inline_with_images_and_notes(
573                            out, module, child, image_map, notes,
574                        );
575                    }
576                }
577                out.push_str("</w:r>");
578            }

1016    fn render_table_row(&self, out: &mut String, module: &SIRModuleV2, node: &Node) {
1017        let is_header = matches!(&node.node_type, NodeType::TableRow { is_header: true });
1018        out.push_str("<w:tr>");
1019        if is_header {
1020            out.push_str("<w:trPr><w:tblHeader/></w:trPr>");
1021        }
1022        for &child_id in &node.child_ids {
1023            if let Some(child) = module.body.get(child_id)
1024                && matches!(&child.node_type, NodeType::TableCell { .. })
1025            {
1026                out.push_str("<w:tc><w:p>");
1027                if is_header {
1028                    out.push_str("<w:pPr><w:jc w:val=\"center\"/></w:pPr>");
1029                }
1030                for &cell_child_id in &child.child_ids {
1031                    if let Some(cell_child) = module.body.get(cell_child_id) {
1032                        if is_header {
1033                            out.push_str("<w:r><w:rPr><w:b/></w:rPr>");
1034                            self.render_inline(out, module, cell_child);
1035                            out.push_str("</w:r>");
1036                        } else {
1037                            self.render_inline(out, module, cell_child);
1038                        }
1039                    }
1040                }
1041                out.push_str("</w:p></w:tc>");
1042            }
1043        }
1044        out.push_str("</w:tr>");

1332fn format_iso8601(secs: u64) -> String {
1333    let secs = secs as i64;
1334    let year = 1970 + secs / 31_536_000;
1335    let rem = secs % 31_536_000;
1336    let month = 1 + rem / 2_592_000;
1337    let rem = rem % 2_592_000;
1338    let day = 1 + rem / 86_400;
1339    format!("{year:04}-{month:02}-{day:02}T00:00:00Z")
1340}

1346impl SimpleZip {
1347    fn new() -> Self {
1348        Self { files: Vec::new() }
1349    }
1350
1351    fn add_file(&mut self, path: &str, data: &[u8], stored: bool) {
1352        self.files.push((path.to_string(), data.to_vec(), stored));
1353    }
1354
1355    fn finish(&self) -> Result<Vec<u8>, String> {
1356        let mut buf = Vec::new();
1357        let mut central_offsets: Vec<(u64, u32, u32, String)> = Vec::new();
1358
1359        for (path, data, stored) in &self.files {
1360            let offset = buf.len() as u64;
1361            let crc = crc32(data);
1362            let size = data.len() as u32;
1363
1364            buf.extend_from_slice(&[0x50, 0x4B, 0x03, 0x04]);
1365            buf.extend_from_slice(&[20, 0]);
1366            buf.extend_from_slice(&[0, 0]);
1367            buf.extend_from_slice(if *stored { &[0, 0] } else { &[8, 0] });
1368            buf.extend_from_slice(&[0, 0]);
1369            buf.extend_from_slice(&[0, 0]);
1370            buf.extend_from_slice(&crc.to_le_bytes());
1371            buf.extend_from_slice(&size.to_le_bytes());
1372            buf.extend_from_slice(&size.to_le_bytes());
1373            let name_bytes = path.as_bytes();
1374            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
1375            buf.extend_from_slice(&[0u8; 10]);
1376            buf.extend_from_slice(name_bytes);
1377            buf.extend_from_slice(data);
1378
1379            central_offsets.push((offset, size, crc, path.clone()));
1380        }
1381
1382        let central_start = buf.len() as u64;
1383
1384        for (offset, size, crc, name) in &central_offsets {
1385            buf.extend_from_slice(&[0x50, 0x4B, 0x01, 0x02]);
1386            buf.extend_from_slice(&[20, 0]);
1387            buf.extend_from_slice(&[20, 0]);
1388            buf.extend_from_slice(&[0, 0]);
1389            buf.extend_from_slice(&[0, 0]);
1390            buf.extend_from_slice(&[0, 0]);
1391            buf.extend_from_slice(&[0, 0]);
1392            buf.extend_from_slice(&crc.to_le_bytes());
1393            buf.extend_from_slice(&size.to_le_bytes());
1394            buf.extend_from_slice(&size.to_le_bytes());
1395            let name_bytes = name.as_bytes();
1396            buf.extend_from_slice(&(name_bytes.len() as u16).to_le_bytes());
1397            buf.extend_from_slice(&[0u8; 10]);
1398            buf.extend_from_slice(&[0, 0, 0]);
1399            buf.extend_from_slice(&(*offset as u32).to_le_bytes());
1400            buf.extend_from_slice(name_bytes);
1401        }
1402
1403        let central_end = buf.len() as u64;
1404        let central_count = central_offsets.len() as u16;
1405
1406        buf.extend_from_slice(&[0x50, 0x4B, 0x05, 0x06]);
1407        buf.extend_from_slice(&[0, 0]);
1408        buf.extend_from_slice(&[0, 0]);
1409        buf.extend_from_slice(&central_count.to_le_bytes());
1410        buf.extend_from_slice(&central_count.to_le_bytes());
1411        buf.extend_from_slice(&(central_start as u32).to_le_bytes());
1412        buf.extend_from_slice(&(central_end as u32).to_le_bytes());
1413        buf.extend_from_slice(&[0u8; 2]);
1414
1415        Ok(buf)
1416    }

1419fn crc32(data: &[u8]) -> u32 {
1420    let mut crc: u32 = 0xFFFFFFFF;
1421    for &byte in data {
1422        crc ^= byte as u32;
1423        for _ in 0..8 {
1424            if crc & 1 != 0 {
1425                crc = (crc >> 1) ^ 0xEDB88320;
1426            } else {
1427                crc >>= 1;
1428            }
1429        }
1430    }
1431    !crc
1432}
