6/// Logical structure element types for PDF tagged (accessible) structure trees.
7pub enum StructureType {
8    /// The document root.
9    Document,
10    /// A part (grouping) element.
11    Part,
12    /// A chapter grouping (custom role).
13    Chapter,
14    /// A section grouping.
15    Section,
16    /// A subsection grouping (custom role).
17    Subsection,
18    /// A level-1 heading.
19    H1,
20    /// A level-2 heading.
21    H2,
22    /// A level-3 heading.
23    H3,
24    /// A level-4 heading.
25    H4,
26    /// A level-5 heading.
27    H5,
28    /// A level-6 heading.
29    H6,
30    /// A paragraph.
31    Paragraph,
32    /// A list container.
33    List,
34    /// A single list entry.
35    ListItem,
36    /// The label (bullet or number) of a list entry.
37    ListLabel,
38    /// The content of a list entry.
39    ListBody,
40    /// A table.
41    Table,
42    /// A table header row group.
43    TableHeader,
44    /// A table body row group.
45    TableBody,
46    /// A table row.
47    TableRow,
48    /// A header cell.
49    TableHeaderCell,
50    /// A data cell.
51    TableDataCell,
52    /// A figure.
53    Figure,
54    /// A caption.
55    Caption,
56    /// A code block (custom role).
57    CodeBlock,
58    /// A block quotation.
59    BlockQuote,
60    /// A display math block (custom role).
61    MathBlock,
62    /// A footnote.
63    Footnote,
64    /// An in-text footnote reference.
65    FootnoteRef,
66    /// The body of a footnote (custom role).
67    FootnoteBody,
68    /// A table of contents.
69    TOC,
70    /// A thematic break (custom role, non-structural).
71    ThematicBreak,
72    /// An inline span.
73    Span,
74    /// Decorative content excluded from reading order.
75    Artifact,
76}

79/// A heading level 1-6, clamped on construction.
80pub struct HeadingLevel(pub u8);
81
82impl HeadingLevel {
83    /// Creates a heading level, clamping to 1-6.
84    pub fn new(level: u8) -> Self {
85        Self(level.clamp(1, 6))
86    }
87
88    /// The corresponding [`StructureType`] heading variant.
89    pub fn to_structure_type(self) -> StructureType {
90        match self.0 {
91            1 => StructureType::H1,
92            2 => StructureType::H2,
93            3 => StructureType::H3,
94            4 => StructureType::H4,
95            5 => StructureType::H5,
96            _ => StructureType::H6,
97        }
98    }
99
100    /// The `pdf-writer` [`StructRole`] for this heading level.
101    pub fn to_struct_role(self) -> StructRole {
102        match self.0 {
103            1 => StructRole::H1,
104            2 => StructRole::H2,
105            3 => StructRole::H3,
106            4 => StructRole::H4,
107            5 => StructRole::H5,
108            _ => StructRole::H6,
109        }
110    }
111}

113impl StructureType {
114    /// Maps to the standard PDF [`StructRole`], using `NonStruct` when none exists.
115    pub fn to_struct_role(self) -> StructRole {
116        match self {
117            Self::Document => StructRole::Document,
118            Self::Part => StructRole::Part,
119            Self::Chapter => StructRole::Sect,
120            Self::Section => StructRole::Sect,
121            Self::Subsection => StructRole::Sect,
122            Self::H1 => StructRole::H1,
123            Self::H2 => StructRole::H2,
124            Self::H3 => StructRole::H3,
125            Self::H4 => StructRole::H4,
126            Self::H5 => StructRole::H5,
127            Self::H6 => StructRole::H6,
128            Self::Paragraph => StructRole::P,
129            Self::List => StructRole::L,
130            Self::ListItem => StructRole::LI,
131            Self::ListLabel => StructRole::Lbl,
132            Self::ListBody => StructRole::LBody,
133            Self::Table => StructRole::Table,
134            Self::TableHeader => StructRole::THead,
135            Self::TableBody => StructRole::TBody,
136            Self::TableRow => StructRole::TR,
137            Self::TableHeaderCell => StructRole::TH,
138            Self::TableDataCell => StructRole::TD,
139            Self::Figure => StructRole::Figure,
140            Self::Caption => StructRole::Caption,
141            Self::CodeBlock => StructRole::Code,
142            Self::BlockQuote => StructRole::BlockQuote,
143            Self::MathBlock => StructRole::Formula,
144            Self::Footnote => StructRole::Note,
145            Self::FootnoteRef => StructRole::Reference,
146            Self::FootnoteBody => StructRole::Note,
147            Self::TOC => StructRole::TOC,
148            Self::ThematicBreak => StructRole::NonStruct,
149            Self::Span => StructRole::Span,
150            Self::Artifact => StructRole::NonStruct,
151        }
152    }
153
154    /// Custom role name for types with no exact standard counterpart (`Chapter`, `CodeBlock`, ...).
155    pub fn custom_role_name(self) -> Option<&'static [u8]> {
156        match self {
157            Self::Chapter => Some(b"Chapter"),
158            Self::Subsection => Some(b"Subsection"),
159            Self::CodeBlock => Some(b"CodeBlock"),
160            Self::MathBlock => Some(b"MathBlock"),
161            Self::FootnoteBody => Some(b"FootnoteBody"),
162            Self::ThematicBreak => Some(b"ThematicBreak"),
163            Self::Artifact => Some(b"Artifact"),
164            _ => None,
165        }
166    }
167
168    /// Whether this is one of the `H1`-`H6` heading types.
169    pub fn is_heading(self) -> bool {
170        matches!(
171            self,
172            Self::H1 | Self::H2 | Self::H3 | Self::H4 | Self::H5 | Self::H6
173        )
174    }
175
176    /// Whether this is an artifact (excluded from reading order).
177    pub fn is_artifact(self) -> bool {
178        self == Self::Artifact
179    }

216#[derive(Debug, Clone)]
217/// A node in the tagged-PDF structure tree.
218pub struct StructureNode {
219    /// Logical type of the element.
220    pub element_type: StructureType,
221    /// Child nodes.
222    pub children: Vec<StructureNode>,
223    /// Alternative description (e.g., for figures).
224    pub alt_text: Option<String>,
225    /// Exact text replacing descendants during extraction.
226    pub actual_text: Option<String>,
227    /// Expanded form of an acronym or abbreviation.
228    pub expanded_text: Option<String>,
229    /// Language override for the whole element (BCP 47).
230    pub language: Option<String>,
231    /// Sub-runs with their own language.
232    pub language_spans: Vec<LanguageSpan>,
233    /// Page the element is marked on.
234    pub page: u32,
235    /// Marked-content ID within the page content stream.
236    pub mcid: u32,
237    /// Position in logical reading order among leaves.
238    pub reading_order: u32,
239    /// Bounding box on the page, if known.
240    pub bbox: Option<BBox>,
241}
242
243impl StructureNode {
244    /// Creates a leaf node of the given type on a page with an MCID.
245    pub fn new(element_type: StructureType, page: u32, mcid: u32) -> Self {
246        Self {
247            element_type,
248            children: Vec::new(),
249            alt_text: None,
250            actual_text: None,
251            expanded_text: None,
252            language: None,
253            language_spans: Vec::new(),
254            page,
255            mcid,
256            reading_order: 0,
257            bbox: None,
258        }
259    }
260
261    /// Creates a container node with children (page and MCID unset).
262    pub fn with_children(element_type: StructureType, children: Vec<StructureNode>) -> Self {
263        Self {
264            element_type,
265            children,
266            alt_text: None,
267            actual_text: None,
268            expanded_text: None,
269            language: None,
270            language_spans: Vec::new(),
271            page: 0,
272            mcid: 0,
273            reading_order: 0,
274            bbox: None,
275        }
276    }
277
278    /// Builder: sets the alternative description.
279    pub fn with_alt_text(mut self, text: impl Into<String>) -> Self {
280        self.alt_text = Some(text.into());
281        self
282    }
283
284    /// Builder: sets the exact text.
285    pub fn with_actual_text(mut self, text: impl Into<String>) -> Self {
286        self.actual_text = Some(text.into());
287        self
288    }
289
290    /// Builder: sets the expanded form.
291    pub fn with_expanded_text(mut self, text: impl Into<String>) -> Self {
292        self.expanded_text = Some(text.into());
293        self
294    }
295
296    /// Builder: sets the element language.
297    pub fn with_language(mut self, lang: impl Into<String>) -> Self {
298        self.language = Some(lang.into());
299        self
300    }
301
302    /// Builder: appends a language-specific sub-run.
303    pub fn with_language_span(mut self, text: impl Into<String>, lang: impl Into<String>) -> Self {
304        self.language_spans.push(LanguageSpan {
305            text: text.into(),
306            lang: lang.into(),
307        });
308        self
309    }
310
311    /// Builder: sets the bounding box.
312    pub fn with_bbox(mut self, bbox: BBox) -> Self {
313        self.bbox = Some(bbox);
314        self
315    }
316
317    /// Builder: sets the reading order explicitly.
318    pub fn with_reading_order(mut self, order: u32) -> Self {
319        self.reading_order = order;
320        self
321    }
322
323    /// Whether the node has no children.
324    pub fn is_leaf(&self) -> bool {
325        self.children.is_empty()
326    }
327
328    /// Numbers leaf nodes in document order, skipping artifacts; returns the number of leaves.
329    pub fn assign_reading_order(&mut self) -> u32 {
330        let mut counter = 0u32;
331        self.assign_reading_order_inner(&mut counter);
332        counter
333    }
334
335    fn assign_reading_order_inner(&mut self, counter: &mut u32) {
336        if self.element_type.is_artifact() {
337            return;
338        }
339        if self.is_leaf() {
340            self.reading_order = *counter;
341            *counter += 1;
342        }
343        for child in &mut self.children {
344            child.assign_reading_order_inner(counter);
345        }
346    }
347
348    /// Concatenates the subtree text, preferring `actual_text` and language spans.
349    pub fn collect_text(&self) -> String {
350        let mut parts = Vec::new();
351        self.collect_text_inner(&mut parts);
352        parts.join("")
353    }
354
355    fn collect_text_inner(&self, parts: &mut Vec<String>) {
356        if let Some(ref text) = self.actual_text
357            && !text.is_empty()
358        {
359            parts.push(text.clone());
360            return;
361        }
362        if !self.language_spans.is_empty() {
363            for span in &self.language_spans {
364                parts.push(span.text.clone());
365            }
366            return;
367        }
368        if !self.children.is_empty() {
369            for child in &self.children {
370                child.collect_text_inner(parts);
371            }
372        }
373    }
374}

406/// Builds a table row from `(cell type, text)` pairs with sequential MCIDs.
407pub fn table_row(cells: Vec<(StructureType, &str)>, page: u32, mcid_start: u32) -> StructureNode {
408    let children: Vec<StructureNode> = cells
409        .into_iter()
410        .enumerate()
411        .map(|(i, (cell_type, text))| {
412            StructureNode::new(cell_type, page, mcid_start + i as u32)
413                .with_actual_text(text.to_string())
414        })
415        .collect();
416    StructureNode::with_children(StructureType::TableRow, children)
417}

419/// Builds a full table with a header row group and body rows.
420pub fn table_with_header(
421    headers: Vec<&str>,
422    rows: Vec<Vec<&str>>,
423    page: u32,
424    mcid_start: u32,
425) -> StructureNode {
426    let mut mcid = mcid_start;
427    let mut children = Vec::new();
428
429    let header_row: Vec<StructureNode> = headers
430        .iter()
431        .enumerate()
432        .map(|(i, text)| {
433            StructureNode::new(StructureType::TableHeaderCell, page, mcid + i as u32)
434                .with_actual_text((*text).to_string())
435        })
436        .collect();
437    mcid += headers.len() as u32;
438
439    let header_group = StructureNode::with_children(
440        StructureType::TableHeader,
441        vec![StructureNode::with_children(
442            StructureType::TableRow,
443            header_row,
444        )],
445    );
446    children.push(header_group);
447
448    let mut body_rows = Vec::new();
449    for row_data in &rows {
450        let row_cells: Vec<StructureNode> = row_data
451            .iter()
452            .enumerate()
453            .map(|(i, text)| {
454                StructureNode::new(StructureType::TableDataCell, page, mcid + i as u32)
455                    .with_actual_text((*text).to_string())
456            })
457            .collect();
458        mcid += row_data.len() as u32;
459        body_rows.push(StructureNode::with_children(
460            StructureType::TableRow,
461            row_cells,
462        ));
463    }
464    children.push(StructureNode::with_children(
465        StructureType::TableBody,
466        body_rows,
467    ));
468
469    StructureNode::with_children(StructureType::Table, children)
470}
