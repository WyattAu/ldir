3pub fn generate_pdfa_xmp(conformance: PdfConformance, title: &str, author: &str) -> Vec<u8> {
4    let part = conformance.pdfaid_part();
5    let conf = conformance.pdfaid_conformance();
6
7    let mut xmp = format!(
8        r#"<?xpacket begin="" id="W5M0MpCehiHzreSzNTczkc9d"?>
9<x:xmpmeta xmlns:x="adobe:ns:meta/">
10  <rdf:RDF xmlns:rdf="http://www.w3.org/1999/02/22-rdf-syntax-ns#"
11           xmlns:dc="http://purl.org/dc/elements/1.1/"
12           xmlns:xmp="http://ns.adobe.com/xap/1.0/"
13           xmlns:pdfaid="http://www.aiim.org/pdfa/ns/id/">
14    <rdf:Description rdf:about=""
15      xmlns:pdfaid="http://www.aiim.org/pdfa/ns/id/">
16      <pdfaid:part>{part}</pdfaid:part>"#
17    );
18
19    if !conf.is_empty() {
20        xmp.push_str(&format!(
21            r#"
22      <pdfaid:conformance>{conf}</pdfaid:conformance>"#
23        ));
24    }
25
26    xmp.push_str(
27        r#"
28    </rdf:Description>
29    <rdf:Description rdf:about=""
30      xmlns:dc="http://purl.org/dc/elements/1.1/">
31      <dc:title>
32        <rdf:Alt>
33          <rdf:li xml:lang="x-default">"#,
34    );
35    xmp.push_str(&xml_escape(title));
36    xmp.push_str(
37        r#"</rdf:li>
38        </rdf:Alt>
39      </dc:title>
40      <dc:creator>
41        <rdf:Seq>
42          <rdf:li>"#,
43    );
44    xmp.push_str(&xml_escape(author));
45    xmp.push_str(
46        r#"</rdf:li>
47        </rdf:Seq>
48      </dc:creator>
49    </rdf:Description>
50    <rdf:Description rdf:about=""
51      xmlns:xmp="http://ns.adobe.com/xap/1.0/">
52      <xmp:CreatorTool>ldir</xmp:CreatorTool>
53      <xmp:CreateDate>"#,
54    );
55    xmp.push_str(&iso8601_now());
56    xmp.push_str(
57        r#"</xmp:CreateDate>
58      <xmp:ModifyDate>"#,
59    );
60    xmp.push_str(&iso8601_now());
61    xmp.push_str(
62        r#"</xmp:ModifyDate>
63    </rdf:Description>
64  </rdf:RDF>
65</x:xmpmeta>
66<?xpacket end="w"?>"#,
67    );
68
69    xmp.into_bytes()
70}

72fn xml_escape(s: &str) -> String {
73    let mut out = String::with_capacity(s.len());
74    for ch in s.chars() {
75        match ch {
76            '&' => out.push_str("&amp;"),
77            '<' => out.push_str("&lt;"),
78            '>' => out.push_str("&gt;"),
79            '"' => out.push_str("&quot;"),
80            '\'' => out.push_str("&apos;"),
81            c => out.push(c),
82        }
83    }
84    out
85}

87fn iso8601_now() -> String {
88    let epoch = std::time::SystemTime::UNIX_EPOCH;
89    std::time::SystemTime::now()
90        .duration_since(epoch)
91        .map(|d| {
92            let secs = d.as_secs();
93            let days = secs / 86400;
94            let z = (days + 719528) as i64;
95            let era = if z >= 0 {
96                z / 146097
97            } else {
98                (z - 146096) / 146097
99            };
100            let doe = z - era * 146097;
101            let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
102            let y = yoe + era * 400;
103            let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
104            let mp = (5 * doy + 2) / 153;
105            let d = doy - (153 * mp + 2) / 5 + 1;
106            let m = if mp < 10 { mp + 3 } else { mp - 9 };
107            let y = if m <= 2 { y + 1 } else { y };
108            format!("{:04}-{:02}-{:02}T00:00:00Z", y, m, d)
109        })
110        .unwrap_or_else(|_| "1970-01-01T00:00:00Z".to_string())
111}
